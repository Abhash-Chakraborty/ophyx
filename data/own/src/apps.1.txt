=== inventory-restock [apps]
--- task
Write a class `Inventory` with `add(sku, qty)`, `remove(sku, qty)` (ValueError if not enough stock, leaving stock unchanged), `set_reorder(sku, level, amount)` and `restock_report()` returning a list of (sku, on_hand, order_amount) for every sku at or below its reorder level, sorted by sku. Quantities must be positive integers (ValueError otherwise). Unknown skus have 0 on hand.
--- solution
class Inventory:
    def __init__(self):
        self._stock = {}
        self._reorder = {}

    @staticmethod
    def _check(qty):
        if not isinstance(qty, int) or isinstance(qty, bool) or qty <= 0:
            raise ValueError("quantity must be a positive integer")

    def on_hand(self, sku):
        return self._stock.get(sku, 0)

    def add(self, sku, qty):
        self._check(qty)
        self._stock[sku] = self.on_hand(sku) + qty

    def remove(self, sku, qty):
        self._check(qty)
        if self.on_hand(sku) < qty:
            raise ValueError(f"not enough {sku}: have {self.on_hand(sku)}, need {qty}")
        self._stock[sku] -= qty

    def set_reorder(self, sku, level, amount):
        self._check(amount)
        self._reorder[sku] = (level, amount)

    def restock_report(self):
        return sorted((sku, self.on_hand(sku), amount) for sku, (level, amount) in self._reorder.items()
                      if self.on_hand(sku) <= level)

=== coupon-cart [apps]
--- task
Write `checkout(items, coupons=())` where items are (name, unit_price, qty) and coupons are strings: "PCT10" (10% off the subtotal, any number of percent), "OFF5" (5 off, after percentage coupons, never below zero) and "BOGO:<name>" (every second unit of that item free). Apply BOGO first, then percentages (only the largest one counts), then fixed amounts. Return a dict {"subtotal", "discount", "total"} as Decimals rounded to cents. Unknown coupons raise ValueError.
--- solution
import re
from decimal import ROUND_HALF_UP, Decimal

CENT = Decimal("0.01")


def checkout(items, coupons=()):
    subtotal = sum(Decimal(str(price)) * qty for _, price, qty in items)
    bogo, pct, off = set(), Decimal(0), Decimal(0)
    for c in coupons:
        if m := re.fullmatch(r"PCT(\d+)", c):
            pct = max(pct, Decimal(m.group(1)))
        elif m := re.fullmatch(r"OFF(\d+(?:\.\d+)?)", c):
            off += Decimal(m.group(1))
        elif c.startswith("BOGO:"):
            bogo.add(c[5:])
        else:
            raise ValueError(f"unknown coupon {c!r}")
    after = subtotal
    for name, price, qty in items:
        if name in bogo:
            after -= Decimal(str(price)) * (qty // 2)
    after -= after * pct / 100
    after = max(Decimal(0), after - off)
    total = after.quantize(CENT, rounding=ROUND_HALF_UP)
    subtotal = subtotal.quantize(CENT, rounding=ROUND_HALF_UP)
    return {"subtotal": subtotal, "discount": subtotal - total, "total": total}

=== grade-book [apps]
--- task
Write `final_grades(scores, weights)` where scores maps student -> {category: list of scores out of 100} and weights maps category -> weight (weights must sum to 1, else ValueError). Each category average drops the lowest score when there are at least 3. A missing category counts as 0. Return student -> (percentage rounded to 1 place, letter) with A >= 90, B >= 80, C >= 70, D >= 60, else F.
--- solution
import math


def _letter(p):
    for cut, letter in ((90, "A"), (80, "B"), (70, "C"), (60, "D")):
        if p >= cut:
            return letter
    return "F"


def final_grades(scores, weights):
    if not math.isclose(sum(weights.values()), 1.0, abs_tol=1e-9):
        raise ValueError("weights must sum to 1")
    out = {}
    for student, cats in scores.items():
        total = 0.0
        for cat, w in weights.items():
            xs = sorted(cats.get(cat, []))
            if len(xs) >= 3:
                xs = xs[1:]
            avg = sum(xs) / len(xs) if xs else 0.0
            total += w * avg
        pct = round(total, 1)
        out[student] = (pct, _letter(pct))
    return out

=== loan-schedule [apps]
--- task
Write `amortization(principal, annual_rate, months)` returning the monthly payment and a schedule list of dicts {"month", "payment", "interest", "principal", "balance"} as Decimals rounded to cents. The payment uses the standard annuity formula (zero rate means principal / months); the last payment is adjusted so the balance ends at exactly 0.00.
--- solution
from decimal import ROUND_HALF_UP, Decimal

CENT = Decimal("0.01")


def _c(x):
    return Decimal(x).quantize(CENT, rounding=ROUND_HALF_UP)


def amortization(principal, annual_rate, months):
    if months <= 0:
        raise ValueError("months must be positive")
    p = Decimal(str(principal))
    r = Decimal(str(annual_rate)) / 12
    if r == 0:
        payment = _c(p / months)
    else:
        payment = _c(p * r / (1 - (1 + r) ** -months))
    balance, rows = p, []
    for m in range(1, months + 1):
        interest = _c(balance * r)
        if m == months:
            pay = balance + interest
        else:
            pay = payment
        princ = pay - interest
        balance = balance - princ
        rows.append({"month": m, "payment": _c(pay), "interest": interest, "principal": _c(princ), "balance": _c(balance)})
    return payment, rows

=== vending-machine [apps]
--- task
Write a class `VendingMachine(prices, coins=(200, 100, 50, 20, 10, 5))` with prices in cents per product. `insert(coin)` accepts only known coins (ValueError otherwise) and adds to the credit; `select(product)` returns (product, change) where change is a list of coins (greedy, largest first) and resets the credit, or raises ValueError("insufficient credit") / KeyError for unknown products; `cancel()` returns the inserted credit as coins and resets it.
--- solution
class VendingMachine:
    def __init__(self, prices, coins=(200, 100, 50, 20, 10, 5)):
        self.prices = dict(prices)
        self.coins = sorted(coins, reverse=True)
        self.credit = 0

    def insert(self, coin):
        if coin not in self.coins:
            raise ValueError(f"coin {coin} not accepted")
        self.credit += coin
        return self.credit

    def _change(self, amount):
        out = []
        for c in self.coins:
            while amount >= c:
                out.append(c)
                amount -= c
        return out

    def select(self, product):
        price = self.prices[product]
        if self.credit < price:
            raise ValueError("insufficient credit")
        change = self._change(self.credit - price)
        self.credit = 0
        return product, change

    def cancel(self):
        coins = self._change(self.credit)
        self.credit = 0
        return coins

=== parking-fees [apps]
--- task
Write `parking_fee(entry, exit)` for datetime pairs: the first 30 minutes are free; after that each started hour costs 2.50, with a daily cap of 15.00 per calendar day of the stay (a stay over several days is charged per day, each day capped, the free 30 minutes only once at the start). An exit before entry raises ValueError. Return a Decimal.
--- solution
import datetime
import math
from decimal import Decimal

RATE, CAP = Decimal("2.50"), Decimal("15.00")


def parking_fee(entry, exit):
    if exit < entry:
        raise ValueError("exit before entry")
    start = entry + datetime.timedelta(minutes=30)
    if exit <= start:
        return Decimal("0.00")
    total, cur = Decimal("0.00"), start
    while cur < exit:
        midnight = datetime.datetime.combine(cur.date() + datetime.timedelta(days=1), datetime.time())
        end = min(exit, midnight)
        hours = math.ceil((end - cur).total_seconds() / 3600)
        total += min(CAP, RATE * hours)
        cur = end
    return total

=== library-loans [apps]
--- task
Write a class `Library` with `add_book(isbn, title, copies=1)`, `checkout(isbn, member, today)` (ValueError if no copy is free or the member already has 3 books or any overdue book), `return_book(isbn, member, today)` returning the late fee (0.25 per day after the 14-day loan period, as a float rounded to 2 places), and `overdue(today)` listing (member, isbn, days_late) sorted by days late descending then member. Dates are datetime.date.
--- solution
import datetime

LOAN_DAYS = 14


class Library:
    def __init__(self):
        self.books = {}
        self.loans = []  # (member, isbn, due)

    def add_book(self, isbn, title, copies=1):
        title0, n = self.books.get(isbn, (title, 0))
        self.books[isbn] = (title0, n + copies)

    def _out(self, isbn):
        return sum(1 for _, i, _ in self.loans if i == isbn)

    def checkout(self, isbn, member, today):
        if isbn not in self.books:
            raise KeyError(isbn)
        mine = [l for l in self.loans if l[0] == member]
        if len(mine) >= 3:
            raise ValueError("loan limit reached")
        if any(due < today for _, _, due in mine):
            raise ValueError("member has overdue books")
        if self._out(isbn) >= self.books[isbn][1]:
            raise ValueError("no copy available")
        due = today + datetime.timedelta(days=LOAN_DAYS)
        self.loans.append((member, isbn, due))
        return due

    def return_book(self, isbn, member, today):
        for k, (m, i, due) in enumerate(self.loans):
            if m == member and i == isbn:
                del self.loans[k]
                return round(max(0, (today - due).days) * 0.25, 2)
        raise ValueError("no such loan")

    def overdue(self, today):
        rows = [(m, i, (today - due).days) for m, i, due in self.loans if due < today]
        return sorted(rows, key=lambda r: (-r[2], r[0]))

=== event-calendar [apps]
--- task
Write a class `Calendar` storing events (title, start, end datetimes). `add(title, start, end)` raises ValueError if end <= start or the event overlaps an existing one (touching ends are fine) unless allow_overlap=True. `on(day)` returns titles of events touching that date, sorted by start. `next_free(after, minutes)` returns the earliest datetime at or after `after` where a slot of that length is free.
--- solution
import datetime


class Calendar:
    def __init__(self):
        self.events = []

    def add(self, title, start, end, allow_overlap=False):
        if end <= start:
            raise ValueError("end must be after start")
        if not allow_overlap:
            for t, s, e in self.events:
                if start < e and s < end:
                    raise ValueError(f"overlaps {t!r}")
        self.events.append((title, start, end))
        self.events.sort(key=lambda ev: ev[1])

    def on(self, day):
        lo = datetime.datetime.combine(day, datetime.time())
        hi = lo + datetime.timedelta(days=1)
        return [t for t, s, e in self.events if s < hi and e > lo]

    def next_free(self, after, minutes):
        need = datetime.timedelta(minutes=minutes)
        cur = after
        for _, s, e in self.events:
            if e <= cur:
                continue
            if s - cur >= need:
                return cur
            cur = max(cur, e)
        return cur

=== tic-tac-toe [apps]
--- task
Write `winner(board)` for a 3x3 tic-tac-toe board given as 3 strings of "X", "O" or "." and return "X", "O", "draw" (full board, no winner) or None (game still going). Raise ValueError for impossible boards: wrong size or characters, counts where O has more marks than X or X has more than one extra, or both players winning, or a winner whose mark count is inconsistent with having just moved.
--- solution
LINES = [[(r, c) for c in range(3)] for r in range(3)] + [[(r, c) for r in range(3)] for c in range(3)] + \
    [[(i, i) for i in range(3)], [(i, 2 - i) for i in range(3)]]


def winner(board):
    if len(board) != 3 or any(len(row) != 3 or set(row) - set("XO.") for row in board):
        raise ValueError("bad board")
    x = sum(row.count("X") for row in board)
    o = sum(row.count("O") for row in board)
    if not (x == o or x == o + 1):
        raise ValueError("impossible move counts")
    wins = {p for p in "XO" for line in LINES if all(board[r][c] == p for r, c in line)}
    if len(wins) == 2:
        raise ValueError("both players win")
    if "X" in wins and x != o + 1:
        raise ValueError("X won but O moved after")
    if "O" in wins and x != o:
        raise ValueError("O won but X moved after")
    if wins:
        return wins.pop()
    return "draw" if x + o == 9 else None

=== sudoku-check [apps]
--- task
Write `sudoku_errors(grid)` for a 9x9 grid (lists of ints, 0 = empty) returning a sorted list of (row, col) cells that conflict with another cell in the same row, column or 3x3 box (both cells of each conflict are listed). Write `is_solved(grid)`: True only for a full grid with no conflicts. A grid of the wrong shape or with values outside 0-9 raises ValueError.
--- solution
def _check(grid):
    if len(grid) != 9 or any(len(r) != 9 for r in grid):
        raise ValueError("grid must be 9x9")
    if any(not isinstance(v, int) or not 0 <= v <= 9 for r in grid for v in r):
        raise ValueError("values must be 0-9")


def sudoku_errors(grid):
    _check(grid)
    bad = set()
    groups = [[(r, c) for c in range(9)] for r in range(9)] + [[(r, c) for r in range(9)] for c in range(9)]
    groups += [[(br + r, bc + c) for r in range(3) for c in range(3)] for br in (0, 3, 6) for bc in (0, 3, 6)]
    for g in groups:
        seen = {}
        for r, c in g:
            v = grid[r][c]
            if v:
                seen.setdefault(v, []).append((r, c))
        for cells in seen.values():
            if len(cells) > 1:
                bad.update(cells)
    return sorted(bad)


def is_solved(grid):
    return not sudoku_errors(grid) and all(v for r in grid for v in r)

=== game-of-life [apps]
--- task
Write `step(live)` for Conway's Game of Life on an unbounded grid, where `live` is a set of (row, col) cells; return the next generation's set. Write `run(live, n)` applying n steps and `bounds(live)` returning (min_row, min_col, max_row, max_col) or None for an empty set.
--- solution
from collections import Counter


def step(live):
    counts = Counter((r + dr, c + dc) for r, c in live for dr in (-1, 0, 1) for dc in (-1, 0, 1) if dr or dc)
    return {cell for cell, n in counts.items() if n == 3 or (n == 2 and cell in live)}


def run(live, n):
    for _ in range(n):
        live = step(live)
    return live


def bounds(live):
    if not live:
        return None
    rows = [r for r, _ in live]
    cols = [c for _, c in live]
    return min(rows), min(cols), max(rows), max(cols)

=== maze-solver [apps]
--- task
Write `solve_maze(lines)` for a maze given as strings with "#" walls, "." open cells, one "S" start and one "E" exit. Return the shortest path as a list of (row, col) from S to E inclusive using up/down/left/right moves, or None if there is none. Write `draw_path(lines, path)` returning the maze with path cells (other than S and E) marked "*". Missing or repeated S or E raises ValueError.
--- solution
from collections import deque


def _find(lines, ch):
    hits = [(r, c) for r, line in enumerate(lines) for c, x in enumerate(line) if x == ch]
    if len(hits) != 1:
        raise ValueError(f"need exactly one {ch}")
    return hits[0]


def solve_maze(lines):
    start, end = _find(lines, "S"), _find(lines, "E")
    prev = {start: None}
    q = deque([start])
    while q:
        cur = q.popleft()
        if cur == end:
            path = []
            while cur is not None:
                path.append(cur)
                cur = prev[cur]
            return path[::-1]
        r, c = cur
        for nr, nc in ((r - 1, c), (r + 1, c), (r, c - 1), (r, c + 1)):
            if 0 <= nr < len(lines) and 0 <= nc < len(lines[nr]) and lines[nr][nc] != "#" and (nr, nc) not in prev:
                prev[(nr, nc)] = cur
                q.append((nr, nc))
    return None


def draw_path(lines, path):
    grid = [list(line) for line in lines]
    for r, c in path or []:
        if grid[r][c] == ".":
            grid[r][c] = "*"
    return ["".join(row) for row in grid]

=== task-order [apps]
--- task
Write `build_order(tasks)` where tasks maps a task name to the list of tasks it depends on. Return an order in which every task comes after its dependencies; among tasks that are ready at the same time, pick them alphabetically. Dependencies that are not keys are tasks with no dependencies of their own. A cycle raises ValueError whose message lists the tasks on the cycle joined by " -> ".
--- solution
import heapq


def build_order(tasks):
    deps = {t: set(d) for t, d in tasks.items()}
    for d in list(deps.values()):
        for x in d:
            deps.setdefault(x, set())
    users = {t: set() for t in deps}
    for t, ds in deps.items():
        for d in ds:
            users[d].add(t)
    left = {t: len(ds) for t, ds in deps.items()}
    ready = [t for t, n in left.items() if n == 0]
    heapq.heapify(ready)
    order = []
    while ready:
        t = heapq.heappop(ready)
        order.append(t)
        for u in users[t]:
            left[u] -= 1
            if left[u] == 0:
                heapq.heappush(ready, u)
    if len(order) < len(deps):
        stuck = {t for t, n in left.items() if n}
        start = min(stuck)
        path, cur = [start], start
        while True:
            cur = min(d for d in deps[cur] if d in stuck)
            if cur in path:
                cycle = path[path.index(cur):] + [cur]
                raise ValueError("cycle: " + " -> ".join(cycle))
            path.append(cur)
    return order

=== route-planner [apps]
--- task
Write `shortest_route(roads, start, goal)` where roads is a list of (a, b, km) two-way roads. Return (total_km, [cities]) for the shortest route using Dijkstra's algorithm, or None if unreachable; start == goal gives (0, [start]). Among equal-length routes return the one whose city list is alphabetically smallest. Negative distances raise ValueError; an unknown start raises KeyError.
--- solution
import heapq
from collections import defaultdict


def shortest_route(roads, start, goal):
    graph = defaultdict(list)
    for a, b, km in roads:
        if km < 0:
            raise ValueError("negative distance")
        graph[a].append((b, km))
        graph[b].append((a, km))
    if start not in graph and start != goal:
        raise KeyError(start)
    heap = [(0, [start])]
    done = set()
    while heap:
        dist, path = heapq.heappop(heap)
        city = path[-1]
        if city in done:
            continue
        if city == goal:
            return dist, path
        done.add(city)
        for nxt, km in graph[city]:
            if nxt not in done:
                heapq.heappush(heap, (dist + km, path + [nxt]))
    return None

=== meeting-rooms [apps]
--- task
Write `rooms_needed(meetings)` returning the minimum number of rooms for meetings given as (start, end) minute pairs (a meeting ending at t frees its room for one starting at t), and `assign_rooms(meetings)` returning a room number (starting at 1) for each meeting in input order, always reusing the lowest-numbered free room, processing meetings by start time (ties by input order).
--- solution
import heapq


def rooms_needed(meetings):
    events = sorted([(s, 1) for s, _ in meetings] + [(e, -1) for _, e in meetings], key=lambda x: (x[0], x[1]))
    cur = best = 0
    for _, d in events:
        cur += d
        best = max(best, cur)
    return best


def assign_rooms(meetings):
    order = sorted(range(len(meetings)), key=lambda i: (meetings[i][0], i))
    busy = []  # (end, room)
    free = []
    next_room = 1
    out = [0] * len(meetings)
    for i in order:
        s, e = meetings[i]
        while busy and busy[0][0] <= s:
            _, room = heapq.heappop(busy)
            heapq.heappush(free, room)
        if free:
            room = heapq.heappop(free)
        else:
            room = next_room
            next_room += 1
        out[i] = room
        heapq.heappush(busy, (e, room))
    return out

=== leaderboard-ties [apps]
--- task
Write `rank(scores)` taking (name, score) pairs and returning (rank, name, score) rows sorted by score descending then name, using competition ranking ("1224": tied players share a rank and the next rank skips). Write `dense_rank(scores)` with dense ranking ("1223"). If a name appears more than once only its best score counts.
--- solution
def _best(scores):
    best = {}
    for name, s in scores:
        if name not in best or s > best[name]:
            best[name] = s
    return sorted(best.items(), key=lambda kv: (-kv[1], kv[0]))


def rank(scores):
    rows, prev, r = [], None, 0
    for i, (name, s) in enumerate(_best(scores), 1):
        if s != prev:
            r, prev = i, s
        rows.append((r, name, s))
    return rows


def dense_rank(scores):
    rows, prev, r = [], None, 0
    for name, s in _best(scores):
        if s != prev:
            r, prev = r + 1, s
        rows.append((r, name, s))
    return rows

=== unit-convert [apps]
--- task
Write `convert(value, from_unit, to_unit)` for lengths (mm, cm, m, km, in, ft, yd, mi), masses (mg, g, kg, oz, lb) and temperatures (C, F, K). Converting between different kinds raises ValueError, as do unknown units. Return a float rounded to 6 significant digits. Also write `parse_quantity(text)` that reads "5.5 km" or "-40F" into (5.5, "km") / (-40.0, "F").
--- solution
import math
import re

LENGTH = {"mm": 0.001, "cm": 0.01, "m": 1.0, "km": 1000.0, "in": 0.0254, "ft": 0.3048, "yd": 0.9144, "mi": 1609.344}
MASS = {"mg": 1e-6, "g": 0.001, "kg": 1.0, "oz": 0.028349523125, "lb": 0.45359237}
TEMP = {"C", "F", "K"}


def _sig(x, digits=6):
    if x == 0:
        return 0.0
    return round(x, digits - 1 - int(math.floor(math.log10(abs(x)))))


def _to_kelvin(v, u):
    return v + 273.15 if u == "C" else (v - 32) * 5 / 9 + 273.15 if u == "F" else v


def _from_kelvin(k, u):
    return k - 273.15 if u == "C" else (k - 273.15) * 9 / 5 + 32 if u == "F" else k


def convert(value, from_unit, to_unit):
    for table in (LENGTH, MASS):
        if from_unit in table and to_unit in table:
            return _sig(value * table[from_unit] / table[to_unit])
    if from_unit in TEMP and to_unit in TEMP:
        return _sig(_from_kelvin(_to_kelvin(value, from_unit), to_unit))
    known = set(LENGTH) | set(MASS) | TEMP
    if from_unit not in known or to_unit not in known:
        raise ValueError("unknown unit")
    raise ValueError(f"cannot convert {from_unit} to {to_unit}")


def parse_quantity(text):
    m = re.fullmatch(r"\s*([+-]?\d+(?:\.\d+)?)\s*([A-Za-z]+)\s*", text)
    if not m:
        raise ValueError(f"bad quantity {text!r}")
    return float(m.group(1)), m.group(2)

=== shift-overlaps [apps]
--- task
Write `coverage_gaps(shifts, open_time, close_time, min_staff=1)` where shifts are (name, "HH:MM", "HH:MM") within one day. Return the time ranges between open and close where fewer than min_staff people are working, as ("HH:MM", "HH:MM", staff_count) tuples in order, merging adjacent ranges with the same count.
--- solution
def _m(t):
    h, m = map(int, t.split(":"))
    return h * 60 + m


def _t(m):
    return f"{m // 60:02d}:{m % 60:02d}"


def coverage_gaps(shifts, open_time, close_time, min_staff=1):
    lo, hi = _m(open_time), _m(close_time)
    points = {lo, hi}
    spans = []
    for _, a, b in shifts:
        a, b = max(_m(a), lo), min(_m(b), hi)
        if a < b:
            spans.append((a, b))
            points.update((a, b))
    edges = sorted(points)
    out = []
    for a, b in zip(edges, edges[1:]):
        n = sum(1 for s, e in spans if s <= a and e >= b)
        if n < min_staff:
            if out and out[-1][1] == a and out[-1][2] == n:
                out[-1] = (out[-1][0], b, n)
            else:
                out.append((a, b, n))
    return [(_t(a), _t(b), n) for a, b, n in out]
