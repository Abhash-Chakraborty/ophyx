=== flatten-dict [data]
--- task
Write `flatten(d, sep=".")` that turns a nested dict into a flat dict with joined keys ({"a": {"b": 1}} -> {"a.b": 1}); lists are kept as values and empty dicts become an empty-dict value at their key. Write `unflatten(flat, sep=".")` for the reverse; it raises ValueError when a key is both a value and a parent (e.g. "a" and "a.b").
--- solution
def flatten(d, sep="."):
    out = {}

    def walk(prefix, value):
        if isinstance(value, dict) and value:
            for k, v in value.items():
                walk(f"{prefix}{sep}{k}" if prefix else str(k), v)
        else:
            out[prefix] = value

    for k, v in d.items():
        walk(str(k), v)
    return out


def unflatten(flat, sep="."):
    root = {}
    for key in sorted(flat, key=lambda k: k.count(sep)):
        parts = key.split(sep)
        cur = root
        for p in parts[:-1]:
            nxt = cur.setdefault(p, {})
            if not isinstance(nxt, dict):
                raise ValueError(f"{key!r} conflicts with a value at {p!r}")
            cur = nxt
        if parts[-1] in cur and isinstance(cur[parts[-1]], dict) and cur[parts[-1]]:
            raise ValueError(f"{key!r} is also a parent")
        cur[parts[-1]] = flat[key]
    return root

=== deep-merge [data]
--- task
Write `deep_merge(base, override)` that returns a new dict: keys from override replace those in base, except that when both values are dicts they are merged recursively. Neither input may be modified. A value of None in override deletes the key from the result.
--- solution
import copy


def deep_merge(base, override):
    out = copy.deepcopy(base)
    for k, v in override.items():
        if v is None:
            out.pop(k, None)
        elif isinstance(v, dict) and isinstance(out.get(k), dict):
            out[k] = deep_merge(out[k], v)
        else:
            out[k] = copy.deepcopy(v)
    return out

=== dotted-path [data]
--- task
Write `get_path(obj, path, default=None)` that reads values from nested dicts and lists with paths like "users[0].name" or "a.b[2][1]" and returns default when any step is missing (missing key, index out of range, wrong type). Write `set_path(obj, path, value)` that sets the value in place, creating dicts for missing keys; list indices must already exist (IndexError otherwise).
--- solution
import re

_TOKEN = re.compile(r"([^.\[\]]+)|\[(-?\d+)\]")


def _steps(path):
    steps, pos = [], 0
    for m in _TOKEN.finditer(path):
        steps.append(m.group(1) if m.group(1) is not None else int(m.group(2)))
    if not steps:
        raise ValueError(f"empty path: {path!r}")
    return steps


def get_path(obj, path, default=None):
    cur = obj
    for s in _steps(path):
        try:
            if isinstance(s, int):
                if not isinstance(cur, list):
                    return default
                cur = cur[s]
            else:
                if not isinstance(cur, dict) or s not in cur:
                    return default
                cur = cur[s]
        except IndexError:
            return default
    return cur


def set_path(obj, path, value):
    steps = _steps(path)
    cur = obj
    for s, nxt in zip(steps, steps[1:]):
        if isinstance(s, int):
            cur = cur[s]
        else:
            if s not in cur:
                cur[s] = [] if isinstance(nxt, int) else {}
            cur = cur[s]
    last = steps[-1]
    if isinstance(last, int):
        cur[last] = value
    else:
        cur[last] = value

=== group-aggregate [data]
--- task
Write `summarize(records, by, field)` that groups a list of dicts by the value of key `by` and returns a dict mapping each group to {"count", "sum", "min", "max", "mean"} of `field` (mean rounded to 2 places). Records missing `field` or with a None value are counted in "missing" for their group and left out of the numbers; a group with no numeric values has None for sum, min, max and mean. --- solution
def summarize(records, by, field):
    groups = {}
    for r in records:
        g = groups.setdefault(r.get(by), {"values": [], "missing": 0})
        v = r.get(field)
        if v is None:
            g["missing"] += 1
        else:
            g["values"].append(v)
    out = {}
    for key, g in groups.items():
        vs = g["values"]
        out[key] = {
            "count": len(vs),
            "sum": sum(vs) if vs else None,
            "min": min(vs) if vs else None,
            "max": max(vs) if vs else None,
            "mean": round(sum(vs) / len(vs), 2) if vs else None,
            "missing": g["missing"],
        }
    return out

=== record-diff [data]
--- task
Write `diff_records(old, new, key="id")` that compares two lists of dicts identified by `key` and returns {"added": [...], "removed": [...], "changed": [...]}: added and removed are the full records (in the order of the list they come from), and changed is a list of {"key": k, "changes": {field: (old_value, new_value)}} for records whose other fields differ, including fields that appear or disappear (shown as None). Duplicate keys within one list raise ValueError.
--- solution
def _index(rows, key):
    out = {}
    for r in rows:
        k = r[key]
        if k in out:
            raise ValueError(f"duplicate key {k!r}")
        out[k] = r
    return out


def diff_records(old, new, key="id"):
    a, b = _index(old, key), _index(new, key)
    added = [r for r in new if r[key] not in a]
    removed = [r for r in old if r[key] not in b]
    changed = []
    for r in new:
        k = r[key]
        if k not in a:
            continue
        before = a[k]
        fields = [f for f in list(before) + [f for f in r if f not in before] if f != key]
        changes = {f: (before.get(f), r.get(f)) for f in fields if before.get(f) != r.get(f) or (f in before) != (f in r)}
        if changes:
            changed.append({"key": k, "changes": changes})
    return {"added": added, "removed": removed, "changed": changed}

=== pivot-sales [data]
--- task
Write `pivot(rows, row_key, col_key, value_key)` that builds a pivot table from a list of dicts: a dict of dicts summing value_key for each (row, column) pair, with every row containing every column seen anywhere (0 where there is no data), plus a "total" column per row and a "total" row with column sums. Rows and columns are sorted.
--- solution
def pivot(rows, row_key, col_key, value_key):
    cols = sorted({r[col_key] for r in rows})
    names = sorted({r[row_key] for r in rows})
    table = {n: {c: 0 for c in cols} for n in names}
    for r in rows:
        table[r[row_key]][r[col_key]] += r[value_key]
    for n in names:
        table[n]["total"] = sum(table[n][c] for c in cols)
    total = {c: sum(table[n][c] for n in names) for c in cols}
    total["total"] = sum(total.values())
    table["total"] = total
    return table

=== chunk-batch [data]
--- task
Write `chunked(iterable, size)` that yields lists of up to `size` items from any iterable (lazily: it must work on an infinite generator), `windowed(seq, n, step=1)` that returns overlapping tuples of length n advancing by step (no partial windows), and `dedupe(iterable, key=None)` that yields items in order, skipping ones whose key was seen. size, n and step must be positive (ValueError).
--- solution
from itertools import islice


def chunked(iterable, size):
    if size < 1:
        raise ValueError("size must be positive")
    it = iter(iterable)
    while True:
        batch = list(islice(it, size))
        if not batch:
            return
        yield batch


def windowed(seq, n, step=1):
    if n < 1 or step < 1:
        raise ValueError("n and step must be positive")
    items = list(seq)
    return [tuple(items[i:i + n]) for i in range(0, len(items) - n + 1, step)]


def dedupe(iterable, key=None):
    seen = set()
    for x in iterable:
        k = key(x) if key else x
        if k not in seen:
            seen.add(k)
            yield x

=== csv-typed-reader [data]
--- task
Write `read_typed_csv(text, schema)` that parses CSV text with a header row into a list of dicts, converting columns named in `schema` with the given type name: "int", "float", "bool" (true/false/yes/no/1/0, any case) or "date" (YYYY-MM-DD, to datetime.date). Empty cells become None. Columns not in the schema stay strings. A bad value raises ValueError mentioning the line number (header is line 1) and the column.
--- solution
import csv
import datetime
import io

_BOOL = {"true": True, "yes": True, "1": True, "false": False, "no": False, "0": False}


def _convert(kind, raw):
    if kind == "int":
        return int(raw)
    if kind == "float":
        return float(raw)
    if kind == "bool":
        if raw.strip().lower() not in _BOOL:
            raise ValueError(raw)
        return _BOOL[raw.strip().lower()]
    if kind == "date":
        return datetime.date.fromisoformat(raw.strip())
    raise ValueError(f"unknown type {kind}")


def read_typed_csv(text, schema):
    reader = csv.DictReader(io.StringIO(text))
    out = []
    for line_no, row in enumerate(reader, start=2):
        rec = {}
        for col, raw in row.items():
            if raw is None or raw == "":
                rec[col] = None
            elif col in schema:
                try:
                    rec[col] = _convert(schema[col], raw)
                except ValueError:
                    raise ValueError(f"line {line_no}, column {col!r}: bad {schema[col]} {raw!r}") from None
            else:
                rec[col] = raw
        out.append(rec)
    return out

=== json-lines-stats [data]
--- task
Write `jsonl_stats(lines)` that reads JSON Lines (an iterable of strings) and returns {"records": n, "bad_lines": [line numbers starting at 1], "fields": {field: count of records having it}} for top-level object fields. Blank lines are skipped; lines that are not valid JSON objects are bad.
--- solution
import json
from collections import Counter


def jsonl_stats(lines):
    n, bad, fields = 0, [], Counter()
    for i, line in enumerate(lines, 1):
        if not line.strip():
            continue
        try:
            obj = json.loads(line)
        except json.JSONDecodeError:
            bad.append(i)
            continue
        if not isinstance(obj, dict):
            bad.append(i)
            continue
        n += 1
        fields.update(obj.keys())
    return {"records": n, "bad_lines": bad, "fields": dict(fields)}

=== running-balance [data]
--- task
Write `statement(opening, transactions)` where transactions are (date string, description, amount) tuples with positive deposits and negative withdrawals. Return a list of dicts in date order (stable for equal dates) with "date", "description", "amount" and "balance" after each one, using Decimal amounts rounded to cents. Also write `overdrafts(rows)` returning the dates on which the balance first went below zero (each dip counted once until it recovers to zero or above).
--- solution
from decimal import ROUND_HALF_UP, Decimal


def _money(x):
    return Decimal(str(x)).quantize(Decimal("0.01"), rounding=ROUND_HALF_UP)


def statement(opening, transactions):
    balance = _money(opening)
    rows = []
    for date, desc, amount in sorted(transactions, key=lambda t: t[0]):
        amt = _money(amount)
        balance += amt
        rows.append({"date": date, "description": desc, "amount": amt, "balance": balance})
    return rows


def overdrafts(rows):
    dates, below = [], False
    for r in rows:
        if r["balance"] < 0 and not below:
            dates.append(r["date"])
            below = True
        elif r["balance"] >= 0:
            below = False
    return dates

=== settle-debts [data]
--- task
Write `settle(expenses)` for splitting group costs: each expense is (payer, amount, participants) and is shared equally by its participants. Return a list of (debtor, creditor, amount) payments, amounts rounded to cents, that settles everyone, using a greedy match of the largest debtor with the largest creditor (names break ties alphabetically). Write `balances(expenses)` too, returning each person's net balance (positive means they are owed).
--- solution
from collections import defaultdict


def balances(expenses):
    net = defaultdict(float)
    for payer, amount, people in expenses:
        share = amount / len(people)
        net[payer] += amount
        for p in people:
            net[p] -= share
    return {k: round(v, 2) for k, v in sorted(net.items())}


def settle(expenses):
    net = balances(expenses)
    debtors = [[-v, k] for k, v in net.items() if v < -0.005]
    creditors = [[v, k] for k, v in net.items() if v > 0.005]
    out = []
    while debtors and creditors:
        debtors.sort(key=lambda x: (-x[0], x[1]))
        creditors.sort(key=lambda x: (-x[0], x[1]))
        d, c = debtors[0], creditors[0]
        pay = round(min(d[0], c[0]), 2)
        out.append((d[1], c[1], pay))
        d[0] = round(d[0] - pay, 2)
        c[0] = round(c[0] - pay, 2)
        if d[0] <= 0.005:
            debtors.pop(0)
        if c[0] <= 0.005:
            creditors.pop(0)
    return out

=== timesheet-hours [data]
--- task
Write `hours_by_week(entries)` where entries are (date "YYYY-MM-DD", start "HH:MM", end "HH:MM") work periods. A period whose end is earlier than its start runs past midnight and counts for its start date. Return a dict mapping ISO week labels like "2024-W05" to total hours rounded to 2 places, sorted by week. Overlapping periods on the same date are counted once (merge them first). Bad times raise ValueError.
--- solution
import datetime


def _minutes(hhmm):
    h, m = hhmm.split(":")
    h, m = int(h), int(m)
    if not (0 <= h < 24 and 0 <= m < 60):
        raise ValueError(f"bad time {hhmm!r}")
    return h * 60 + m


def hours_by_week(entries):
    by_date = {}
    for day, start, end in entries:
        s, e = _minutes(start), _minutes(end)
        if e <= s:
            e += 24 * 60
        by_date.setdefault(day, []).append((s, e))
    weeks = {}
    for day, spans in by_date.items():
        spans.sort()
        total, cur_s, cur_e = 0, None, None
        for s, e in spans:
            if cur_e is None or s > cur_e:
                if cur_e is not None:
                    total += cur_e - cur_s
                cur_s, cur_e = s, e
            else:
                cur_e = max(cur_e, e)
        total += cur_e - cur_s
        y, w, _ = datetime.date.fromisoformat(day).isocalendar()
        label = f"{y}-W{w:02d}"
        weeks[label] = weeks.get(label, 0) + total
    return {k: round(v / 60, 2) for k, v in sorted(weeks.items())}
--- tests
import pytest


def test_simple_week():
    entries = [("2024-01-29", "09:00", "17:30"), ("2024-01-30", "09:00", "12:15")]
    assert hours_by_week(entries) == {"2024-W05": 11.75}


def test_overnight():
    assert hours_by_week([("2024-02-02", "22:00", "02:00")]) == {"2024-W05": 4.0}


def test_overlap_merged():
    entries = [("2024-02-05", "09:00", "12:00"), ("2024-02-05", "11:00", "13:00"), ("2024-02-05", "14:00", "15:00")]
    assert hours_by_week(entries) == {"2024-W06": 5.0}


def test_year_boundary():
    assert hours_by_week([("2024-12-30", "08:00", "09:00")]) == {"2025-W01": 1.0}


def test_bad_time():
    with pytest.raises(ValueError):
        hours_by_week([("2024-01-01", "25:00", "26:00")])

=== free-slots [data]
--- task
Write `free_slots(busy, day_start, day_end, min_minutes=30)` that takes busy intervals as ("HH:MM", "HH:MM") pairs (unsorted, possibly overlapping or extending outside the day) and returns the free intervals within [day_start, day_end] that are at least min_minutes long, as "HH:MM" pairs in order.
--- solution
def _m(t):
    h, m = map(int, t.split(":"))
    return h * 60 + m


def _t(m):
    return f"{m // 60:02d}:{m % 60:02d}"


def free_slots(busy, day_start, day_end, min_minutes=30):
    start, end = _m(day_start), _m(day_end)
    spans = sorted((max(_m(a), start), min(_m(b), end)) for a, b in busy)
    out, cur = [], start
    for a, b in spans:
        if b <= cur:
            continue
        if a - cur >= min_minutes:
            out.append((_t(cur), _t(a)))
        cur = max(cur, b)
    if end - cur >= min_minutes:
        out.append((_t(cur), _t(end)))
    return out
--- tests
def test_basic():
    busy = [("10:00", "11:00"), ("09:00", "09:30"), ("13:00", "14:30")]
    assert free_slots(busy, "09:00", "17:00") == [("09:30", "10:00"), ("11:00", "13:00"), ("14:30", "17:00")]


def test_overlapping_busy():
    busy = [("10:00", "12:00"), ("11:00", "13:00")]
    assert free_slots(busy, "10:00", "14:00") == [("13:00", "14:00")]

    busy = [("09:00", "09:50"), ("10:00", "17:00")]
    assert free_slots(busy, "09:00", "17:00") == []
    assert free_slots(busy, "09:00", "17:00", min_minutes=10) == [("09:50", "10:00")]

    assert free_slots([("07:00", "09:30"), ("16:00", "20:00")], "09:00", "17:00") == [("09:30", "16:00")]

    assert free_slots([], "08:00", "09:00") == [("08:00", "09:00")]

=== business-days [data]
--- task
Write `business_days(start, end, holidays=())` counting weekdays (Mon-Fri) from start to end inclusive, skipping the given holiday dates; if end is before start the count is negative (the same days counted backwards). Write `add_business_days(start, n, holidays=())` that returns the date n business days after start (n may be negative; n = 0 returns start, or the next business day if start is not one). Dates are datetime.date objects.
--- solution
import datetime

ONE = datetime.timedelta(days=1)


def _is_bday(d, hol):
    return d.weekday() < 5 and d not in hol


def business_days(start, end, holidays=()):
    hol = set(holidays)
    if end < start:
        return -business_days(end, start, hol)
    n, d = 0, start
    while d <= end:
        n += _is_bday(d, hol)
        d += ONE
    return n


def add_business_days(start, n, holidays=()):
    hol = set(holidays)
    d = start
    if n == 0:
        while not _is_bday(d, hol):
            d += ONE
        return d
    step = ONE if n > 0 else -ONE
    left = abs(n)
    while left:
        d += step
        if _is_bday(d, hol):
            left -= 1
    return d

=== resample-series [data]
--- task
Write `resample(points, minutes, how="mean")` that groups (datetime, value) points into consecutive buckets of `minutes` length aligned to midnight, and returns (bucket_start, aggregate) pairs for every bucket from the first to the last point, in order. `how` is "mean", "sum", "max" or "last" (the latest point in time). Empty buckets in between get None. Unsorted input is allowed.
--- solution
import datetime


def resample(points, minutes, how="mean"):
    if not points:
        return []
    width = datetime.timedelta(minutes=minutes)

    def floor(t):
        midnight = t.replace(hour=0, minute=0, second=0, microsecond=0)
        return midnight + ((t - midnight) // width) * width

    buckets = {}
    for t, v in sorted(points, key=lambda p: p[0]):
        buckets.setdefault(floor(t), []).append(v)
    agg = {"mean": lambda vs: sum(vs) / len(vs), "sum": sum, "max": max, "last": lambda vs: vs[-1]}[how]
    out, cur, last = [], min(buckets), max(buckets)
    while cur <= last:
        vs = buckets.get(cur)
        out.append((cur, agg(vs) if vs else None))
        cur += width
    return out

=== fill-missing-dates [data]
--- task
Write `fill_dates(rows, fill="previous")` where rows are (date string "YYYY-MM-DD", value) pairs, possibly unsorted and with gaps. Return a sorted list covering every day from the first to the last date. Missing days get the previous known value when fill is "previous", 0 when fill is "zero", or a linear interpolation between neighbours (rounded to 2 places) when fill is "linear". Duplicate dates raise ValueError.
--- solution
import datetime


def fill_dates(rows, fill="previous"):
    known = {}
    for d, v in rows:
        day = datetime.date.fromisoformat(d)
        if day in known:
            raise ValueError(f"duplicate date {d}")
        known[day] = v
    if not known:
        return []
    days = sorted(known)
    out = []
    d = days[0]
    while d <= days[-1]:
        if d in known:
            out.append((d.isoformat(), known[d]))
        elif fill == "zero":
            out.append((d.isoformat(), 0))
        elif fill == "previous":
            out.append((d.isoformat(), out[-1][1]))
        elif fill == "linear":
            before = max(k for k in days if k < d)
            after = min(k for k in days if k > d)
            frac = (d - before).days / (after - before).days
            out.append((d.isoformat(), round(known[before] + frac * (known[after] - known[before]), 2)))
        else:
            raise ValueError(f"unknown fill {fill!r}")
        d += datetime.timedelta(days=1)
    return out
--- tests

        fill_dates([("2024-01-01", 1), ("2024-01-01", 2)])

=== percentile-summary [data]
--- task
Write `percentile(values, p)` using linear interpolation between closest ranks (the same as numpy's default), for p from 0 to 100, raising ValueError on an empty list or p out of range. Write `describe(values)` returning {"n", "min", "p25", "median", "p75", "max", "iqr_outliers"} where outliers are values outside 1.5 IQR from the quartiles, sorted.
--- solution
def percentile(values, p):
    if not values:
        raise ValueError("no values")
    if not 0 <= p <= 100:
        raise ValueError("p must be between 0 and 100")
    xs = sorted(values)
    k = (len(xs) - 1) * p / 100
    lo = int(k)
    hi = min(lo + 1, len(xs) - 1)
    return xs[lo] + (xs[hi] - xs[lo]) * (k - lo)


def describe(values):
    q1, q3 = percentile(values, 25), percentile(values, 75)
    iqr = q3 - q1
    lo, hi = q1 - 1.5 * iqr, q3 + 1.5 * iqr
    return {
        "n": len(values),
        "min": min(values),
        "p25": q1,
        "median": percentile(values, 50),
        "p75": q3,
        "max": max(values),
        "iqr_outliers": sorted(v for v in values if v < lo or v > hi),
    }
--- tests
import pytest


def test_percentiles():
    xs = [15, 20, 35, 40, 50]
    assert percentile(xs, 0) == 15
    assert percentile(xs, 100) == 50
    assert percentile(xs, 50) == 35
    assert percentile(xs, 40) == 29.0


def test_unsorted_input():
    assert percentile([3, 1, 2], 50) == 2


def test_describe():
    d = describe([1, 2, 3, 4, 5, 6, 7, 8, 100])
    assert d["median"] == 5 and d["p25"] == 3 and d["p75"] == 7
    assert d["iqr_outliers"] == [100]


def test_errors():
    with pytest.raises(ValueError):
        percentile([], 50)
    with pytest.raises(ValueError):
        percentile([1], 101)


def test_single():
    assert describe([4])["iqr_outliers"] == []
