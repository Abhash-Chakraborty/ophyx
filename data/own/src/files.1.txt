=== dotenv-parser [files]
--- task
Write `parse_env(text)` for .env files: KEY=VALUE lines, optional "export " prefix, "#" comments (full-line, or after an unquoted value when preceded by a space), single-quoted values taken literally, double-quoted values with \n and \" escapes, and ${OTHER} references expanded from keys defined earlier in the file (unknown ones become ""). Invalid lines raise ValueError with the line number.
--- solution
import re

_LINE = re.compile(r"^(?:export\s+)?([A-Za-z_][A-Za-z0-9_]*)\s*=\s*(.*)$")
_REF = re.compile(r"\$\{([A-Za-z_][A-Za-z0-9_]*)\}")


def parse_env(text):
    env = {}
    for n, raw in enumerate(text.splitlines(), 1):
        line = raw.strip()
        if not line or line.startswith("#"):
            continue
        m = _LINE.match(line)
        if not m:
            raise ValueError(f"line {n}: cannot parse {raw!r}")
        key, value = m.group(1), m.group(2)
        if value.startswith("'"):
            if not value.endswith("'") or len(value) < 2:
                raise ValueError(f"line {n}: unterminated quote")
            env[key] = value[1:-1]
            continue
        if value.startswith('"'):
            end = value.rfind('"')
            if end == 0:
                raise ValueError(f"line {n}: unterminated quote")
            value = value[1:end].replace("\\n", "\n").replace('\\"', '"')
        else:
            value = re.split(r"\s+#", value, maxsplit=1)[0].strip()
        env[key] = _REF.sub(lambda r: env.get(r.group(1), ""), value)
    return env

=== requirements-parser [files]
--- task
Write `parse_requirements(text)` that reads a requirements.txt and returns a list of dicts {"name", "extras", "specifier", "marker"}: the name normalised (lowercase, runs of "-", "_", "." turned into "-"), extras as a sorted list, the version specifier string without spaces (e.g. ">=1.2,<2"), and the environment marker after ";" stripped (or None). Comments, blank lines and option lines starting with "-" are skipped; "\" line continuations are joined.
--- solution
import re

_REQ = re.compile(r"^([A-Za-z0-9][A-Za-z0-9._-]*)\s*(?:\[([^\]]*)\])?\s*([^;]*?)\s*(?:;\s*(.*))?$")


def parse_requirements(text):
    text = re.sub(r"\\\n", " ", text)
    out = []
    for raw in text.splitlines():
        line = raw.split(" #", 1)[0].strip()
        if not line or line.startswith("#") or line.startswith("-"):
            continue
        m = _REQ.match(line)
        if not m:
            raise ValueError(f"bad requirement: {raw!r}")
        name = re.sub(r"[-_.]+", "-", m.group(1)).lower()
        extras = sorted(e.strip() for e in (m.group(2) or "").split(",") if e.strip())
        spec = m.group(3).replace(" ", "")
        marker = m.group(4).strip() if m.group(4) else None
        out.append({"name": name, "extras": extras, "specifier": spec, "marker": marker})
    return out

=== ini-config [files]
--- task
Write `load_config(path, defaults=None)` that reads an INI file with configparser and returns a dict of sections, each a dict of options converted to int, float or bool ("true"/"false"/"yes"/"no"/"on"/"off") when they look like one, otherwise strings. `defaults` is a nested dict merged underneath (file values win). A missing file raises FileNotFoundError with the path in the message.
--- solution
import configparser
import os
import re

_BOOLS = {"true": True, "yes": True, "on": True, "false": False, "no": False, "off": False}


def _convert(v):
    low = v.strip().lower()
    if low in _BOOLS:
        return _BOOLS[low]
    if re.fullmatch(r"[+-]?\d+", low):
        return int(low)
    if re.fullmatch(r"[+-]?(\d+\.\d*|\.\d+|\d+(\.\d*)?e[+-]?\d+)", low):
        return float(low)
    return v


def load_config(path, defaults=None):
    if not os.path.exists(path):
        raise FileNotFoundError(f"config file not found: {path}")
    parser = configparser.ConfigParser(interpolation=None)
    parser.read(path, encoding="utf-8")
    out = {s: dict(v) for s, v in (defaults or {}).items()}
    for section in parser.sections():
        out.setdefault(section, {})
        for key, value in parser.items(section):
            out[section][key] = _convert(value)
    return out

=== find-duplicate-files [files]
--- task
Write `find_duplicates(root)` that walks a directory tree and returns groups of files with identical contents: a list of lists of paths relative to root (as POSIX strings), each group sorted, groups sorted by their first path. Files are first grouped by size and only same-size files are hashed (SHA-256, read in chunks). Empty files are ignored.
--- solution
import hashlib
import os
from collections import defaultdict
from pathlib import Path


def _digest(path, chunk=1 << 16):
    h = hashlib.sha256()
    with open(path, "rb") as f:
        while True:
            block = f.read(chunk)
            if not block:
                break
            h.update(block)
    return h.hexdigest()


def find_duplicates(root):
    root = Path(root)
    by_size = defaultdict(list)
    for dirpath, _, files in os.walk(root):
        for name in files:
            p = Path(dirpath) / name
            if p.is_file() and not p.is_symlink():
                size = p.stat().st_size
                if size:
                    by_size[size].append(p)
    groups = []
    for paths in by_size.values():
        if len(paths) < 2:
            continue
        by_hash = defaultdict(list)
        for p in paths:
            by_hash[_digest(p)].append(p.relative_to(root).as_posix())
        groups += [sorted(g) for g in by_hash.values() if len(g) > 1]
    return sorted(groups)

=== bulk-rename [files]
--- task
Write `plan_renames(directory, pattern, template)` that matches file names in one directory (not recursive) against a regex `pattern` and builds new names from `template` using {name} groups and {n} (1-based counter in sorted order of the original names, zero-padded with {n:03d} style formats). Return a list of (old, new) pairs. Raise ValueError if two files would get the same name or a new name collides with an existing file that is not itself being renamed. Write `apply_renames(directory, plan)` that performs it safely (via temporary names, so swaps work).
--- solution
import os
import re
import uuid


def plan_renames(directory, pattern, template):
    rx = re.compile(pattern)
    names = sorted(n for n in os.listdir(directory) if os.path.isfile(os.path.join(directory, n)))
    plan, n = [], 0
    for name in names:
        m = rx.fullmatch(name)
        if not m:
            continue
        n += 1
        new = template.format(n=n, **m.groupdict())
        if new != name:
            plan.append((name, new))
    targets = [new for _, new in plan]
    if len(set(targets)) != len(targets):
        raise ValueError("two files would get the same name")
    moving = {old for old, _ in plan}
    for t in targets:
        if t in names and t not in moving:
            raise ValueError(f"{t} already exists")
    return plan


def apply_renames(directory, plan):
    tmp = []
    for old, new in plan:
        t = f".rename-{uuid.uuid4().hex}"
        os.rename(os.path.join(directory, old), os.path.join(directory, t))
        tmp.append((t, new))
    for t, new in tmp:
        os.rename(os.path.join(directory, t), os.path.join(directory, new))

--- task
Write `count_lines(root, extensions=(".py",))` that walks a project and returns {"files", "code", "comment", "blank"} totals for files with the given extensions, skipping directories named .git, __pycache__, node_modules and any starting with ".venv" or "venv". import os

SKIP = {".git", "__pycache__", "node_modules"}


def count_lines(root, extensions=(".py",)):
    totals = {"files": 0, "code": 0, "comment": 0, "blank": 0}
    for dirpath, dirnames, filenames in os.walk(root):
        dirnames[:] = [d for d in dirnames if d not in SKIP and not d.startswith((".venv", "venv"))]
        for name in filenames:
            if not name.endswith(tuple(extensions)):
                continue
            totals["files"] += 1
            with open(os.path.join(dirpath, name), encoding="utf-8", errors="replace") as f:
                for line in f:
                    s = line.strip()
                    if not s:
                        totals["blank"] += 1
                    elif s.startswith("#"):
                        totals["comment"] += 1
                    else:
                        totals["code"] += 1
    return totals

=== log-level-report [files]
--- task
Write `log_report(lines)` for log lines like "2024-05-01 12:00:03,123 ERROR [db] connection lost". Return {"levels": count per level, "by_logger": {logger: {level: count}}, "first_error": the timestamp string of the first ERROR or CRITICAL line or None, "unparsed": number of lines that don't match}. Lines that start with whitespace are continuation lines (e.g. tracebacks) and are neither counted nor unparsed.
--- solution
import re
from collections import Counter, defaultdict

_LINE = re.compile(r"^(\d{4}-\d{2}-\d{2} \d{2}:\d{2}:\d{2},\d{3}) (DEBUG|INFO|WARNING|ERROR|CRITICAL) \[([^\]]+)\] ")


def log_report(lines):
    levels = Counter()
    by_logger = defaultdict(Counter)
    first_error, unparsed = None, 0
    for line in lines:
        if not line.strip() or line[0].isspace():
            continue
        m = _LINE.match(line)
        if not m:
            unparsed += 1
            continue
        ts, level, logger = m.groups()
        levels[level] += 1
        by_logger[logger][level] += 1
        if first_error is None and level in ("ERROR", "CRITICAL"):
            first_error = ts
    return {"levels": dict(levels), "by_logger": {k: dict(v) for k, v in by_logger.items()},
            "first_error": first_error, "unparsed": unparsed}

=== access-log-stats [files]
--- task
Write `access_stats(lines)` for web server logs in the Common Log Format, e.g. '127.0.0.1 - frank [10/Oct/2000:13:55:36 -0700] "GET /a.gif HTTP/1.0" 200 2326'. Return {"requests": n, "status": count per status class like "2xx", "top_paths": the 3 most requested paths without query strings as (path, count) with ties alphabetical, "bytes": total bytes (a "-" size counts as 0), "unique_ips": number of distinct client IPs}. import re
from collections import Counter

_CLF = re.compile(r'^(\S+) \S+ \S+ \[[^\]]+\] "(\S+) (\S+)[^"]*" (\d{3}) (\d+|-)')


def access_stats(lines):
    n, status, paths, total, ips = 0, Counter(), Counter(), 0, set()
    for line in lines:
        m = _CLF.match(line)
        if not m:
            continue
        ip, _, path, code, size = m.groups()
        n += 1
        status[code[0] + "xx"] += 1
        paths[path.split("?", 1)[0]] += 1
        total += 0 if size == "-" else int(size)
        ips.add(ip)
    top = sorted(paths.items(), key=lambda kv: (-kv[1], kv[0]))[:3]
    return {"requests": n, "status": dict(status), "top_paths": top, "bytes": total, "unique_ips": len(ips)}

=== tail-file [files]
--- task
Write `tail(path, n=10)` that returns the last n lines of a text file (without their newline characters) by reading blocks backwards from the end, so it stays fast on huge files. It must handle files with or without a trailing newline, files shorter than n lines, empty files and n = 0.
--- solution
import os


def tail(path, n=10, block=4096):
    if n <= 0:
        return []
    with open(path, "rb") as f:
        f.seek(0, os.SEEK_END)
        pos = f.tell()
        data = b""
        while pos > 0 and data.count(b"\n") <= n:
            step = min(block, pos)
            pos -= step
            f.seek(pos)
            data = f.read(step) + data
    text = data.decode("utf-8", errors="replace")
    if text.endswith("\n"):
        text = text[:-1]
    if not text and pos == 0:
        return []
    return text.split("\n")[-n:]

=== rotate-backups [files]
--- task
Write `prune_backups(directory, keep_daily=7, keep_weekly=4)` for files named like "backup-2024-05-31.tar.gz". Keep the newest file of each of the last keep_daily distinct days that have backups, plus the newest file of each of the last keep_weekly ISO weeks (counting weeks that have backups, newest first). Delete every other matching file and return the sorted list of deleted names. Files that don't match the pattern are never touched.
--- solution
import datetime
import os
import re

_NAME = re.compile(r"^backup-(\d{4}-\d{2}-\d{2})\.tar\.gz$")


def prune_backups(directory, keep_daily=7, keep_weekly=4):
    dated = []
    for name in os.listdir(directory):
        m = _NAME.match(name)
        if m:
            try:
                dated.append((datetime.date.fromisoformat(m.group(1)), name))
            except ValueError:
                continue
    dated.sort(reverse=True)
    keep = set()
    days, weeks = [], []
    for d, name in dated:
        if d not in days and len(days) < keep_daily:
            days.append(d)
            keep.add(name)
        week = d.isocalendar()[:2]
        if week not in weeks and len(weeks) < keep_weekly:
            weeks.append(week)
            keep.add(name)
    deleted = sorted(name for _, name in dated if name not in keep)
    for name in deleted:
        os.remove(os.path.join(directory, name))
    return deleted

=== safe-json-write [files]
--- task
Write `save_json(path, data)` that writes JSON atomically: dump to a temporary file in the same directory, flush and fsync it, then os.replace it over the target, so a crash never leaves a half-written file; on any error the temporary file is removed and the original file stays untouched. Write `load_json(path, default)` that returns default when the file does not exist.
--- solution
import json
import os
import tempfile


def save_json(path, data):
    directory = os.path.dirname(os.path.abspath(path))
    fd, tmp = tempfile.mkstemp(prefix=".tmp-", dir=directory)
    try:
        with os.fdopen(fd, "w", encoding="utf-8") as f:
            json.dump(data, f, indent=2, sort_keys=True)
            f.flush()
            os.fsync(f.fileno())
        os.replace(tmp, path)
    except BaseException:
        try:
            os.remove(tmp)
        except FileNotFoundError:
            pass
        raise


def load_json(path, default):
    try:
        with open(path, encoding="utf-8") as f:
            return json.load(f)
    except FileNotFoundError:
        return default

=== directory-tree [files]
--- task
Write `tree(root, max_depth=None)` that returns a text drawing of a directory like the `tree` command: the root name on the first line, then entries with "├── " / "└── " branches and "│   " / "    " indents, directories first then files, each group sorted case-insensitively, directories shown with a trailing "/". max_depth limits how many levels below the root are shown.
--- solution
from pathlib import Path


def tree(root, max_depth=None):
    root = Path(root)
    lines = [root.name + "/"]

    def walk(d, prefix, depth):
        if max_depth is not None and depth > max_depth:
            return
        entries = sorted(d.iterdir(), key=lambda p: (not p.is_dir(), p.name.lower()))
        for i, p in enumerate(entries):
            last = i == len(entries) - 1
            lines.append(prefix + ("└── " if last else "├── ") + p.name + ("/" if p.is_dir() else ""))
            if p.is_dir():
                walk(p, prefix + ("    " if last else "│   "), depth + 1)

    walk(root, "", 1)
    return "\n".join(lines)

=== cli-word-count [files]
--- task
Write `main(argv=None)` for a command-line tool like wc: it takes file paths and the flags -l, -w, -c (lines, words, bytes; default all three), prints one line per file with right-aligned counts (width 7 each) followed by the name, plus a "total" line when there is more than one file, and returns 0. A missing file prints "wc: <name>: No such file or directory" to stderr, continues, and makes the return code 1. Use argparse.
--- solution
import argparse
import sys


def _counts(data):
    return {"l": data.count(b"\n"), "w": len(data.split()), "c": len(data)}


def main(argv=None):
    ap = argparse.ArgumentParser(prog="wc")
    ap.add_argument("-l", action="store_true")
    ap.add_argument("-w", action="store_true")
    ap.add_argument("-c", action="store_true")
    ap.add_argument("files", nargs="+")
    a = ap.parse_args(argv)
    keys = [k for k in "lwc" if getattr(a, k)] or ["l", "w", "c"]
    total = {k: 0 for k in "lwc"}
    rc, shown = 0, 0
    for name in a.files:
        try:
            with open(name, "rb") as f:
                c = _counts(f.read())
        except FileNotFoundError:
            print(f"wc: {name}: No such file or directory", file=sys.stderr)
            rc = 1
            continue
        shown += 1
        for k in "lwc":
            total[k] += c[k]
        print("".join(f"{c[k]:>7}" for k in keys) + f" {name}")
    if len(a.files) > 1:
        print("".join(f"{total[k]:>7}" for k in keys) + " total")
    return rc

=== csv-merge-files [files]
--- task
Write `merge_csv(paths, out_path)` that concatenates CSV files with possibly different columns into one file whose header is the union of all columns in order of first appearance; missing cells are empty. Add a "source" column (the file's base name) as the first column. Return the number of data rows written. Files are read with the csv module (quoted fields with commas must survive).
--- solution
import csv
import os


def merge_csv(paths, out_path):
    columns, rows = [], []
    for p in paths:
        with open(p, newline="", encoding="utf-8") as f:
            reader = csv.DictReader(f)
            for c in reader.fieldnames or []:
                if c not in columns:
                    columns.append(c)
            for r in reader:
                rows.append({"source": os.path.basename(p), **r})
    with open(out_path, "w", newline="", encoding="utf-8") as f:
        w = csv.DictWriter(f, fieldnames=["source"] + columns, restval="")
        w.writeheader()
        w.writerows(rows)
    return len(rows)
