=== snake-camel-case [text]
--- task
Write `to_snake(name)` that turns camelCase or PascalCase identifiers into snake_case (keeping runs of capitals such as "HTTPServer" together as "http_server"), and `to_camel(name)` that turns snake_case into camelCase. Leading and trailing underscores are dropped.
--- solution
import re


def to_snake(name):
    name = name.strip("_")
    s = re.sub(r"([A-Z]+)([A-Z][a-z])", r"\1_\2", name)
    s = re.sub(r"([a-z0-9])([A-Z])", r"\1_\2", s)
    return s.replace("-", "_").lower()


def to_camel(name):
    parts = [p for p in name.strip("_").split("_") if p]
    if not parts:
        return ""
    return parts[0].lower() + "".join(p[:1].upper() + p[1:].lower() for p in parts[1:])

=== human-bytes [text]
--- task
Write `human_bytes(n)` that formats a byte count with binary units: "512 B", "1.0 KiB", "1.5 MiB", up to "TiB", with one decimal place above bytes. Negative numbers raise ValueError. Also write `parse_bytes(text)` that reads strings like "10 KiB", "2.5MiB", "7 B" or "300" (bytes) back into an int, rounding down.
--- solution
UNITS = ["B", "KiB", "MiB", "GiB", "TiB"]


def human_bytes(n):
    if n < 0:
        raise ValueError("negative size")
    if n < 1024:
        return f"{n} B"
    value = float(n)
    for unit in UNITS[1:]:
        value /= 1024
        if value < 1024 or unit == UNITS[-1]:
            return f"{value:.1f} {unit}"


def parse_bytes(text):
    text = text.strip()
    for i, unit in reversed(list(enumerate(UNITS))):
        if text.endswith(unit):
            number = text[: -len(unit)].strip()
            return int(float(number) * 1024**i)
    return int(text)

=== parse-duration [text]
--- task
Write `parse_duration(text)` that converts strings such as "1h30m", "45s", "2d 4h", "1h 5m 10s" or "90" (plain seconds) into a number of seconds (int). Units are d, h, m and s; each may appear at most once and in that order. Anything else raises ValueError.
--- solution
import re

_PART = re.compile(r"\s*(\d+)\s*([dhms])")
_ORDER = "dhms"
_SECONDS = {"d": 86400, "h": 3600, "m": 60, "s": 1}


def parse_duration(text):
    text = text.strip()
    if text.isdigit():
        return int(text)
    pos, total, last = 0, 0, -1
    if not text:
        raise ValueError("empty duration")
    while pos < len(text):
        m = _PART.match(text, pos)
        if not m:
            raise ValueError(f"bad duration: {text!r}")
        idx = _ORDER.index(m.group(2))
        if idx <= last:
            raise ValueError(f"units out of order or repeated: {text!r}")
        last = idx
        total += int(m.group(1)) * _SECONDS[m.group(2)]
        pos = m.end()
    return total

import re
from collections import Counter


def top_words(text, n, stopwords=()):
    stop = {w.lower() for w in stopwords}
    words = [w.lower().strip("'") for w in re.findall(r"[A-Za-z']+", text)]
    counts = Counter(w for w in words if w and w not in stop)
    return sorted(counts.items(), key=lambda kv: (-kv[1], kv[0]))[:n]

=== wrap-paragraphs [text]
--- task
Write `wrap(text, width)` that re-wraps text into lines of at most `width` characters, keeping paragraphs (separated by blank lines) apart with a single blank line. Words longer than `width` go on their own line unbroken. Return the result as one string without a trailing newline.
--- solution
def wrap(text, width):
    if width < 1:
        raise ValueError("width must be positive")
    paragraphs = [p.split() for p in text.replace("\r\n", "\n").split("\n\n")]
    out = []
    for words in paragraphs:
        if not words:
            continue
        lines, line = [], ""
        for w in words:
            if not line:
                line = w
            elif len(line) + 1 + len(w) <= width:
                line += " " + w
            else:
                lines.append(line)
                line = w
        lines.append(line)
        out.append("\n".join(lines))
    return "\n\n".join(out)

=== mask-card [text]
--- task
Write `luhn_valid(number)` that checks a card number with the Luhn algorithm (spaces and dashes allowed, at least 12 digits, otherwise False), and `mask_card(number)` that returns the digits with all but the last four replaced by "*", grouped in fours from the left with spaces, e.g. "**** **** **** 1111". mask_card raises ValueError for an invalid number.
--- solution
def _digits(number):
    return number.replace(" ", "").replace("-", "")


def luhn_valid(number):
    d = _digits(number)
    if not d.isdigit() or len(d) < 12:
        return False
    total = 0
    for i, ch in enumerate(reversed(d)):
        n = int(ch)
        if i % 2 == 1:
            n *= 2
            if n > 9:
                n -= 9
        total += n
    return total % 10 == 0


def mask_card(number):
    if not luhn_valid(number):
        raise ValueError("invalid card number")
    d = _digits(number)
    masked = "*" * (len(d) - 4) + d[-4:]
    return " ".join(masked[i:i + 4] for i in range(0, len(masked), 4))

=== extract-mentions [text]
--- task
Write `extract(post)` that returns a dict with three lists, in order of first appearance and without duplicates: "hashtags" (words after "#", lowercased), "mentions" (names after "@", made of letters, digits and underscores) and "urls" (http or https URLs up to the next whitespace, with trailing punctuation .,!?) removed). --- solution
import re

_URL = re.compile(r"https?://\S+")
_TAG = re.compile(r"(?<![\w&])#(\w+)")
_MENTION = re.compile(r"(?<![\w.])@(\w+)")


def _unique(items):
    seen, out = set(), []
    for x in items:
        if x not in seen:
            seen.add(x)
            out.append(x)
    return out


def extract(post):
    urls = [u.rstrip(".,!?)") for u in _URL.findall(post)]
    rest = _URL.sub(" ", post)
    return {
        "hashtags": _unique(t.lower() for t in _TAG.findall(rest)),
        "mentions": _unique(_MENTION.findall(rest)),
        "urls": _unique(urls),
    }

=== render-template [text]
--- task
Write `render(template, values)` for a tiny template language: "{{ name }}" is replaced by values["name"] (spaces inside the braces optional), dotted names such as "{{ user.email }}" look up nested dicts, "{{ name | upper }}" and "| lower" and "| title" apply a filter, and "{{{{" renders a literal "{{". A missing name raises KeyError naming it.
--- solution
import re

_FIELD = re.compile(r"\{\{\s*([\w.]+)\s*(?:\|\s*(\w+)\s*)?\}\}")
_FILTERS = {"upper": str.upper, "lower": str.lower, "title": str.title}


def _lookup(values, dotted):
    cur = values
    for part in dotted.split("."):
        if not isinstance(cur, dict) or part not in cur:
            raise KeyError(dotted)
        cur = cur[part]
    return cur


def render(template, values):
    marker = "\x00"
    template = template.replace("{{{{", marker)

    def sub(m):
        value = str(_lookup(values, m.group(1)))
        if m.group(2):
            if m.group(2) not in _FILTERS:
                raise ValueError(f"unknown filter {m.group(2)}")
            value = _FILTERS[m.group(2)](value)
        return value

    return _FIELD.sub(sub, template).replace(marker, "{{")

=== run-length [text]
--- task

--- solution
import re


def rle_encode(s):
    if not s:
        return ""
    out, prev, count = [], s[0], 1
    for ch in s[1:]:
        if ch == prev:
            count += 1
        else:
            out.append(f"{count}{prev}")
            prev, count = ch, 1
    out.append(f"{count}{prev}")
    return "".join(out)


def rle_decode(s):
    if not re.fullmatch(r"(\d+\D)*", s):
        raise ValueError(f"malformed run-length string: {s!r}")
    return "".join(ch * int(n) for n, ch in re.findall(r"(\d+)(\D)", s))

=== caesar-vigenere [text]
--- task
Write `caesar(text, shift)` that shifts letters by `shift` positions (wrapping, keeping case, leaving other characters alone; negative shifts decode) and `vigenere(text, key, decrypt=False)` that applies a Vigenere cipher with an alphabetic key, advancing the key only on letters. An empty or non-alphabetic key raises ValueError.
--- solution
def _shift(ch, k):
    if ch.isupper():
        return chr((ord(ch) - 65 + k) % 26 + 65)
    if ch.islower():
        return chr((ord(ch) - 97 + k) % 26 + 97)
    return ch


def caesar(text, shift):
    return "".join(_shift(ch, shift) for ch in text)


def vigenere(text, key, decrypt=False):
    if not key or not key.isalpha():
        raise ValueError("key must be letters")
    shifts = [ord(c) - 97 for c in key.lower()]
    out, i = [], 0
    for ch in text:
        if ch.isalpha() and ch.isascii():
            k = shifts[i % len(shifts)]
            out.append(_shift(ch, -k if decrypt else k))
            i += 1
        else:
            out.append(ch)
    return "".join(out)

def group_anagrams(words):
    groups = {}
    for w in words:
        key = "".join(sorted(w.lower().replace(" ", "")))
        group = groups.setdefault(key, [])
        if w not in group:

    return list(groups.values())

=== balanced-brackets [text]
--- task

--- solution
PAIRS = {")": "(", "]": "[", "}": "{"}


def check_brackets(text):
    stack = []
    quote = None
    i = 0
    while i < len(text):
        ch = text[i]
        if quote:
            if ch == "\\":
                i += 2
                continue
            if ch == quote:
                quote = None
        elif ch in "'\"":
            quote = ch
        elif ch in "([{":
            stack.append((ch, i))
        elif ch in PAIRS:
            if not stack or stack[-1][0] != PAIRS[ch]:
                return i
            stack.pop()
        i += 1
    return stack[0][1] if stack else None

=== levenshtein-suggest [text]
--- task
Write `edit_distance(a, b)` (Levenshtein: insertions, deletions, substitutions cost 1) and `suggest(word, vocabulary, max_distance=2, limit=3)` that returns up to `limit` vocabulary words within `max_distance` of `word` (case-insensitive), closest first, ties alphabetical. An exact match returns just that word.
--- solution
def edit_distance(a, b):
    if len(a) < len(b):
        a, b = b, a
    prev = list(range(len(b) + 1))
    for i, ca in enumerate(a, 1):
        cur = [i]
        for j, cb in enumerate(b, 1):
            cur.append(min(prev[j] + 1, cur[j - 1] + 1, prev[j - 1] + (ca != cb)))
        prev = cur
    return prev[-1]


def suggest(word, vocabulary, max_distance=2, limit=3):
    w = word.lower()
    scored = []
    for v in vocabulary:
        d = edit_distance(w, v.lower())
        if d == 0:
            return [v]
        if d <= max_distance:
            scored.append((d, v.lower(), v))
    return [v for _, _, v in sorted(scored)[:limit]]

=== markdown-toc [text]
--- task
Write `toc(markdown)` that builds a table of contents from ATX headings ("#" to "###") as nested bullet lines: two spaces of indent per level below the top level found, each item a link "[Title](#slug)". The slug is lowercase, with spaces turned into "-" and characters other than letters, digits, "-" and "_" removed; repeated slugs get "-1", "-2" suffixes. Headings inside ``` fenced code blocks are ignored.
--- solution
import re


def _slug(title):
    s = re.sub(r"[^\w\- ]", "", title.strip().lower())
    return s.replace(" ", "-")


def toc(markdown):
    lines, in_code, seen = [], False, {}
    heads = []
    for line in markdown.splitlines():
        if line.lstrip().startswith("```"):
            in_code = not in_code
            continue
        m = re.match(r"^(#{1,3})\s+(.+?)\s*#*\s*$", line)
        if m and not in_code:
            heads.append((len(m.group(1)), m.group(2)))
    if not heads:
        return ""
    top = min(level for level, _ in heads)
    for level, title in heads:
        slug = _slug(title)
        n = seen.get(slug, 0)
        seen[slug] = n + 1
        if n:
            slug = f"{slug}-{n}"
        lines.append("  " * (level - top) + f"- [{title}](#{slug})")
    return "\n".join(lines)

=== strip-html [text]
--- task
Write `html_to_text(html)` that removes tags, drops the contents of <script> and <style> elements entirely, turns <br> and the end of <p>, <div>, <li> and <h1>-<h6> into newlines, unescapes HTML entities, collapses runs of spaces within a line and strips each line, and removes blank lines.

import html as _html
import re


def html_to_text(html):
    s = re.sub(r"(?is)<(script|style)\b.*?</\1\s*>", "", html)
    s = re.sub(r"(?i)<br\s*/?>", "\n", s)
    s = re.sub(r"(?i)</(p|div|li|h[1-6])\s*>", "\n", s)
    s = re.sub(r"<[^>]+>", "", s)
    s = _html.unescape(s)
    lines = [re.sub(r"[ \t\xa0]+", " ", line).strip() for line in s.split("\n")]
    return "\n".join(line for line in lines if line)

=== sentence-split [text]
--- task
Write `sentences(text)` that splits text into sentences ending in ".", "!" or "?" (possibly followed by closing quotes or brackets), when followed by whitespace and an uppercase letter, digit or quote, or by the end of the text. It must not split after common abbreviations (Mr., Mrs., Dr., e.g., i.e., etc., vs.) or after single capital initials such as "J.". Return stripped sentences; a trailing fragment without punctuation is kept.
--- solution
import re

ABBREV = {"mr.", "mrs.", "ms.", "dr.", "e.g.", "i.e.", "etc.", "vs.", "prof.", "st."}


def sentences(text):
    out, start = [], 0
    for m in re.finditer(r"[.!?]+['\")\]]*(?=\s+['\"(A-Z0-9]|\s*$)", text):
        end = m.end()
        last_word = text[start:m.start() + 1].split()[-1].lower() if text[start:m.start() + 1].split() else ""
        if m.group().startswith(".") and (last_word in ABBREV or re.fullmatch(r"[a-z]\.", last_word)):
            continue
        piece = text[start:end].strip()
        if piece:
            out.append(piece)
        start = end
    rest = text[start:].strip()
    if rest:
        out.append(rest)
    return out
--- tests

=== normalize-phone [text]
--- task
Write `normalize_phone(raw, default_country="1")` that turns phone numbers written in common ways ("(555) 123-4567", "555.123.4567", "+44 20 7946 0958", "0044 20 7946 0958", "1-555-123-4567") into E.164 form "+<country><number>". A number without a country prefix gets default_country. The result must have 8 to 15 digits after "+", otherwise ValueError; letters also raise ValueError.
--- solution
import re


def normalize_phone(raw, default_country="1"):
    s = raw.strip()
    if re.search(r"[A-Za-z]", s):
        raise ValueError(f"letters in phone number: {raw!r}")
    plus = s.startswith("+")
    digits = re.sub(r"\D", "", s)
    if plus:
        full = digits
    elif digits.startswith("00"):
        full = digits[2:]
    elif default_country == "1" and len(digits) == 11 and digits.startswith("1"):
        full = digits
    else:
        full = default_country + digits
    if not 8 <= len(full) <= 15:
        raise ValueError(f"bad phone number length: {raw!r}")
    return "+" + full

=== justify-text [text]
--- task
Write `justify(words, width)` that lays words out in fully justified lines of exactly `width` characters: extra spaces are spread as evenly as possible, with the leftmost gaps getting more. A line with one word, and the last line, are left-aligned and padded with spaces on the right. A word longer than width raises ValueError.
--- solution
def justify(words, width):
    if any(len(w) > width for w in words):
        raise ValueError("word longer than width")
    lines, cur, length = [], [], 0
    for w in words:
        if cur and length + len(cur) + len(w) > width:
            lines.append(cur)
            cur, length = [], 0
        cur.append(w)
        length += len(w)
    if cur:
        lines.append(cur)
    out = []
    for i, line in enumerate(lines):
        if i == len(lines) - 1 or len(line) == 1:
            out.append(" ".join(line).ljust(width))
            continue
        spaces = width - sum(len(w) for w in line)
        gaps = len(line) - 1
        q, r = divmod(spaces, gaps)
        s = ""
        for j, w in enumerate(line[:-1]):
            s += w + " " * (q + (1 if j < r else 0))
        out.append(s + line[-1])
    return out
--- tests

=== csv-table-markdown [text]
--- task
Write `markdown_table(rows, headers, align=None)` that renders a list of rows (lists of values) as a GitHub Markdown table with columns padded to equal width. `align` is an optional list of "l", "r" or "c" per column, shown in the separator row as ":---", "---:" or ":---:" and applied to the padding. Values are converted with str(); a "|" inside a value is escaped with a backslash. Rows with the wrong number of cells raise ValueError.
--- solution
def markdown_table(rows, headers, align=None):
    ncol = len(headers)
    align = align or ["l"] * ncol
    if len(align) != ncol or any(a not in "lrc" for a in align):
        raise ValueError("bad align")
    cells = [[str(h).replace("|", "\\|") for h in headers]]
    for r in rows:
        if len(r) != ncol:
            raise ValueError(f"row has {len(r)} cells, expected {ncol}")
        cells.append([str(v).replace("|", "\\|") for v in r])
    widths = [max(3, max(len(row[i]) for row in cells)) for i in range(ncol)]

    def pad(text, w, a):
        return text.rjust(w) if a == "r" else text.center(w) if a == "c" else text.ljust(w)

    def line(row):
        return "| " + " | ".join(pad(t, w, a) for t, w, a in zip(row, widths, align)) + " |"

    seps = []
    for w, a in zip(widths, align):
        dash = "-" * w
        seps.append(":" + dash[1:] if a == "l" else dash[:-1] + ":" if a == "r" else ":" + dash[2:] + ":")
    return "\n".join([line(cells[0]), "| " + " | ".join(seps) + " |"] + [line(r) for r in cells[1:]])
--- tests

def test_basic():
    out = markdown_table([["apple", 3], ["kiwi", 12]], ["fruit", "n"], align=["l", "r"])
    assert out == "| fruit |   n |\n| :---- | --: |\n| apple |   3 |\n| kiwi  |  12 |"

    assert markdown_table([[1]], ["x"]) == "| x   |\n| :-- |\n| 1   |"

    out = markdown_table([["ab"]], ["title"], align=["c"])
    assert out.splitlines()[1] == "| :---: |"
