def first_bad(n, is_bad):
    lo, hi = 0, n
    while lo < hi:
        mid = (lo + hi) // 2
        if is_bad(mid):
            hi = mid
        else:
            lo = mid + 1
    return lo


def floor_root(n, k):
    if n < 0 or k < 1:
        raise ValueError("need n >= 0 and k >= 1")
    lo, hi = 0, 1 << (n.bit_length() // k + 1)
    while lo < hi:
        mid = (lo + hi + 1) // 2
        if mid ** k <= n:
            lo = mid
        else:
            hi = mid - 1
    return lo

def longest_within_budget(sizes, budget):
    best_start, best_len = 0, 0
    start = total = 0
    for end, s in enumerate(sizes):
        total += s
        while total > budget and start <= end:
            total -= sizes[start]
            start += 1
        if end - start + 1 > best_len:
            best_start, best_len = start, end - start + 1
    return best_start, best_len


def _at_most(a, b, limit):
    j, count = len(b) - 1, 0
    for x in a:
        while j >= 0 and x + b[j] > limit:
            j -= 1
        count += j + 1
    return count


def count_pairs_between(a, b, lo, hi):
    if lo > hi:
        return 0
    return _at_most(a, b, hi) - _at_most(a, b, lo - 1)

import re

_RUNS = re.compile(r"(\d+)")


def natural_key(s):
    key = []
    for i, part in enumerate(_RUNS.split(s)):
        if not part:
            continue
        key.append((0, int(part), "") if i % 2 else (1, 0, part.lower()))
    return tuple(key)


def sort_by_spec(rows, spec):
    out = list(rows)
    for field, order in reversed(spec):
        if order not in ("asc", "desc"):
            raise ValueError(f"bad order {order!r}")
        out.sort(key=lambda r: r[field], reverse=order == "desc")
    return out

from collections import Counter
from decimal import Decimal
from itertools import combinations


def duplicate_groups(invoices):
    groups = {}
    for inv in invoices:
        key = (inv["customer"].strip().casefold(), Decimal(inv["amount"]).quantize(Decimal("0.01")), inv["date"])
        groups.setdefault(key, []).append(inv["id"])
    return [ids for ids in groups.values() if len(ids) > 1]


def top_pairs(baskets, k):
    counts = Counter()
    for basket in baskets:
        counts.update(combinations(sorted(set(basket)), 2))
    return sorted(counts.items(), key=lambda kv: (-kv[1], kv[0]))[:k]

def nearest_blocker(heights):
    stack, out = [], []
    for i, h in enumerate(heights):
        while stack and heights[stack[-1]] < h:
            stack.pop()
        out.append(stack[-1] if stack else -1)
        stack.append(i)
    return out


def _div(a, b):
    q = abs(a) // abs(b)
    return q if (a >= 0) == (b > 0) else -q


_OPS = {"+": lambda a, b: a + b, "-": lambda a, b: a - b, "*": lambda a, b: a * b, "/": _div}


def eval_postfix(tokens):
    stack = []

    def pop():
        if not stack:
            raise ValueError("stack underflow")
        return stack.pop()
    for tok in tokens:
        if tok in _OPS:
            b, a = pop(), pop()
            stack.append(_OPS[tok](a, b))
        elif tok == "dup":
            x = pop()
            stack += [x, x]
        elif tok == "swap":
            b, a = pop(), pop()
            stack += [b, a]
        elif tok == "drop":
            pop()
        else:
            try:
                stack.append(int(tok))
            except ValueError:
                raise ValueError(f"unknown token {tok!r}") from None
    if len(stack) != 1:
        raise ValueError(f"expected one result, stack has {len(stack)}")
    return stack[0]

from collections import deque


def min_moves(grid, start, goal):
    rows = len(grid)
    cols = len(grid[0]) if rows else 0
    for r, c in (start, goal):
        if not (0 <= r < rows and 0 <= c < cols) or grid[r][c] == "#":
            raise ValueError(f"bad cell {(r, c)}")
    pads = {}
    for r, row in enumerate(grid):
        for c, ch in enumerate(row):
            if ch.isdigit():
                pads.setdefault(ch, []).append((r, c))
    start, goal = tuple(start), tuple(goal)
    dist = {start: 0}
    queue = deque([start])
    while queue:
        cell = queue.popleft()
        d = dist[cell]
        if cell == goal:
            return d
        r, c = cell
        nxt = [(r - 1, c), (r + 1, c), (r, c - 1), (r, c + 1)]
        nxt += pads.pop(grid[r][c], [])
        for nr, nc in nxt:
            if 0 <= nr < rows and 0 <= nc < cols and grid[nr][nc] != "#" and (nr, nc) not in dist:
                dist[(nr, nc)] = d + 1
                queue.append((nr, nc))
    return -1

def brute(grid, start, goal):
    rows, cols = len(grid), len(grid[0])
    best = {start: 0}
    heap = [(0, start)]
    while heap:
        d, (r, c) = heapq.heappop(heap)
        if (r, c) == goal:
            return d
        if d > best[(r, c)]:
            continue
        nbrs = [(r - 1, c), (r + 1, c), (r, c - 1), (r, c + 1)]
        if grid[r][c].isdigit():
            nbrs += [(i, j) for i in range(rows) for j in range(cols) if grid[i][j] == grid[r][c]]
        for nr, nc in nbrs:
            if 0 <= nr < rows and 0 <= nc < cols and grid[nr][nc] != "#" and d + 1 < best.get((nr, nc), 1e9):
                best[(nr, nc)] = d + 1
                heapq.heappush(heap, (d + 1, (nr, nc)))
    return -1

def all_rosters(shifts, available, max_shifts):
    shifts = list(shifts)
    chosen, load = [], {}

    def extend(i):
        if i == len(shifts):
            yield dict(zip(shifts, chosen))
            return
        for w in sorted(set(available.get(shifts[i], ()))):
            if (chosen and chosen[-1] == w) or load.get(w, 0) >= max_shifts:
                continue
            chosen.append(w)
            load[w] = load.get(w, 0) + 1
            yield from extend(i + 1)
            chosen.pop()
            load[w] -= 1
    yield from extend(0)

import functools


def min_cost_memo(offers, need):
    offers = tuple(offers)
    if need > 0 and not offers:
        raise ValueError("no offers")

    @functools.lru_cache(maxsize=None)
    def cost(n):
        if n <= 0:
            return 0
        return min(price + cost(n - length) for length, price in offers)
    return cost(need)


def _table(offers, need):
    if need > 0 and not offers:
        raise ValueError("no offers")
    cost = [0] * (max(need, 0) + 1)
    choice = [-1] * len(cost)
    for n in range(1, len(cost)):
        best = None
        for idx, (length, price) in enumerate(offers):
            c = price + cost[max(n - length, 0)]
            if best is None or c < best:
                best, choice[n] = c, idx
        cost[n] = best
    return cost, choice


def min_cost_table(offers, need):
    return _table(offers, need)[0][max(need, 0)]


def cheapest_plan(offers, need):
    _, choice = _table(offers, need)
    plan, n = [], need
    while n > 0:
        plan.append(choice[n])
        n -= offers[choice[n]][0]
    return sorted(plan)

import heapq
import itertools


def cheapest_route(roads, start, goal):
    graph = {}
    for a, b, toll in roads:
        if toll < 0:
            raise ValueError(f"negative toll on {a!r} -> {b!r}")
        graph.setdefault(a, []).append((b, toll))
    tie = itertools.count()
    dist, prev, done = {start: 0}, {}, set()
    heap = [(0, next(tie), start)]
    while heap:
        d, _, u = heapq.heappop(heap)
        if u in done:
            continue
        done.add(u)
        if u == goal:
            path = [u]
            while path[-1] != start:
                path.append(prev[path[-1]])
            return d, path[::-1]
        for v, w in graph.get(u, ()):
            nd = d + w
            if v not in dist or nd < dist[v]:
                dist[v], prev[v] = nd, u
                heapq.heappush(heap, (nd, next(tie), v))
    return None

def bellman_ford(roads, start, goal):
    nodes = {start, goal} | {a for a, _, _ in roads} | {b for _, b, _ in roads}
    dist = {n: float("inf") for n in nodes}
    dist[start] = 0
    for _ in range(len(nodes)):
        for a, b, w in roads:
            if dist[a] + w < dist[b]:
                dist[b] = dist[a] + w
    return None if dist[goal] == float("inf") else dist[goal]

import heapq


class CycleError(ValueError):
    def __init__(self, cycle):
        super().__init__("dependency cycle: " + " -> ".join(cycle))
        self.cycle = cycle


def _graph(deps):
    nodes = set(deps)
    for ds in deps.values():
        nodes.update(ds)
    prereq = {n: set(deps.get(n, ())) for n in nodes}
    users = {n: [] for n in nodes}
    for n, ps in prereq.items():
        for p in ps:
            users[p].append(n)
    return prereq, users


def _find_cycle(remaining, prereq):
    node, index, path = min(remaining), {}, []
    while node not in index:
        index[node] = len(path)
        path.append(node)
        node = min(p for p in prereq[node] if p in remaining)
    return path[index[node]:] + [node]


def build_order(deps):
    prereq, users = _graph(deps)
    indegree = {n: len(ps) for n, ps in prereq.items()}
    ready = [n for n, d in indegree.items() if d == 0]
    heapq.heapify(ready)
    order = []
    while ready:
        n = heapq.heappop(ready)
        order.append(n)
        for u in users[n]:
            indegree[u] -= 1
            if indegree[u] == 0:
                heapq.heappush(ready, u)
    if len(order) < len(prereq):
        raise CycleError(_find_cycle(set(prereq) - set(order), prereq))
    return order


def parallel_stages(deps):
    prereq, _ = _graph(deps)
    stage = {}
    for n in build_order(deps):
        stage[n] = 1 + max((stage[p] for p in prereq[n]), default=-1)
    out = [[] for _ in range(max(stage.values(), default=-1) + 1)]
    for n in sorted(stage):
        out[stage[n]].append(n)
    return out

@@@ reference
class DisjointSet:
    def __init__(self, n):
        self.parent = list(range(n))
        self._size = [1] * n
        self.count = n

    def find(self, x):
        root = x
        while self.parent[root] != root:
            root = self.parent[root]
        while self.parent[x] != root:
            self.parent[x], x = root, self.parent[x]
        return root

    def union(self, a, b):
        ra, rb = self.find(a), self.find(b)
        if ra == rb:
            return False
        if self._size[ra] < self._size[rb]:
            ra, rb = rb, ra
        self.parent[rb] = ra
        self._size[ra] += self._size[rb]
        self.count -= 1
        return True

    def connected(self, a, b):
        return self.find(a) == self.find(b)

    def size(self, x):
        return self._size[self.find(x)]


def first_full_connection(n, cables):
    if n < 2:
        raise ValueError("need at least two offices")
    ds = DisjointSet(n)
    for i, (a, b) in enumerate(cables):
        ds.union(a, b)
        if ds.count == 1:
            return i
    return -1


def redundant_cables(n, cables):
    ds = DisjointSet(n)
    return [i for i, (a, b) in enumerate(cables) if not ds.union(a, b)]
