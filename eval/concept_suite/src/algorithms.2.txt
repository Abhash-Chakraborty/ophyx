import heapq


def _push(heap, k, entry):
    if len(heap) < k:
        heapq.heappush(heap, entry)
    elif entry[:2] > heap[0][:2]:
        heapq.heapreplace(heap, entry)


def _ranked(heap):
    return [e[2] for e in sorted(heap, key=lambda e: e[:2], reverse=True)]


def top_k(iterable, k, key=None):
    if k <= 0:
        return []
    heap = []
    for i, item in enumerate(iterable):
        _push(heap, k, (item if key is None else key(item), -i, item))
    return _ranked(heap)


def rolling_top_k(stream, k):
    heap = []
    for i, x in enumerate(stream):
        if k > 0:
            _push(heap, k, (x, -i, x))
        yield _ranked(heap)

def merge_bookings(bookings):
    items = sorted(bookings)
    merged = []
    for start, end in items:
        if start >= end:
            raise ValueError(f"empty or reversed booking {(start, end)}")
        if merged and start <= merged[-1][1]:
            if end > merged[-1][1]:
                merged[-1] = (merged[-1][0], end)
        else:
            merged.append((start, end))
    return merged


def common_free(calendars, day_start, day_end, min_len):
    busy = merge_bookings([b for cal in calendars for b in cal])
    free, cursor = [], day_start
    for start, end in busy + [(day_end, day_end)]:
        start = min(max(start, day_start), day_end)
        end = min(max(end, day_start), day_end)
        if start > cursor and start - cursor >= min_len:
            free.append((cursor, start))
        cursor = max(cursor, end)
    return free

def prefix_function(s):
    pi = [0] * len(s)
    for i in range(1, len(s)):
        k = pi[i - 1]
        while k and s[i] != s[k]:
            k = pi[k - 1]
        if s[i] == s[k]:
            k += 1
        pi[i] = k
    return pi


def find_all(text, pattern):
    if not pattern:
        raise ValueError("empty pattern")
    pi = prefix_function(pattern)
    hits, k, m = [], 0, len(pattern)
    for i, ch in enumerate(text):
        while k and ch != pattern[k]:
            k = pi[k - 1]
        if ch == pattern[k]:
            k += 1
        if k == m:
            hits.append(i - m + 1)
            k = pi[k - 1]
    return hits


def smallest_period(s):
    return len(s) - prefix_function(s)[-1] if s else 0

class Node:
    __slots__ = ("value", "next")

    def __init__(self, value, next=None):
        self.value = value
        self.next = next


class LinkedList:
    def __init__(self, iterable=()):
        self.head = self._tail = None
        self._len = 0
        for v in iterable:
            self.append(v)

    def append(self, value):
        node = Node(value)
        if self._tail is None:
            self.head = self._tail = node
        else:
            self._tail.next = node
            self._tail = node
        self._len += 1

    def appendleft(self, value):
        self.head = Node(value, self.head)
        if self._tail is None:
            self._tail = self.head
        self._len += 1

    def popleft(self):
        if self.head is None:
            raise IndexError("pop from empty list")
        node = self.head
        self.head = node.next
        if self.head is None:
            self._tail = None
        self._len -= 1
        return node.value

    def remove(self, value):
        prev, cur = None, self.head
        while cur is not None and cur.value != value:
            prev, cur = cur, cur.next
        if cur is None:
            raise ValueError(f"{value!r} not in list")
        if prev is None:
            self.head = cur.next
        else:
            prev.next = cur.next
        if cur is self._tail:
            self._tail = prev
        self._len -= 1

    def reverse(self):
        prev, cur = None, self.head
        self._tail = cur
        while cur is not None:
            cur.next, prev, cur = prev, cur, cur.next
        self.head = prev

    def __iter__(self):
        cur = self.head
        while cur is not None:
            yield cur.value
            cur = cur.next

    def __len__(self):
        return self._len

    def __repr__(self):
        return f"LinkedList({list(self)!r})"

from collections import deque


class TreeNode:
    def __init__(self, value, left=None, right=None):
        self.value = value
        self.left = left
        self.right = right


def preorder(root):
    out, stack = [], [root] if root else []
    while stack:
        node = stack.pop()
        out.append(node.value)
        if node.right:
            stack.append(node.right)
        if node.left:
            stack.append(node.left)
    return out


def inorder(root):
    out, stack, node = [], [], root
    while stack or node:
        while node:
            stack.append(node)
            node = node.left
        node = stack.pop()
        out.append(node.value)
        node = node.right
    return out


def postorder(root):
    out, stack = [], [root] if root else []
    while stack:
        node = stack.pop()
        out.append(node.value)
        if node.left:
            stack.append(node.left)
        if node.right:
            stack.append(node.right)
    return out[::-1]


def levels(root):
    out, queue = [], deque([root] if root else [])
    while queue:
        out.append([])
        for _ in range(len(queue)):
            node = queue.popleft()
            out[-1].append(node.value)
            queue.extend(c for c in (node.left, node.right) if c)
    return out


def height(root):
    return len(levels(root))

class _Node:
    __slots__ = ("key", "left", "right")

    def __init__(self, key):
        self.key, self.left, self.right = key, None, None


class BST:
    def __init__(self):
        self.root = None
        self._n = 0

    def insert(self, key):
        if self.root is None:
            self.root = _Node(key)
            self._n = 1
            return True
        cur = self.root
        while True:
            if key == cur.key:
                return False
            side = "left" if key < cur.key else "right"
            nxt = getattr(cur, side)
            if nxt is None:
                setattr(cur, side, _Node(key))
                self._n += 1
                return True
            cur = nxt

    def __contains__(self, key):
        cur = self.root
        while cur is not None and cur.key != key:
            cur = cur.left if key < cur.key else cur.right
        return cur is not None

    def delete(self, key):
        parent, cur = None, self.root
        while cur is not None and cur.key != key:
            parent, cur = cur, cur.left if key < cur.key else cur.right
        if cur is None:
            raise KeyError(key)
        if cur.left is not None and cur.right is not None:
            sp, s = cur, cur.right
            while s.left is not None:
                sp, s = s, s.left
            cur.key = s.key
            parent, cur = sp, s
        child = cur.left if cur.left is not None else cur.right
        if parent is None:
            self.root = child
        elif parent.left is cur:
            parent.left = child
        else:
            parent.right = child
        self._n -= 1

    def __len__(self):
        return self._n

    def __iter__(self):
        stack, node = [], self.root
        while stack or node:
            while node:
                stack.append(node)
                node = node.left
            node = stack.pop()
            yield node.key
            node = node.right

    def range(self, lo, hi):
        out, stack, node = [], [], self.root
        while stack or node:
            while node:
                stack.append(node)
                node = node.left if lo < node.key else None
            node = stack.pop()
            if node.key >= hi:
                break
            if node.key >= lo:
                out.append(node.key)
            node = node.right
        return out

    def floor(self, x):
        best, cur = None, self.root
        while cur is not None:
            if cur.key <= x:
                best, cur = cur.key, cur.right
            else:
                cur = cur.left
        return best

    def ceiling(self, x):
        best, cur = None, self.root
        while cur is not None:
            if cur.key >= x:
                best, cur = cur.key, cur.left
            else:
                cur = cur.right
        return best

    def height(self):
        h, level = 0, [self.root] if self.root else []

def _check(m):
    if not m or not m[0] or any(len(r) != len(m[0]) for r in m):
        raise ValueError("matrix must be non-empty and rectangular")


def identity(n):
    return [[int(i == j) for j in range(n)] for i in range(n)]


def rotate(m, quarter_turns=1):
    _check(m)
    out = [list(r) for r in m]
    for _ in range(quarter_turns % 4):
        out = [list(row) for row in zip(*out[::-1])]
    return out


def matmul(a, b, mod=None):
    _check(a)
    _check(b)
    if len(a[0]) != len(b):
        raise ValueError(f"cannot multiply {len(a)}x{len(a[0])} by {len(b)}x{len(b[0])}")
    cols = list(zip(*b))
    out = [[sum(x * y for x, y in zip(row, col)) for col in cols] for row in a]
    if mod is not None:
        out = [[v % mod for v in row] for row in out]
    return out


def mat_pow(m, k, mod=None):
    _check(m)
    if len(m) != len(m[0]) or k < 0:
        raise ValueError("need a square matrix and k >= 0")
    result, base = identity(len(m)), [list(r) for r in m]
    while k:
        if k & 1:
            result = matmul(result, base, mod)
        base = matmul(base, base, mod)
        k >>= 1
    return result

def _check(n):
    if n < 0:
        raise ValueError("expected a non-negative int")


def popcount(n):
    _check(n)
    return bin(n).count("1")


def iter_bits(n):
    _check(n)
    while n:
        low = n & -n
        yield low.bit_length() - 1
        n &= n - 1


def lowest_set_bit(n):
    _check(n)
    return n & -n


def submasks(mask):
    _check(mask)
    out, s = [], mask
    while True:
        out.append(s)
        if s == 0:
            return out
        s = (s - 1) & mask


def gray_code(bits):
    _check(bits)
    return [i ^ (i >> 1) for i in range(1 << bits)]

def max_workshops(requests):
    """Choose a maximum set of pairwise compatible half-open intervals.

    Proof sketch (exchange argument): let g1 be the request that finishes first. Take any optimal
    schedule O and let o1 be its earliest-finishing request. Since g1 ends no later than o1, we can
    exchange o1 for g1 in O without creating an overlap, so some optimal schedule starts with g1.
    Removing g1 and every request that starts before it ends leaves the same problem on a smaller
    input, and induction shows the greedy choice at every step stays within an optimal schedule.
    """
    order = sorted(range(len(requests)), key=lambda i: (requests[i][1], requests[i][0], i))
    chosen, free_at = [], None
    for i in order:
        start, end = requests[i]
        if free_at is None or start >= free_at:
            chosen.append(i)
            free_at = end
    return chosen


def greedy_counterexample(coins, limit):
    """Return the smallest amount <= limit where greedy change-making is not optimal, else None.

    Greedy is only optimal for "canonical" coin systems. Taking the largest coin can block a
    better combination of smaller coins: with [1, 3, 4] greedy pays 6 as 4 + 1 + 1 (three coins)
    while 3 + 3 needs two. So the greedy count is compared with an exact DP optimum per amount.
    """
    coins = sorted(set(coins), reverse=True)
    if 1 not in coins:
        raise ValueError("coin system must contain 1")
    best = [0] * (limit + 1)
    for amount in range(1, limit + 1):
        best[amount] = 1 + min(best[amount - c] for c in coins if c <= amount)
        rest, used = amount, 0
        for c in coins:
            used += rest // c
            rest %= c
        if used > best[amount]:
            return amount
    return None

from itertools import accumulate


class Grid2DSum:
    def __init__(self, matrix):
        self.rows = len(matrix)
        self.cols = len(matrix[0]) if self.rows else 0
        p = [[0] * (self.cols + 1) for _ in range(self.rows + 1)]
        for r in range(self.rows):
            run = 0
            for c in range(self.cols):
                run += matrix[r][c]
                p[r + 1][c + 1] = p[r][c + 1] + run
        self._p = p

    def query(self, r1, c1, r2, c2):
        if not (0 <= r1 <= r2 < self.rows and 0 <= c1 <= c2 < self.cols):
            raise IndexError("bad rectangle")
        p = self._p
        return p[r2 + 1][c2 + 1] - p[r1][c2 + 1] - p[r2 + 1][c1] + p[r1][c1]


def occupancy(n_nights, stays):
    diff = [0] * (n_nights + 1)
    for checkin, checkout, guests in stays:
        if not 0 <= checkin < checkout <= n_nights:
            raise ValueError(f"bad stay {(checkin, checkout)}")
        diff[checkin] += guests
        diff[checkout] -= guests
    return list(accumulate(diff[:n_nights]))

from collections import Counter
from math import factorial


def distinct_permutations(items):
    a = sorted(items)
    while True:
        yield tuple(a)
        i = len(a) - 2
        while i >= 0 and a[i] >= a[i + 1]:
            i -= 1
        if i < 0:
            return
        j = len(a) - 1
        while a[j] <= a[i]:
            j -= 1
        a[i], a[j] = a[j], a[i]
        a[i + 1:] = reversed(a[i + 1:])


def _multinomial(counts):
    total = factorial(sum(counts.values()))
    for c in counts.values():
        total //= factorial(c)
    return total


def count_distinct_permutations(items):
    return _multinomial(Counter(items))


def nth_permutation(items, k):
    pool = Counter(items)
    if not 0 <= k < _multinomial(pool):
        raise IndexError("k out of range")
    out = []
    for _ in range(len(items)):
        for x in sorted(pool):
            if pool[x] == 0:
                continue
            pool[x] -= 1
            c = _multinomial(pool)
            if k < c:
                out.append(x)
                break
            k -= c
            pool[x] += 1
    return tuple(out)

    assert p == nth_permutation(items[::-1], k)
    assert nth_permutation(items, 0) == tuple(items) and nth_permutation(items, math.factorial(20) - 1) == tuple(items[::-1])
    assert nth_permutation("aabbc", 29) == tuple("cbbaa")


def test_nth_out_of_range():
    with pytest.raises(IndexError):
        nth_permutation("abc", 6)
    with pytest.raises(IndexError):
        nth_permutation("abc", -1)
    assert nth_permutation([], 0) == ()
