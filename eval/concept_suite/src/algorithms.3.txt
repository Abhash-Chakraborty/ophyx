def best_lineup(talks, minutes):
    if minutes < 0 or any(d <= 0 for d, _ in talks):
        raise ValueError("minutes must be >= 0 and durations positive")
    best = [0] * (minutes + 1)
    took = []
    for duration, score in talks:
        row = bytearray(minutes + 1)
        for t in range(minutes, duration - 1, -1):
            cand = best[t - duration] + score
            if cand > best[t]:
                best[t] = cand
                row[t] = 1
        took.append(row)
    chosen, t = [], minutes
    for i in range(len(talks) - 1, -1, -1):
        if took[i][t]:
            chosen.append(i)
            t -= talks[i][0]
    return best[minutes], sorted(chosen)

import heapq


def best_route(roads, start, goal):
    graph = {}
    for a, b, cost in roads:
        if cost < 0:
            raise ValueError(f"negative cost on {a} -> {b}")
        graph.setdefault(a, []).append((b, cost))
    heap = [(0, 0, (start,))]
    settled = set()
    while heap:
        cost, hops, path = heapq.heappop(heap)
        node = path[-1]
        if node in settled:
            continue
        settled.add(node)
        if node == goal:
            return cost, list(path)
        for nxt, w in graph.get(node, ()):
            if nxt not in settled:
                heapq.heappush(heap, (cost + w, hops + 1, path + (nxt,)))
    return None

def brute(roads, start, goal):
    graph = {}
    for a, b, w in roads:
        graph.setdefault(a, []).append((b, w))
    best = None

    def dfs(path, cost):
        nonlocal best
        node = path[-1]
        if node == goal:
            key = (cost, len(path), path)
            if best is None or key < best:
                best = key
            return
        for nxt, w in graph.get(node, ()):
            if nxt not in path:
                dfs(path + [nxt], cost + w)
    dfs([start], 0)
    return None if best is None else (best[0], best[2])

def connection_times(n, cables, queries):
    parent = list(range(n))
    size = [1] * n
    stamp = [-1] * n

    def root(x):
        while parent[x] != x:
            x = parent[x]
        return x

    for i, (a, b) in enumerate(cables):
        ra, rb = root(a), root(b)
        if ra == rb:
            continue
        if size[ra] < size[rb]:
            ra, rb = rb, ra
        parent[rb] = ra
        size[ra] += size[rb]
        stamp[rb] = i

    def chain(x):
        out = [x]
        while parent[x] != x:
            x = parent[x]
            out.append(x)
        return out

    answers = []
    for a, b in queries:
        if a == b:
            answers.append(-1)
            continue
        pa, pb = chain(a), chain(b)
        if pa[-1] != pb[-1]:
            answers.append(None)
            continue
        on_b = set(pb)
        best = -1
        for x in pa:
            if x in on_b:
                lca = x
                break
            best = max(best, stamp[x])
        for x in pb:
            if x == lca:
                break
            best = max(best, stamp[x])
        answers.append(best)
    return answers

from collections import deque


def critical_path(tasks):
    users = {name: [] for name in tasks}
    indegree = {}
    for name, (duration, deps) in tasks.items():
        if duration <= 0:
            raise ValueError(f"duration of {name} must be positive")
        indegree[name] = len(set(deps))
        for d in set(deps):
            if d not in tasks:
                raise ValueError(f"unknown dependency {d!r} of {name!r}")
            users[d].append(name)
    queue = deque(sorted(n for n, k in indegree.items() if k == 0))
    finish, seen = {}, 0
    while queue:
        n = queue.popleft()
        seen += 1
        duration, deps = tasks[n]
        finish[n] = duration + max((finish[d] for d in deps), default=0)
        for u in users[n]:
            indegree[u] -= 1
            if indegree[u] == 0:
                queue.append(u)
    if seen < len(tasks):
        raise ValueError("dependency cycle")
    if not tasks:
        return 0, []
    length = max(finish.values())
    cur = min(n for n, f in finish.items() if f == length)
    chain = [cur]
    while tasks[cur][1]:
        start = finish[cur] - tasks[cur][0]
        cur = min(d for d in tasks[cur][1] if finish[d] == start)
        chain.append(cur)
    return length, chain[::-1]

from collections import deque


class OrgChart:
    def __init__(self, managers):
        roots = [e for e, m in managers.items() if m is None]
        if len(roots) != 1:
            raise ValueError("need exactly one root")
        self._ids = {e: i for i, e in enumerate(managers)}
        self._names = list(managers)
        n = len(self._names)
        parent = [0] * n
        children = [[] for _ in range(n)]
        for e, m in managers.items():
            i = self._ids[e]
            if m is None:
                parent[i] = i
                continue
            if m not in self._ids:
                raise ValueError(f"unknown manager {m!r}")
            parent[i] = self._ids[m]
            children[parent[i]].append(i)
        root = self._ids[roots[0]]
        self._depth = [-1] * n
        self._depth[root] = 0
        queue = deque([root])
        while queue:
            x = queue.popleft()
            for c in children[x]:
                self._depth[c] = self._depth[x] + 1
                queue.append(c)
        if -1 in self._depth:
            raise ValueError("cycle in management chain")
        self._parent = parent
        self._up = [parent]
        for _ in range(1, max(1, max(self._depth).bit_length())):
            prev = self._up[-1]
            self._up.append([prev[prev[i]] for i in range(n)])

    def depth(self, e):
        return self._depth[self._ids[e]]

    def _lca(self, a, b):
        if self._depth[a] < self._depth[b]:
            a, b = b, a
        diff, k = self._depth[a] - self._depth[b], 0
        while diff:
            if diff & 1:
                a = self._up[k][a]
            diff >>= 1
            k += 1
        if a == b:
            return a
        for row in reversed(self._up):
            if row[a] != row[b]:
                a, b = row[a], row[b]
        return self._parent[a]

    def lowest_common_manager(self, a, b):
        return self._names[self._lca(self._ids[a], self._ids[b])]

    def chain(self, a, b):
        ia, ib = self._ids[a], self._ids[b]
        top = self._lca(ia, ib)

        def climb(x):
            out = [x]
            while x != top:
                x = self._parent[x]
                out.append(x)
            return out
        up_part, down_part = climb(ia), climb(ib)
        return [self._names[i] for i in up_part + down_part[-2::-1]]
