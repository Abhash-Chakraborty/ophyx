def run_in_threads(fn, args_list):

class Registry:
    def __init__(self):
        self._lock = threading.RLock()
        self._items = {}

    def register(self, name, value):

            if name in self._items:

import threading


class CountDownLatch:
    def __init__(self, count: int):
        if count < 0:
            raise ValueError("count must be >= 0")
        self._count = count
        self._cond = threading.Condition()

    @property
    def count(self):
        with self._cond:
            return self._count

    def count_down(self):
        with self._cond:
            if self._count > 0:
                self._count -= 1
                if self._count == 0:
                    self._cond.notify_all()

    def wait(self, timeout=None):
        with self._cond:
            return self._cond.wait_for(lambda: self._count == 0, timeout)

import threading


class ConcurrencyLimiter:
    def __init__(self, max_concurrent: int):
        self._sem = threading.BoundedSemaphore(max_concurrent)
        self._lock = threading.Lock()
        self._active = 0
        self._peak = 0

    @property
    def active(self):
        with self._lock:
            return self._active

    @property
    def peak(self):
        with self._lock:
            return self._peak

    def run(self, fn, *args, **kwargs):
        with self._sem:
            with self._lock:
                self._active += 1
                self._peak = max(self._peak, self._active)
            try:
                return fn(*args, **kwargs)
            finally:
                with self._lock:
                    self._active -= 1

import queue
import threading

_STOP = object()


def run_pipeline(items, worker, n_workers=4):
    q = queue.Queue()
    results, errors = {}, {}
    lock = threading.Lock()

    def consume():
        while True:
            item = q.get()
            try:
                if item is _STOP:
                    return
                try:
                    r = worker(item)
                except Exception as exc:
                    with lock:
                        errors[item] = exc
                else:
                    with lock:
                        results[item] = r
            finally:
                q.task_done()

    threads = [threading.Thread(target=consume, daemon=True) for _ in range(n_workers)]
    for t in threads:
        t.start()
    for item in items:
        q.put(item)
    for _ in threads:
        q.put(_STOP)
    for t in threads:
        t.join()
    return results, errors

import threading
import time
from collections import deque


class Closed(Exception):
    pass


class BoundedBuffer:
    def __init__(self, capacity):
        self.capacity = capacity
        self._items = deque()
        self._closed = False
        self._cond = threading.Condition()

    def _wait(self, pred, timeout):
        if not self._cond.wait_for(pred, timeout):
            raise TimeoutError

    def put(self, item, timeout=None):
        with self._cond:
            self._wait(lambda: self._closed or len(self._items) < self.capacity, timeout)
            if self._closed:
                raise Closed
            self._items.append(item)
            self._cond.notify_all()

    def get(self, timeout=None):
        with self._cond:
            self._wait(lambda: self._closed or self._items, timeout)
            if self._items:
                item = self._items.popleft()
                self._cond.notify_all()
                return item
            raise Closed

    def close(self):
        with self._cond:
            self._closed = True
            self._cond.notify_all()

    def __len__(self):
        with self._cond:
            return len(self._items)

def fetch_all(urls, fetch, max_workers=8):
    unique = list(dict.fromkeys(urls))
    out = {}
    with ThreadPoolExecutor(max_workers=max_workers) as ex:
        futs = {ex.submit(fetch, u): u for u in unique}

            u = futs[fut]
            try:
                out[u] = fut.result()
            except Exception as exc:
                out[u] = exc

import math
from concurrent.futures import ProcessPoolExecutor


def _is_prime(n):
    if n < 2:
        return False
    if n % 2 == 0:
        return n == 2
    for d in range(3, math.isqrt(n) + 1, 2):
        if n % d == 0:
            return False
    return True


def count_primes_in_range(lo, hi):
    return sum(1 for n in range(max(lo, 0), hi) if _is_prime(n))


def count_primes_parallel(limit, chunks=4, workers=2):
    if limit <= 2:
        return 0
    chunks = max(1, min(chunks, limit))
    step = math.ceil(limit / chunks)
    bounds = [(i, min(i + step, limit)) for i in range(0, limit, step)]
    with ProcessPoolExecutor(max_workers=workers) as ex:
        return sum(ex.map(count_primes_in_range, *zip(*bounds)))

import multiprocessing
from collections import Counter


def _count_words(text):
    return Counter(text.lower().split())


def parallel_word_count(texts, processes=2):
    total = Counter()
    with multiprocessing.Pool(processes) as pool:
        for c in pool.map(_count_words, list(texts)):
            total.update(c)
    return total


def _bump(value, n):
    for _ in range(n):
        with value.get_lock():
            value.value += 1


def shared_counter(n_procs, increments):
    value = multiprocessing.Value("i", 0)
    procs = [multiprocessing.Process(target=_bump, args=(value, increments)) for _ in range(n_procs)]
    for p in procs:
        p.start()
    for p in procs:
        p.join()
    return value.value

import asyncio


async def graceful_cancel(tasks, timeout: float):
    tasks = list(tasks)
    finished = [t for t in tasks if t.done()]
    pending = [t for t in tasks if not t.done()]
    for t in pending:
        t.cancel()
    if pending:
        await asyncio.wait(pending, timeout=timeout)
    return {
        "finished": finished,
        "cancelled": [t for t in pending if t.done()],
        "stuck": [t for t in pending if not t.done()],
    }
