import asyncio


async def crawl(start, get_links, max_workers=3, max_pages=100):
    queue = asyncio.Queue()
    seen = {start}
    await queue.put(start)

    async def worker():
        while True:
            page = await queue.get()
            try:
                try:
                    links = await get_links(page)
                except Exception:
                    links = []
                for link in links:
                    if link not in seen and len(seen) < max_pages:
                        seen.add(link)
                        queue.put_nowait(link)
            finally:
                queue.task_done()

    workers = [asyncio.create_task(worker()) for _ in range(max_workers)]
    try:
        await queue.join()
    finally:
        for w in workers:
            w.cancel()
        await asyncio.gather(*workers, return_exceptions=True)
    return seen

import asyncio

_DONE = object()


async def pipeline(source, transform, sink, queue_size=2):
    q = asyncio.Queue(maxsize=queue_size)

    async def produce():
        async for item in source:
            await q.put(item)
        await q.put(_DONE)

    producer = asyncio.create_task(produce())
    count = 0
    try:

            if item is _DONE:

            await sink(await transform(item))
            count += 1

        producer.cancel()
        await asyncio.gather(producer, return_exceptions=True)
        raise
    await producer
    return count

import asyncio


async def abatch(aiterable, size):
    if size < 1:
        raise ValueError("size must be >= 1")
    batch = []
    async for item in aiterable:
        batch.append(item)
        if len(batch) == size:
            yield batch
            batch = []
    if batch:
        yield batch


class Ticker:
    def __init__(self, n, interval, sleep=asyncio.sleep):
        self.n = n
        self.interval = interval
        self.sleep = sleep
        self._i = 0

    def __aiter__(self):
        self._i = 0
        return self

    async def __anext__(self):
        if self._i >= self.n:
            raise StopAsyncIteration
        if self._i > 0:
            await self.sleep(self.interval)
        self._i += 1
        return self._i - 1

class Account:
    def __init__(self, id: int, balance: int):
        self.id = id
        self.balance = balance
        self.lock = threading.Lock()


def transfer(src, dst, amount):
    if src is dst or src.id == dst.id:
        raise ValueError("cannot transfer to the same account")
    first, second = (src, dst) if src.id < dst.id else (dst, src)
    with first.lock:
        with second.lock:
            if src.balance < amount:
                raise ValueError("insufficient funds")
            src.balance -= amount
            dst.balance += amount

import time
from contextlib import contextmanager


@contextmanager
def acquire_all(locks, timeout: float):
    ordered = sorted({id(l): l for l in locks}.values(), key=id)
    deadline = time.monotonic() + timeout
    taken = []
    try:
        for lock in ordered:
            remaining = max(0.0, deadline - time.monotonic())
            if not lock.acquire(timeout=remaining):
                raise TimeoutError("could not acquire all locks")
            taken.append(lock)
        yield
    finally:
        for lock in reversed(taken):
            lock.release()

import threading

_UNSET = object()


class Lazy:
    def __init__(self, factory):
        self._factory = factory
        self._value = _UNSET
        self._lock = threading.Lock()

    @property
    def initialized(self):
        return self._value is not _UNSET

    def get(self):
        if self._value is _UNSET:
            with self._lock:
                if self._value is _UNSET:
                    self._value = self._factory()
        return self._value

import threading


class SingletonMeta(type):
    _instances = {}
    _lock = threading.Lock()

    def __call__(cls, *args, **kwargs):
        inst = SingletonMeta._instances.get(cls)
        if inst is None:
            with SingletonMeta._lock:
                inst = SingletonMeta._instances.get(cls)
                if inst is None:
                    inst = super().__call__(*args, **kwargs)
                    SingletonMeta._instances[cls] = inst
        return inst

    def reset(cls):
        with SingletonMeta._lock:
            SingletonMeta._instances.pop(cls, None)

Write a `WorkerPool(n_workers, handler)` that starts `n_workers` threads taking items from a `queue.Queue` and calling `handler(item)`. `submit(item)` enqueues work and raises `RuntimeError` after shutdown has begun. Exceptions from `handler` are appended to the `errors` list and do not kill the worker. `shutdown(drain=True) -> int` stops accepting work and stops the workers gracefully: with `drain=True` every queued item is processed first; with `drain=False` items not yet started are discarded, while items already being handled finish. It joins all worker threads and returns the number of discarded items. `processed` counts handler calls that completed (successfully or not).
@@@ reference
import queue
import threading

_STOP = object()


class WorkerPool:
    def __init__(self, n_workers, handler):
        self.handler = handler
        self.errors = []
        self.processed = 0
        self._q = queue.Queue()
        self._lock = threading.Lock()
        self._closed = False
        self._threads = [threading.Thread(target=self._run, daemon=True) for _ in range(n_workers)]
        for t in self._threads:
            t.start()

    def _run(self):
        while True:
            item = self._q.get()
            if item is _STOP:
                return
            try:
                self.handler(item)
            except Exception as exc:
                with self._lock:
                    self.errors.append(exc)
            finally:
                with self._lock:
                    self.processed += 1

    def submit(self, item):
        with self._lock:
            if self._closed:
                raise RuntimeError("pool is shut down")
            self._q.put(item)

    def shutdown(self, drain=True):
        with self._lock:
            self._closed = True
        discarded = 0
        if not drain:
            while True:
                try:
                    item = self._q.get_nowait()
                except queue.Empty:
                    break
                if item is not _STOP:
                    discarded += 1
        for _ in self._threads:
            self._q.put(_STOP)
        for t in self._threads:
            t.join()
        return discarded

Write an `AsyncWorkerPool(n_workers, handler)` for asyncio, where `handler(item)` is a coroutine function. `await pool.start()` creates the worker tasks, which take items from an `asyncio.Queue`. `await pool.submit(item)` enqueues an item. `await pool.stop(timeout)` shuts down gracefully: it waits up to `timeout` seconds for the queue to be fully processed (`queue.join()`), then cancels the workers and waits for them to exit; it returns True if everything was processed within the timeout and False otherwise. Handler exceptions are appended to `pool.errors` and the worker keeps going. `pool.processed` counts finished handler calls.
@@@ reference
import asyncio


class AsyncWorkerPool:
    def __init__(self, n_workers, handler):
        self.n_workers = n_workers
        self.handler = handler
        self.errors = []
        self.processed = 0
        self._queue = None
        self._workers = []

    async def start(self):
        self._queue = asyncio.Queue()
        self._workers = [asyncio.create_task(self._run()) for _ in range(self.n_workers)]

    async def _run(self):
        while True:
            item = await self._queue.get()
            try:
                await self.handler(item)
            except Exception as exc:
                self.errors.append(exc)
            finally:
                self.processed += 1
                self._queue.task_done()

    async def submit(self, item):
        await self._queue.put(item)

    async def stop(self, timeout):
        try:
            await asyncio.wait_for(self._queue.join(), timeout)
            drained = True
        except asyncio.TimeoutError:
            drained = False
        for w in self._workers:
            w.cancel()
        await asyncio.gather(*self._workers, return_exceptions=True)
        return drained
