from pathlib import Path


def index_tree(root):
    root = Path(root)
    out = {}
    for p in root.rglob("*"):
        rel = p.relative_to(root)
        if any(part.startswith(".") for part in rel.parts) or not p.is_file():
            continue
        out.setdefault(p.suffix.lower(), []).append(rel.as_posix())
    for names in out.values():
        names.sort()
    return out


def relocate(path, old_root, new_root, new_suffix=None):
    rel = Path(path).relative_to(old_root)
    target = Path(new_root) / rel
    if new_suffix is not None:
        target = target.with_suffix(new_suffix)
    return target

class ExtendedEncoder(json.JSONEncoder):
    def default(self, o):
        if isinstance(o, datetime.datetime):
            return {"__type__": "datetime", "value": o.isoformat()}
        if isinstance(o, datetime.date):

import csv


def normalize_csv(src, dst, columns):
    with open(src, newline="", encoding="utf-8-sig") as fin, \
            open(dst, "w", newline="", encoding="utf-8") as fout:
        reader = csv.reader(fin)
        header = [h.strip().lower() for h in next(reader, [])]
        writer = csv.writer(fout)
        writer.writerow(columns)
        written = 0
        for row in reader:
            rec = dict(zip(header, row))
            out = [rec.get(c, "") for c in columns]
            if not any(v.strip() for v in out):
                continue
            writer.writerow(out)
            written += 1
    return written

from zoneinfo import ZoneInfo


def to_zone(timestamp, zone):
    dt = datetime.fromisoformat(timestamp)
    if dt.tzinfo is None:

    return dt.astimezone(ZoneInfo(zone)).isoformat()

import re
import shutil
import tempfile
from pathlib import Path

_SNAP = re.compile(r"snap-(\d+)")


def _snapshots(dest_root):
    found = []
    for p in Path(dest_root).iterdir():
        m = _SNAP.fullmatch(p.name)
        if m and p.is_dir():
            found.append((int(m.group(1)), p))
    return sorted(found)


def snapshot(src, dest_root, *, ignore=("*.tmp", "__pycache__")):
    dest_root = Path(dest_root)
    dest_root.mkdir(parents=True, exist_ok=True)
    existing = _snapshots(dest_root)
    target = dest_root / f"snap-{existing[-1][0] + 1 if existing else 1}"
    staging = Path(tempfile.mkdtemp(prefix=".staging-", dir=dest_root))
    try:
        shutil.copytree(src, staging, ignore=shutil.ignore_patterns(*ignore), dirs_exist_ok=True)
        staging.rename(target)
    except BaseException:
        shutil.rmtree(staging, ignore_errors=True)
        raise
    return target


def prune(dest_root, keep):
    if keep < 0:
        raise ValueError("keep must be >= 0")
    snaps = _snapshots(dest_root)
    doomed = [p for _, p in snaps[:max(len(snaps) - keep, 0)]]
    for p in doomed:
        shutil.rmtree(p)
    return doomed

import hashlib
import hmac
import secrets


def hash_password(password, *, salt=None, iterations=100_000):
    if salt is None:
        salt = secrets.token_bytes(16)
    key = hashlib.pbkdf2_hmac("sha256", password.encode("utf-8"), salt, iterations)
    return f"pbkdf2_sha256${iterations}${salt.hex()}${key.hex()}"


def verify_password(password, stored):
    try:
        algo, iterations, salt_hex, key_hex = stored.split("$")
        if algo != "pbkdf2_sha256" or int(iterations) < 1:
            return False
        key = hashlib.pbkdf2_hmac("sha256", password.encode("utf-8"), bytes.fromhex(salt_hex), int(iterations))
        expected = bytes.fromhex(key_hex)
    except ValueError:
        return False
    return hmac.compare_digest(key, expected)


def new_token(nbytes=32):
    return secrets.token_urlsafe(nbytes)


def file_sha256(path, chunk_size=65536):
    h = hashlib.sha256()
    with open(path, "rb") as f:
        for block in iter(lambda: f.read(chunk_size), b""):
            h.update(block)
    return h.hexdigest()

import math
import statistics


def describe(values):
    vals = list(values)
    if not vals:
        raise ValueError("no values")
    many = len(vals) >= 2
    return {
        "count": len(vals),
        "mean": statistics.fmean(vals),
        "median": statistics.median(vals),
        "stdev": statistics.stdev(vals) if many else None,
        "p90": statistics.quantiles(vals, n=10, method="inclusive")[-1] if many else None,
        "total": math.fsum(vals),
    }


def zscores(values):
    vals = list(values)
    mu = statistics.fmean(vals)
    sigma = statistics.pstdev(vals, mu)
    if sigma == 0:
        return [0.0] * len(vals)
    return [(v - mu) / sigma for v in vals]


def growth_rate(values):
    vals = list(values)
    if len(vals) < 2:
        raise ValueError("need at least two values")
    return statistics.geometric_mean([b / a for a, b in zip(vals, vals[1:])]) - 1

import difflib
import re
import string
import textwrap


def format_paragraphs(text, width, indent=""):
    paras = [p for p in re.split(r"\n\s*\n", textwrap.dedent(text)) if p.strip()]
    return "\n\n".join(
        textwrap.fill(" ".join(p.split()), width=width, initial_indent=indent, subsequent_indent=indent,
                      break_long_words=False, break_on_hyphens=False)
        for p in paras)


def fill_template(template, values):
    return string.Template(template).substitute(values)


def suggest(word, choices, n=3):
    return difflib.get_close_matches(word, choices, n=n)


def diff_lines(old, new, name):
    return list(difflib.unified_diff(old.splitlines(), new.splitlines(), fromfile=f"a/{name}",
                                     tofile=f"b/{name}", lineterm=""))

from urllib.parse import parse_qsl, urlencode, urlsplit, urlunsplit

DEFAULT_PORTS = {"http": 80, "https": 443}


def canonicalize(url):
    parts = urlsplit(url)
    scheme = parts.scheme.lower()
    netloc = parts.hostname or ""
    if parts.port is not None and DEFAULT_PORTS.get(scheme) != parts.port:
        netloc += f":{parts.port}"
    pairs = sorted(parse_qsl(parts.query, keep_blank_values=True), key=lambda kv: kv[0])
    return urlunsplit((scheme, netloc, parts.path or "/", urlencode(pairs), ""))


def _expand(key, value):
    if value is None:
        return []
    if isinstance(value, (list, tuple)):
        return [(key, str(v)) for v in value]
    return [(key, str(value))]


def with_query(url, params):
    parts = urlsplit(url)
    out, done = [], set()
    for k, v in parse_qsl(parts.query, keep_blank_values=True):
        if k in params:
            if k not in done:
                done.add(k)
                out.extend(_expand(k, params[k]))
        else:
            out.append((k, v))
    for k, v in params.items():
        if k not in done:
            out.extend(_expand(k, v))
    return urlunsplit(parts._replace(query=urlencode(out)))

A value out of range for its field, or of the wrong type, raises `ValueError` chained from the `struct.error`. `decode_batch(data) -> list[tuple]` reverses it and raises `ValueError` for a short header, a wrong magic, an unsupported version, or a total length that does not match the count exactly.
@@@ reference
import struct

MAGIC = b"SNSR"
HEADER = struct.Struct("<4sBI")
RECORD = struct.Struct("<HIdB")


def encode_batch(records):
    records = list(records)
    try:
        body = b"".join(RECORD.pack(*r) for r in records)
    except struct.error as exc:
        raise ValueError(f"bad record: {exc}") from exc
    return HEADER.pack(MAGIC, 1, len(records)) + body


def decode_batch(data):
    if len(data) < HEADER.size:
        raise ValueError("truncated header")
    magic, version, count = HEADER.unpack_from(data)
    if magic != MAGIC:
        raise ValueError("bad magic")
    if version != 1:
        raise ValueError(f"unsupported version {version}")
    if len(data) != HEADER.size + count * RECORD.size:
        raise ValueError("length does not match record count")
    return list(RECORD.iter_unpack(memoryview(data)[HEADER.size:]))

Write archive helpers with `zipfile` and `gzip`. `zip_dir(src_dir, zip_path) -> list[str]` stores every file under `src_dir` with `ZIP_DEFLATED`, using POSIX paths relative to `src_dir` as archive names, added in sorted order, and returns those names. `safe_extract(zip_path, dest) -> list[Path]` guards against "zip slip". Otherwise it extracts all members into `dest` and returns `Path(dest) / name` for each file member (not directories), sorted. `write_gzip_lines(path, lines) -> int` writes each string plus `"\n"` into a gzip-compressed UTF-8 text file and returns the count. `gzip_lines(path)` is a generator that lazily yields the lines of such a file without their trailing newline.
@@@ reference
import gzip
import zipfile
from pathlib import Path, PurePosixPath


def zip_dir(src_dir, zip_path):
    src_dir = Path(src_dir)
    names = sorted(p.relative_to(src_dir).as_posix() for p in src_dir.rglob("*") if p.is_file())
    with zipfile.ZipFile(zip_path, "w", compression=zipfile.ZIP_DEFLATED) as zf:
        for name in names:
            zf.write(src_dir / name, arcname=name)
    return names


def _unsafe(name):
    p = PurePosixPath(name)
    return p.is_absolute() or ".." in p.parts or "\\" in name or (len(name) >= 2 and name[1] == ":")


def safe_extract(zip_path, dest):
    with zipfile.ZipFile(zip_path) as zf:
        members = zf.infolist()
        for m in members:
            if _unsafe(m.filename):
                raise ValueError(f"unsafe member name {m.filename!r}")
        zf.extractall(dest, members)
    return sorted(Path(dest) / m.filename for m in members if not m.is_dir())


def write_gzip_lines(path, lines):
    n = 0
    with gzip.open(path, "wt", encoding="utf-8") as f:
        for line in lines:
            f.write(line + "\n")
            n += 1
    return n


def gzip_lines(path):
    with gzip.open(path, "rt", encoding="utf-8") as f:
        for line in f:
            yield line.rstrip("\n")
