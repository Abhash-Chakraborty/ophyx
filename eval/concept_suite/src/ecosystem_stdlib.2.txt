Write DST-aware time arithmetic with `datetime` and `zoneinfo`. Local times are naive ISO strings such as `"2026-03-29T02:30"`, interpreted in an IANA `zone`. `local_time_exists(local, zone) -> bool` is False for wall-clock times skipped by a spring-forward transition (attach the zone, convert to UTC and back, and see whether the wall time survives). `is_ambiguous(local, zone) -> bool` is True for times that occur twice at a fall-back transition: the time exists and its UTC offset differs between `fold=0` and `fold=1`. `elapsed_hours(start_local, end_local, zone) -> float` returns the real elapsed time in hours between two local times, so midnight to midnight across the spring-forward day in Berlin is 23.0. `add_elapsed(local, zone, hours) -> str` adds real elapsed hours (do the arithmetic in UTC) and returns the resulting local time's `isoformat()` with its offset. Both of the last two functions raise `ValueError` for a nonexistent input time and use `fold=0` for ambiguous ones.
@@@ reference
from datetime import datetime, timedelta, timezone
from zoneinfo import ZoneInfo


def _attach(local, zone, fold=0):
    naive = datetime.fromisoformat(local)
    if naive.tzinfo is not None:
        raise ValueError("expected a naive local time")
    return naive, naive.replace(tzinfo=ZoneInfo(zone), fold=fold)


def local_time_exists(local, zone):
    naive, aware = _attach(local, zone)
    back = aware.astimezone(timezone.utc).astimezone(aware.tzinfo)
    return back.replace(tzinfo=None) == naive


def is_ambiguous(local, zone):
    _, aware = _attach(local, zone)
    return local_time_exists(local, zone) and aware.utcoffset() != aware.replace(fold=1).utcoffset()


def _checked(local, zone):
    if not local_time_exists(local, zone):
        raise ValueError(f"{local} does not exist in {zone}")
    return _attach(local, zone)[1]


def elapsed_hours(start_local, end_local, zone):
    start, end = _checked(start_local, zone), _checked(end_local, zone)
    return (end.astimezone(timezone.utc) - start.astimezone(timezone.utc)) / timedelta(hours=1)


def add_elapsed(local, zone, hours):
    start = _checked(local, zone)
    return (start.astimezone(timezone.utc) + timedelta(hours=hours)).astimezone(start.tzinfo).isoformat()

Write account transfers on `sqlite3` with explicit transactions. `open_bank(path)` connects with `isolation_level=None` (autocommit mode, so you issue `BEGIN`, `COMMIT`, `ROLLBACK` and `SAVEPOINT` yourself) and creates `accounts(id TEXT PRIMARY KEY, balance INTEGER NOT NULL CHECK (balance >= 0))`. `create_account(conn, account_id, balance=0)` inserts an account. `transfer(conn, src, dst, amount)` moves money atomically inside one `BEGIN ... COMMIT`. A non-positive amount raises `ValueError` before touching the database. An unknown account raises `KeyError`, and a balance that would go negative raises `InsufficientFunds` (a `ValueError` subclass) chained from the `IntegrityError` raised by the CHECK constraint. In both cases everything done so far is rolled back, including a debit already applied. `apply_batch(conn, transfers) -> list[bool]` runs a list of `(src, dst, amount)` in one outer transaction, each inside its own `SAVEPOINT`. A failing transfer is rolled back to its savepoint and reported as False, the others as True, and the batch commits once at the end. After any of these calls, `conn.in_transaction` must be False.
@@@ reference
import sqlite3


class InsufficientFunds(ValueError):
    pass


def open_bank(path):
    conn = sqlite3.connect(path, isolation_level=None)
    conn.execute("CREATE TABLE IF NOT EXISTS accounts ("
                 "id TEXT PRIMARY KEY, balance INTEGER NOT NULL CHECK (balance >= 0))")
    return conn


def create_account(conn, account_id, balance=0):
    conn.execute("INSERT INTO accounts (id, balance) VALUES (?, ?)", (account_id, balance))


def _move(conn, src, dst, amount):
    try:
        cur = conn.execute("UPDATE accounts SET balance = balance - ? WHERE id = ?", (amount, src))
    except sqlite3.IntegrityError as exc:
        raise InsufficientFunds(f"{src} cannot pay {amount}") from exc
    if cur.rowcount != 1:
        raise KeyError(src)
    if conn.execute("UPDATE accounts SET balance = balance + ? WHERE id = ?", (amount, dst)).rowcount != 1:
        raise KeyError(dst)


def transfer(conn, src, dst, amount):
    if amount <= 0:
        raise ValueError("amount must be positive")
    conn.execute("BEGIN")
    try:
        _move(conn, src, dst, amount)
    except BaseException:
        conn.execute("ROLLBACK")
        raise
    conn.execute("COMMIT")


def apply_batch(conn, transfers):
    results = []
    conn.execute("BEGIN")
    try:
        for src, dst, amount in transfers:
            conn.execute("SAVEPOINT one")
            try:
                if amount <= 0:
                    raise ValueError("amount must be positive")
                _move(conn, src, dst, amount)
            except (ValueError, KeyError):
                conn.execute("ROLLBACK TO one")
                results.append(False)
            else:
                results.append(True)
            conn.execute("RELEASE one")
    except BaseException:
        conn.execute("ROLLBACK")
        raise
    conn.execute("COMMIT")
    return results

Write CSV helpers built on dialects, `DictWriter` and `Sniffer`. `register_pipe_dialect()` registers a dialect named `"pipes"` with `delimiter="|"`, `quotechar='"'`, `quoting=csv.QUOTE_MINIMAL` and `lineterminator="\n"`, and calling it twice is harmless. `write_records(path, records, *, dialect="excel") -> list[str]` writes a list of dicts with `csv.DictWriter`. The header is the union of all keys in first-seen order, missing values are written as `""` (`restval`), and the file is opened with `newline=""`. It returns the header. `detect_delimiter(path) -> str` sniffs the first 4096 characters with `csv.Sniffer().sniff(sample, delimiters=",;\t|")` and returns the delimiter, raising `ValueError` (chained from `csv.Error`) when sniffing fails. `convert(src, dst, dialect) -> int` reads `src` with its detected delimiter using `csv.DictReader`, writes the rows to `dst` with `write_records` in the given dialect, and returns the number of data rows.
@@@ reference
import csv


def register_pipe_dialect():
    csv.register_dialect("pipes", delimiter="|", quotechar='"', quoting=csv.QUOTE_MINIMAL, lineterminator="\n")


def write_records(path, records, *, dialect="excel"):
    fields = {}
    for rec in records:
        for key in rec:
            fields.setdefault(key, None)
    header = list(fields)
    with open(path, "w", newline="", encoding="utf-8") as f:
        writer = csv.DictWriter(f, fieldnames=header, restval="", dialect=dialect)
        writer.writeheader()
        writer.writerows(records)
    return header


def detect_delimiter(path):
    with open(path, newline="", encoding="utf-8") as f:
        sample = f.read(4096)
    try:
        return csv.Sniffer().sniff(sample, delimiters=",;\t|").delimiter
    except csv.Error as exc:
        raise ValueError(f"cannot detect delimiter of {path}") from exc


def convert(src, dst, dialect):
    delim = detect_delimiter(src)
    with open(src, newline="", encoding="utf-8") as f:
        rows = list(csv.DictReader(f, delimiter=delim))
    write_records(dst, rows, dialect=dialect)
    return len(rows)
