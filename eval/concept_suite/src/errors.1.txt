import functools
import random
import time


def retry(attempts=3, base_delay=0.1, max_delay=2.0, exceptions=(Exception,), jitter=True,
          sleep=time.sleep, rng=random.random):
    if attempts < 1:
        raise ValueError("attempts must be >= 1")

    def deco(fn):
        @functools.wraps(fn)
        def wrapper(*args, **kwargs):
            for i in range(1, attempts + 1):
                try:
                    return fn(*args, **kwargs)
                except exceptions:
                    if i == attempts:
                        raise
                    delay = min(max_delay, base_delay * 2 ** (i - 1))
                    if jitter:
                        delay *= rng()
                    sleep(delay)
        return wrapper
    return deco

import random
import time


class RetryError(Exception):
    def __init__(self, attempts, last):
        super().__init__(f"gave up after {attempts} attempts: {last!r}")
        self.attempts = attempts
        self.last = last


def backoff_delays(base, factor, cap, attempts, jitter="none", rng=random.random):
    if jitter not in ("none", "full", "equal"):
        raise ValueError(f"unknown jitter {jitter!r}")
    for i in range(attempts - 1):
        raw = min(cap, base * factor ** i)
        if jitter == "full":
            yield raw * rng()
        elif jitter == "equal":
            yield raw / 2 + rng() * raw / 2
        else:
            yield raw


def call_with_retry(fn, delays, retry_on=(Exception,), sleep=time.sleep, on_retry=None):
    delays = iter(delays)
    attempt = 0
    while True:
        attempt += 1
        try:
            return fn()
        except retry_on as exc:
            delay = next(delays, None)
            if delay is None:
                raise RetryError(attempt, exc) from exc
            if on_retry is not None:
                on_retry(attempt, exc, delay)
            sleep(delay)

import functools
import time


class CircuitOpenError(Exception):
    pass


class CircuitBreaker:
    def __init__(self, failure_threshold=5, recovery_timeout=30.0, expected_exceptions=(Exception,),
                 clock=time.monotonic):
        self.failure_threshold = failure_threshold
        self.recovery_timeout = recovery_timeout
        self.expected_exceptions = expected_exceptions
        self.clock = clock
        self._failures = 0
        self._opened_at = None

    @property
    def state(self):
        if self._opened_at is None:
            return "closed"
        if self.clock() - self._opened_at >= self.recovery_timeout:
            return "half_open"
        return "open"

    def call(self, fn, *args, **kwargs):
        state = self.state
        if state == "open":
            raise CircuitOpenError("circuit is open")
        try:
            result = fn(*args, **kwargs)
        except self.expected_exceptions:
            self._failures += 1
            if state == "half_open" or self._failures >= self.failure_threshold:
                self._opened_at = self.clock()
            raise
        self._failures = 0
        self._opened_at = None
        return result

    def __call__(self, fn):
        @functools.wraps(fn)
        def wrapper(*args, **kwargs):
            return self.call(fn, *args, **kwargs)
        return wrapper

import threading


def run_with_timeout(fn, timeout: float, *args, **kwargs):
    box = {}
    done = threading.Event()

    def work():
        try:
            box["result"] = fn(*args, **kwargs)
        except BaseException as exc:
            box["error"] = exc
        finally:
            done.set()

    t = threading.Thread(target=work, daemon=True)
    t.start()
    if not done.wait(timeout):
        raise TimeoutError(f"{getattr(fn, '__name__', 'call')} did not finish in {timeout}s")
    if "error" in box:
        raise box["error"]
    return box["result"]

import queue
import time


def drain_until(q, deadline, max_items=None, clock=time.monotonic):
    end = clock() + deadline
    out = []
    while max_items is None or len(out) < max_items:
        remaining = end - clock()
        if remaining <= 0:
            break
        try:
            out.append(q.get(timeout=remaining))
        except queue.Empty:
            break
    return out

def parse_port(value):
    if isinstance(value, bool) or not isinstance(value, (int, str)):
        raise TypeError(f"port must be int or str, got {type(value).__name__}")
    if isinstance(value, str):
        s = value.strip()
        if not s.isdigit() or not s.isascii():
            raise ValueError(f"port is not a number: {value!r}")
        n = int(s)
    else:
        n = value
    if not 1 <= n <= 65535:
        raise ValueError(f"port out of range 1-65535: {value!r}")
    return n

_FIELDS = {"name", "email", "age"}


class ValidationError(Exception):

        self.errors = errors
        super().__init__("invalid fields: " + ", ".join(sorted(errors)))


def validate_user(payload):
    if not isinstance(payload, dict):
        raise TypeError(f"payload must be a dict, got {type(payload).__name__}")
    errors = {}
    name = payload.get("name")
    if name is None:
        errors["name"] = "required"
    elif not isinstance(name, str) or not name.strip():
        errors["name"] = "must be a non-empty string"
    elif len(name.strip()) > 50:
        errors["name"] = "must be at most 50 characters"
    email = payload.get("email")
    if email is None:
        errors["email"] = "required"
    elif not isinstance(email, str) or not _EMAIL.match(email):
        errors["email"] = "must be a valid email address"
    age = payload.get("age")
    if "age" in payload and age is not None:
        if isinstance(age, bool) or not isinstance(age, int):

    for key in payload:
        if key not in _FIELDS:
            errors[key] = "unknown field"
    if errors:
        raise ValidationError(errors)
    return {"name": name.strip(), "email": email.lower(), "age": age}

def run_all(tasks):
    results, errors = {}, []
    for name, fn in tasks.items():
        try:
            results[name] = fn()
        except Exception as exc:
            exc.add_note(f"task: {name}")
            errors.append(exc)
    if errors:
        raise ExceptionGroup(f"{len(errors)} of {len(tasks)} tasks failed", errors)
    return results

Write `process(fn) -> list[str]` that calls `fn()` and uses `except*` to handle `ValueError`s selectively. `fn` may return normally, raise a single exception, or raise an `ExceptionGroup` (possibly nested). Every `ValueError` anywhere in what was raised is handled: return the sorted list of their messages (`str(e)`). All other exceptions must keep propagating, which is exactly what `except*` does: the remainder of a group is re-raised as an exception group containing only the unhandled ones, and a single non-ValueError exception propagates unchanged. If `fn` raises nothing, return `[]`.
@@@ reference
def process(fn):
    messages = []
    try:
        fn()
    except* ValueError as group:
        stack = [group]
        while stack:
            g = stack.pop()
            for e in g.exceptions:
                if isinstance(e, BaseExceptionGroup):
                    stack.append(e)
                else:
                    messages.append(str(e))
    return sorted(messages)

import pytest
from solution import process


def raiser(exc):
    def f():
        raise exc
    return f


def test_no_error():
    assert process(lambda: 42) == []


def test_naked_value_error():
    assert process(raiser(ValueError("bad"))) == ["bad"]

def test_nested_groups():
    eg = ExceptionGroup("outer", [ValueError("x"), ExceptionGroup("inner", [ValueError("y"), ValueError("z")])])
    assert process(raiser(eg)) == ["x", "y", "z"]


def test_other_errors_propagate():
    eg = ExceptionGroup("g", [ValueError("handled"), KeyError("k"), TypeError("t")])
    with pytest.raises(ExceptionGroup) as ei:
        process(raiser(eg))
    remaining = ei.value.exceptions
    assert sorted(type(e).__name__ for e in remaining) == ["KeyError", "TypeError"]
