import sys

_MARK = "_configured_by_configure_logging"


def configure_logging(name, level="INFO", stream=None, fmt="%(levelname)s %(name)s: %(message)s"):
    if isinstance(level, str):
        resolved = logging.getLevelName(level.upper())
        if not isinstance(resolved, int):
            raise ValueError(f"unknown log level {level!r}")
        level = resolved
    logger = logging.getLogger(name)
    for h in list(logger.handlers):
        if getattr(h, _MARK, False):

    logger.addHandler(handler)
    logger.setLevel(level)
    logger.propagate = False
    return logger

import functools
import warnings


def deprecated(reason, *, version=None):
    def deco(fn):
        msg = f"{fn.__name__} is deprecated: {reason}"
        if version is not None:
            msg += f" (since {version})"

        @functools.wraps(fn)
        def wrapper(*args, **kwargs):
            warnings.warn(msg, DeprecationWarning, stacklevel=2)
            return fn(*args, **kwargs)
        wrapper.__deprecated__ = reason
        return wrapper
    return deco

import signal


class GracefulShutdown:
    def __init__(self, signals=(signal.SIGINT, signal.SIGTERM)):
        self.signals = tuple(signals)
        self.requested = False
        self.received = None
        self._previous = {}

    def _handler(self, signum, frame):
        self.requested = True
        self.received = signum

    def __enter__(self):
        for s in self.signals:
            self._previous[s] = signal.getsignal(s)
            signal.signal(s, self._handler)
        return self

    def __exit__(self, *exc):
        for s, h in self._previous.items():
            signal.signal(s, h)
        self._previous = {}
        return False

import os
import tempfile
from pathlib import Path


def atomic_write(path, data):
    path = Path(path)
    if isinstance(data, str):
        chunks = [data.encode("utf-8")]
    elif isinstance(data, (bytes, bytearray, memoryview)):
        chunks = [bytes(data)]
    else:
        chunks = data
    fd, tmp = tempfile.mkstemp(dir=path.parent, prefix=f".{path.name}.", suffix=".tmp")
    try:
        with os.fdopen(fd, "wb") as f:
            for chunk in chunks:
                f.write(chunk)
            f.flush()
            os.fsync(f.fileno())
        os.replace(tmp, path)
    except BaseException:
        try:
            os.unlink(tmp)
        except FileNotFoundError:
            pass
        raise
    return path

def _scan(path):
    if not os.path.exists(path):
        return [], False, 0
    with open(path, "rb") as f:
        lines = f.readlines()
    records, good = [], 0
    for i, raw in enumerate(lines):

        good += len(raw)
    return records, False, good

Write `load_jsonl(text: str) -> tuple[list, list]` that parses JSON Lines input from an untrusted source without crashing on bad lines. Return `(records, errors)`: `records` holds every line that parsed to a JSON object (dict); `errors` holds `(line_number, message)` pairs (1-based) for lines that are not valid JSON (the message should include the decoder's `msg`, e.g. "Expecting value") or that parse to something other than an object (message `"expected an object, got <type name>"`, e.g. `list`). Blank or whitespace-only lines are skipped silently. A UTF-8 byte-order mark at the start of the text is ignored.
@@@ reference
import json


def load_jsonl(text):
    if text.startswith("﻿"):
        text = text[1:]
    records, errors = [], []
    for n, line in enumerate(text.splitlines(), 1):
        if not line.strip():
            continue
        try:
            obj = json.loads(line)
        except json.JSONDecodeError as exc:
            errors.append((n, f"invalid JSON: {exc.msg}"))
            continue
        if not isinstance(obj, dict):
            errors.append((n, f"expected an object, got {type(obj).__name__}"))
            continue
        records.append(obj)
    return records, errors

import csv
import io


class CSVFormatError(ValueError):
    def __init__(self, line, column, reason):
        super().__init__(f"line {line}, column {column}: {reason}")
        self.line, self.column, self.reason = line, column, reason


def parse_csv(text, columns):
    reader = csv.reader(io.StringIO(text, newline=""))
    try:
        header = next(reader)
    except StopIteration:
        raise CSVFormatError(1, None, "empty input") from None
    for name in columns:
        if name not in header:
            raise CSVFormatError(1, name, "missing column")
    out = []
    for row in reader:
        if not row:
            continue
        if len(row) != len(header):
            raise CSVFormatError(reader.line_num, None, f"expected {len(header)} fields, got {len(row)}")
        rec = {}
        for name, raw in zip(header, row):
            conv = columns.get(name, str)
            try:
                rec[name] = conv(raw)
            except ValueError as exc:
                raise CSVFormatError(reader.line_num, name, f"bad value {raw!r}") from exc
        out.append(rec)
    return out

@@@ reference
from dataclasses import dataclass
from typing import Any


@dataclass(frozen=True)
class Ok:
    value: Any

    def is_ok(self):
        return True

    def map(self, fn):
        return Ok(fn(self.value))

    def unwrap(self):
        return self.value

    def unwrap_or(self, default):
        return self.value


@dataclass(frozen=True)
class Err:
    error: Any

    def is_ok(self):
        return False

    def map(self, fn):
        return self

    def unwrap(self):
        if isinstance(self.error, BaseException):
            raise self.error
        raise ValueError(self.error)

    def unwrap_or(self, default):
        return default


def try_call(fn, *args, catch=(Exception,), **kwargs):
    try:
        return Ok(fn(*args, **kwargs))
    except catch as exc:
        return Err(exc)


def parse_age(text):
    s = text.strip() if isinstance(text, str) else ""
    if not s.isdigit() or not s.isascii():
        return Err(f"not a whole number: {text!r}")
    n = int(s)
    if n > 150:
        return Err(f"age out of range: {n}")
    return Ok(n)

class NotFound(Exception):
    pass


class Forbidden(Exception):
    pass


class InvalidInput(Exception):
    def __init__(self, details=None):
        super().__init__("invalid input")
        self.details = details or {}


def handle_request(handler, payload, logger):
    try:
        result = handler(payload)
    except NotFound:
        return 404, {"error": "not found"}
    except Forbidden:
        return 403, {"error": "forbidden"}
    except InvalidInput as exc:
        return 422, {"error": "invalid input", "details": exc.details}
    except Exception:
        logger.exception("unhandled error in handler")
        return 500, {"error": "internal error"}
    return 200, {"data": result}
