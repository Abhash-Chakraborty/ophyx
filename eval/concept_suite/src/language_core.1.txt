import functools
import re

_RE = re.compile(r"(0|[1-9]\d*)\.(0|[1-9]\d*)\.(0|[1-9]\d*)(?:-([0-9A-Za-z-]+(?:\.[0-9A-Za-z-]+)*))?")


@functools.total_ordering
class SemVer:
    def __init__(self, major, minor, patch, prerelease=()):
        self.major, self.minor, self.patch = major, minor, patch
        self.prerelease = tuple(prerelease)

    @classmethod
    def parse(cls, text):
        m = _RE.fullmatch(text)
        if not m:
            raise ValueError(f"invalid version {text!r}")
        pre = tuple(m.group(4).split(".")) if m.group(4) else ()
        return cls(int(m.group(1)), int(m.group(2)), int(m.group(3)), pre)

    def __str__(self):
        base = f"{self.major}.{self.minor}.{self.patch}"
        return base + ("-" + ".".join(self.prerelease) if self.prerelease else "")

    def __repr__(self):
        return f"SemVer({str(self)!r})"

    def _key(self):
        return (self.major, self.minor, self.patch, self.prerelease)

    def __eq__(self, other):
        if not isinstance(other, SemVer):
            return NotImplemented
        return self._key() == other._key()

    def __hash__(self):
        return hash(self._key())

    @staticmethod
    def _pre_key(pre):
        return [(0, int(p), "") if p.isdigit() else (1, 0, p) for p in pre]

    def __lt__(self, other):
        if not isinstance(other, SemVer):
            return NotImplemented
        a, b = self._key()[:3], other._key()[:3]
        if a != b:
            return a < b
        if not self.prerelease or not other.prerelease:
            return bool(self.prerelease) and not other.prerelease
        return self._pre_key(self.prerelease) < self._pre_key(other.prerelease)

class GridIterator:
    def __init__(self, rows, cols):
        self.rows, self.cols = rows, cols
        self._i = 0

    def __iter__(self):
        return self

    def __next__(self):
        if self._i >= self.rows * self.cols:
            raise StopIteration
        r, c = divmod(self._i, self.cols)
        self._i += 1
        return (r, c)


class Grid:
    def __init__(self, rows, cols):
        if rows < 0 or cols < 0:
            raise ValueError("sizes must be non-negative")
        self.rows, self.cols = rows, cols

    def __iter__(self):
        return GridIterator(self.rows, self.cols)

    def __len__(self):
        return self.rows * self.cols

import re

_LINE = re.compile(r"([A-Z]+):\s*(.*)")


def parse(lines):
    for line in lines:
        m = _LINE.fullmatch(line.rstrip("\n"))
        if m:
            yield m.group(1), m.group(2).strip()


def only(records, level):
    return (r for r in records if r[0] == level)


def take(n, iterable):
    if n <= 0:
        return
    for i, item in enumerate(iterable, 1):
        yield item
        if i >= n:
            return


def errors_pipeline(lines, n):
    return (msg for _, msg in take(n, only(parse(lines), "ERROR")))

def stream_stats(numbers):
    count = 0
    total = 0
    lo = hi = None
    for x in numbers:
        count += 1
        total += x
        lo = x if lo is None or x < lo else lo
        hi = x if hi is None or x > hi else hi
    return {"count": count, "total": total, "minimum": lo, "maximum": hi,
            "mean": total / count if count else None}

    return sum(i * i for i in range(n))

def validate_range(**bounds):
    def deco(fn):
        sig = inspect.signature(fn)
        unknown = set(bounds) - set(sig.parameters)
        if unknown:
            raise TypeError(f"{fn.__name__} has no parameters {sorted(unknown)}")

        fields = {}
        for base in reversed(bases):

        own = {k: v for k, v in ns.items() if isinstance(v, Field)}

from contextlib import contextmanager

_MISSING = object()


@contextmanager
def patched(mapping, **overrides):
    saved = {k: mapping.get(k, _MISSING) if k in mapping else _MISSING for k in overrides}
    try:
        for k, v in overrides.items():
            if v is None:
                mapping.pop(k, None)
            else:
                mapping[k] = v
        yield mapping
    finally:
        for k, old in saved.items():
            if old is _MISSING:
                mapping.pop(k, None)
            else:
                mapping[k] = old

from urllib.parse import quote, urlencode


def make_url(host, /, *path, scheme="https", port=None, **query):
    if port is not None and (not isinstance(port, int) or isinstance(port, bool) or not 1 <= port <= 65535):
        raise ValueError(f"invalid port {port!r}")
    netloc = host if port is None else f"{host}:{port}"
    url = f"{scheme}://{netloc}/" + "/".join(quote(str(seg), safe="") for seg in path)
    params = [(k, v) for k, v in query.items() if v is not None]
    if params:
        url += "?" + urlencode(params)
    return url

import re

_SETTING = re.compile(r"\s*([A-Za-z_]\w*)\s*=\s*(.*?)\s*$")


def split_ends(seq):
    if len(seq) < 2:
        raise ValueError("need at least two items")
    first, *middle, last = seq
    return first, middle, last


def read_chunks(read, size):
    chunks = []
    while (chunk := read(size)):
        chunks.append(chunk)
    return chunks


def parse_settings(lines):
    out = {}
    for line in lines:
        if (m := _SETTING.match(line)):
            out[m.group(1)] = m.group(2)
    return out
