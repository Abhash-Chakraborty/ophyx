from dataclasses import dataclass


@dataclass(frozen=True)
class Num:
    value: object


@dataclass(frozen=True)
class Var:
    name: str


@dataclass(frozen=True)
class Add:
    left: object
    right: object


@dataclass(frozen=True)
class Mul:
    left: object
    right: object


def simplify(expr):
    match expr:
        case Num() | Var():
            return expr
        case Add(left, right):
            match simplify(left), simplify(right):
                case Num(a), Num(b):
                    return Num(a + b)
                case (Num(0), x) | (x, Num(0)):
                    return x
                case x, y if x == y:
                    return Mul(Num(2), x)
                case x, y:
                    return Add(x, y)
        case Mul(left, right):
            match simplify(left), simplify(right):
                case Num(a), Num(b):
                    return Num(a * b)
                case (Num(0), _) | (_, Num(0)):
                    return Num(0)
                case (Num(1), x) | (x, Num(1)):
                    return x
                case x, y:
                    return Mul(x, y)
    raise TypeError(f"not an expression: {expr!r}")


def evaluate(expr, env):
    match expr:
        case Num(value):
            return value
        case Var(name):
            return env[name]
        case Add(left, right):
            return evaluate(left, env) + evaluate(right, env)
        case Mul(left, right):
            return evaluate(left, env) * evaluate(right, env)
    raise TypeError(f"not an expression: {expr!r}")

import enum

_BITS = (("r", 4), ("w", 2), ("x", 1))


class Perm(enum.Flag):
    X = 1
    W = 2
    R = 4
    RWX = 7


def parse_mode(text):
    if not isinstance(text, str) or len(text) != 3:
        raise ValueError(f"bad mode {text!r}")
    perm = Perm(0)
    for ch, (letter, bit) in zip(text, _BITS):
        if ch == letter:
            perm |= Perm(bit)
        elif ch != "-":
            raise ValueError(f"bad mode {text!r}")
    return perm


def format_mode(perm):
    return "".join(letter if Perm(bit) in perm else "-" for letter, bit in _BITS)


class Level(enum.IntEnum):
    LOW = 10
    MEDIUM = 20
    HIGH = 30

    @classmethod
    def parse(cls, text):
        if text.isdigit():
            return cls(int(text))
        try:
            return cls[text.upper()]
        except KeyError:
            raise ValueError(f"unknown level {text!r}") from None


class Status(enum.Enum):
    ACTIVE = "active"
    SUSPENDED = "suspended"
    CLOSED = "closed"

    @classmethod
    def _missing_(cls, value):
        if isinstance(value, str):
            for member in cls:
                if member.value == value.lower():
                    return member
        return None

import functools
from decimal import Decimal, ROUND_HALF_UP

_CENT = Decimal("0.01")


@functools.total_ordering
class Money:
    __slots__ = ("_amount", "_currency")

    def __init__(self, amount, currency="EUR"):
        if isinstance(amount, bool) or not isinstance(amount, (int, str, Decimal)):
            raise TypeError("amount must be int, str or Decimal")
        object.__setattr__(self, "_amount", Decimal(amount).quantize(_CENT, ROUND_HALF_UP))
        object.__setattr__(self, "_currency", currency)

    def __setattr__(self, name, value):
        raise AttributeError("Money is immutable")

    @property
    def amount(self):
        return self._amount

    @property
    def currency(self):
        return self._currency

    def _same(self, other):
        if other._currency != self._currency:
            raise ValueError(f"currency mismatch: {self._currency} vs {other._currency}")

    def __add__(self, other):
        if not isinstance(other, Money):
            return NotImplemented
        self._same(other)
        return Money(self._amount + other._amount, self._currency)

    def __radd__(self, other):
        if isinstance(other, int) and not isinstance(other, bool) and other == 0:
            return self
        return NotImplemented

    def __sub__(self, other):
        if not isinstance(other, Money):
            return NotImplemented
        self._same(other)
        return Money(self._amount - other._amount, self._currency)

    def __mul__(self, factor):
        if isinstance(factor, bool) or not isinstance(factor, (int, Decimal)):
            return NotImplemented
        return Money(self._amount * factor, self._currency)

    __rmul__ = __mul__

    def __neg__(self):
        return Money(-self._amount, self._currency)

    def __abs__(self):
        return Money(abs(self._amount), self._currency)

    def __bool__(self):
        return bool(self._amount)

    def __eq__(self, other):
        if not isinstance(other, Money):
            return NotImplemented
        return (self._amount, self._currency) == (other._amount, other._currency)

    def __hash__(self):
        return hash((self._amount, self._currency))

    def __lt__(self, other):
        if not isinstance(other, Money):
            return NotImplemented
        self._same(other)
        return self._amount < other._amount

    def __str__(self):
        return f"{self._amount} {self._currency}"

    def __repr__(self):
        return f"Money('{self._amount}', '{self._currency}')"

TOTAL = 0
REGISTRY = {}


def register(name):
    global TOTAL
    if name in REGISTRY:
        raise ValueError(f"{name!r} already registered")
    TOTAL += 1
    REGISTRY[name] = TOTAL
    return TOTAL


def reset_registry():
    global TOTAL
    TOTAL = 0
    REGISTRY.clear()


def make_tally(start=0):
    total = start

    def tally(n=1):
        nonlocal total
        total += n
        return total

    def reset():
        nonlocal total
        total = start

    return tally, reset


def make_multipliers(n):
    return [lambda x, i=i: x * i for i in range(n)]

import functools


def compose(*fns):
    if not fns:
        return lambda x: x
    *outer, inner = fns

    def composed(*args, **kwargs):
        value = inner(*args, **kwargs)
        for f in reversed(outer):
            value = f(value)
        return value
    return composed


def flip(fn):
    @functools.wraps(fn)
    def flipped(a, b, *args, **kwargs):
        return fn(b, a, *args, **kwargs)
    return flipped


def rounder(ndigits):
    return functools.partial(round, ndigits=ndigits)


def apply_n(fn, n):
    if n < 0:
        raise ValueError("n must be non-negative")

    def repeated(x):
        for _ in range(n):
            x = fn(x)
        return x
    return repeated

def running_balance(start, deltas):

import itertools


def rle(text):
    return [(ch, sum(1 for _ in group)) for ch, group in itertools.groupby(text)]


def paginate(iterable, size):
    if size < 1:
        raise ValueError("size must be at least 1")
    return _pages(iter(iterable), size)


def _pages(it, size):
    while page := list(itertools.islice(it, size)):
        yield page


def budget_combos(prices, k, budget):
    return [combo for combo in itertools.combinations(prices, k)
            if sum(prices[name] for name in combo) <= budget]


def seat_labels(rows, count):
    return [f"{row}{n}" for row, n in itertools.product(rows, range(1, count + 1))]

import re
from collections import ChainMap, Counter, defaultdict, deque, namedtuple

Reading = namedtuple("Reading", "sensor value unit", defaults=("C",))


def top_words(text, n):
    return Counter(re.findall(r"[a-z']+", text.lower())).most_common(n)


def by_sensor(readings):
    groups = defaultdict(list)
    for r in readings:
        groups[r.sensor].append(r.value)
    return groups


def moving_average(values, window):
    if window < 1:
        raise ValueError("window must be at least 1")
    buf = deque(maxlen=window)
    out = []
    for v in values:
        buf.append(v)
        out.append(sum(buf) / len(buf))
    return out


def layered(*maps):
    return ChainMap({}, *maps)

def money(x):
    digits = f"{abs(x):,.2f}"
    sign = "-" if x < 0 and digits != "0.00" else ""
    return f"{sign}${digits}"


def pct(x, places=1):
    return f"{x:.{places}%}"


def _cell(v):
    return f"{v:.2f}" if isinstance(v, float) else str(v)


def _numeric(v):
    return isinstance(v, (int, float)) and not isinstance(v, bool)


def table(headers, rows):
    cells = [[_cell(v) for v in row] for row in rows]
    widths = [max([len(h)] + [len(r[i]) for r in cells]) for i, h in enumerate(headers)]
    lines = [" | ".join(f"{h:<{w}}" for h, w in zip(headers, widths)),
             "-+-".join("-" * w for w in widths)]
    for raw, row in zip(rows, cells):
        lines.append(" | ".join(f"{c:>{w}}" if _numeric(v) else f"{c:<{w}}"
                                for v, c, w in zip(raw, row, widths)))
    return "\n".join(line.rstrip() for line in lines)


class Duration:
    def __init__(self, seconds):
        if seconds < 0:
            raise ValueError("duration must be non-negative")
        self.seconds = int(seconds)

    def __format__(self, spec):
        h, rem = divmod(self.seconds, 3600)
        m, s = divmod(rem, 60)
        if spec in ("", "hms"):
            return f"{h}:{m:02d}:{s:02d}"
        if spec == "s":
            return str(self.seconds)
        if spec == "short":
            return "".join(f"{v}{u}" for v, u in ((h, "h"), (m, "m"), (s, "s")) if v) or "0s"
        raise ValueError(f"unknown format spec {spec!r}")

    def __str__(self):
        return format(self, "")

import codecs


def encode_frames(messages):
    out = bytearray()
    for m in messages:
        b = m.encode("utf-8")
        if len(b) > 0xFFFF:
            raise ValueError("message too long")
        out += len(b).to_bytes(2, "big")
        out += b
    return bytes(out)


def decode_frames(data):
    view = memoryview(data).cast("B")
    msgs, i = [], 0
    while i < len(view):
        if i + 2 > len(view):
            raise ValueError("truncated header")
        n = int.from_bytes(view[i:i + 2], "big")
        if i + 2 + n > len(view):
            raise ValueError("truncated frame")
        msgs.append(str(view[i + 2:i + 2 + n], "utf-8"))
        i += 2 + n
    return msgs


def xor_inplace(buf, key):
    if not isinstance(buf, bytearray):
        raise TypeError("buf must be a bytearray")
    if not key:
        raise ValueError("key must not be empty")
    for i in range(len(buf)):
        buf[i] ^= key[i % len(key)]


def sniff_decode(data):
    data = bytes(data)
    if data.startswith(codecs.BOM_UTF8):
        return data[len(codecs.BOM_UTF8):].decode("utf-8")
    if data.startswith((codecs.BOM_UTF16_LE, codecs.BOM_UTF16_BE)):
        return data.decode("utf-16")
    try:
        return data.decode("utf-8")
    except UnicodeDecodeError:
        return data.decode("latin-1")

import re
from fractions import Fraction

_MIXED = re.compile(r"(\d+) (\d+)/(\d+)")
_FRAC = re.compile(r"(\d+)/(\d+)")
_NUM = re.compile(r"\d+(?:\.\d+)?")


def parse_quantity(text):
    text = text.strip()
    if m := _MIXED.fullmatch(text):
        whole, num, den = map(int, m.groups())
        if den == 0 or num >= den:
            raise ValueError(f"bad mixed number {text!r}")
        return whole + Fraction(num, den)
    if m := _FRAC.fullmatch(text):
        num, den = map(int, m.groups())
        if den == 0:
            raise ValueError("zero denominator")
        return Fraction(num, den)
    if _NUM.fullmatch(text):
        return Fraction(text)
    raise ValueError(f"bad quantity {text!r}")


def format_quantity(q):
    q = Fraction(q)
    if q < 0:
        raise ValueError("negative quantity")
    whole, rest = divmod(q, 1)
    if rest == 0:
        return str(whole)
    frac = f"{rest.numerator}/{rest.denominator}"
    return f"{whole} {frac}" if whole else frac


def scale_recipe(ingredients, factor):
    f = parse_quantity(factor) if isinstance(factor, str) else Fraction(factor)
    return {name: format_quantity(parse_quantity(q) * f) for name, q in ingredients.items()}


def continued_fraction(x):
    x = Fraction(x)
    if x < 0:
        raise ValueError("negative value")
    terms = []
    num, den = x.numerator, x.denominator
    while den:
        a, r = divmod(num, den)
        terms.append(a)
        num, den = den, r
    return terms


def from_continued_fraction(terms):
    value = Fraction(terms[-1])
    for a in reversed(terms[:-1]):
        value = a + 1 / value
    return value

import copy


class Document:
    def __init__(self, title, sections=None, owner=None):
        self.title = title
        self.sections = sections if sections is not None else []
        self.owner = owner
        self.cache = {}

    def __copy__(self):
        new = type(self).__new__(type(self))
        new.__dict__.update(self.__dict__)
        new.cache = {}
        return new

    def __deepcopy__(self, memo):
        new = type(self).__new__(type(self))
        memo[id(self)] = new
        for name, value in self.__dict__.items():
            if name == "owner":
                new.owner = value
            elif name == "cache":
                new.cache = {}
            else:
                setattr(new, name, copy.deepcopy(value, memo))
        return new


def make_grid(rows, cols, fill):
    return [[copy.deepcopy(fill) for _ in range(cols)] for _ in range(rows)]

import importlib
import importlib.util
import os
import sys
import types


def load_from_path(path, module_name):
    if module_name in sys.modules:
        return sys.modules[module_name]
    path = os.fspath(path)
    if not os.path.isfile(path):
        raise FileNotFoundError(path)
    spec = importlib.util.spec_from_file_location(module_name, path)
    module = importlib.util.module_from_spec(spec)
    sys.modules[module_name] = module
    try:
        spec.loader.exec_module(module)
    except BaseException:
        sys.modules.pop(module_name, None)
        raise
    return module


def public_api(module):
    if hasattr(module, "__all__"):
        return list(module.__all__)
    return sorted(n for n, v in vars(module).items()
                  if not n.startswith("_") and not isinstance(v, types.ModuleType))


def star_import(module):
    if hasattr(module, "__all__"):
        return {name: getattr(module, name) for name in module.__all__}
    return {n: v for n, v in vars(module).items() if not n.startswith("_")}


def import_from_root(root, dotted):
    root = os.fspath(root)
    sys.path.insert(0, root)
    try:
        importlib.invalidate_caches()
        return importlib.import_module(dotted)
    finally:
        sys.path.remove(root)
