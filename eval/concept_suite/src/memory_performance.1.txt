import gc
import weakref


class TreeNode:
    def __init__(self, name):
        self.name = name
        self.children = []
        self._parent = None

    @property
    def parent(self):
        return self._parent() if self._parent is not None else None

    def add_child(self, child):
        if child.parent is not None:
            raise ValueError(f"{child.name!r} already has a parent")
        child._parent = weakref.ref(self)
        self.children.append(child)
        return child

    def path(self):

        return "/".join(reversed(parts))

import weakref


class ObjectCache:
    def __init__(self, loader):
        self._loader = loader
        self._items = weakref.WeakValueDictionary()
        self.loads = 0

    def get(self, key):
        obj = self._items.get(key)
        if obj is None:
            obj = self._loader(key)
            self.loads += 1
            self._items[key] = obj
        return obj

    def peek(self, key):
        return self._items.get(key)

    def __len__(self):
        return len(self._items)

    def __contains__(self, key):
        return key in self._items

import keyword


def make_record_class(name, fields):
    fields = tuple(fields)
    if len(set(fields)) != len(fields) or not all(f.isidentifier() and not keyword.iskeyword(f) for f in fields):
        raise ValueError(f"bad field names {fields!r}")

    def __init__(self, *args, **kwargs):
        if len(args) > len(fields):
            raise TypeError(f"{name} takes at most {len(fields)} positional arguments")
        values = dict(zip(fields, args))
        for k, v in kwargs.items():
            if k not in fields:
                raise TypeError(f"unknown field {k!r}")
            if k in values:
                raise TypeError(f"field {k!r} given twice")
            values[k] = v
        missing = [f for f in fields if f not in values]
        if missing:
            raise TypeError(f"missing fields {missing}")
        for f in fields:
            setattr(self, f, values[f])

    def __repr__(self):
        return f"{name}(" + ", ".join(f"{f}={getattr(self, f)!r}" for f in fields) + ")"

    def __eq__(self, other):
        if type(other) is not type(self):
            return NotImplemented
        return tuple(self) == tuple(other)

    def __iter__(self):
        return (getattr(self, f) for f in fields)

    ns = {"__slots__": fields + ("__weakref__",), "__init__": __init__, "__repr__": __repr__,
          "__eq__": __eq__, "__iter__": __iter__, "__hash__": None}
    return type(name, (), ns)

import sys
import tracemalloc


def _slot_names(cls):
    for klass in cls.__mro__:
        slots = klass.__dict__.get("__slots__", ())
        if isinstance(slots, str):
            slots = (slots,)
        for s in slots:
            if s not in ("__dict__", "__weakref__"):
                yield s


def deep_sizeof(obj):
    seen, total, stack = set(), 0, [obj]
    while stack:
        o = stack.pop()
        if id(o) in seen:
            continue
        seen.add(id(o))
        total += sys.getsizeof(o)
        if isinstance(o, dict):
            stack.extend(o.keys())
            stack.extend(o.values())
        elif isinstance(o, (list, tuple, set, frozenset)):
            stack.extend(o)
        elif not isinstance(o, type):
            d = getattr(o, "__dict__", None)
            if isinstance(d, dict):
                stack.append(d)
            for name in _slot_names(type(o)):
                try:
                    stack.append(getattr(o, name))
                except AttributeError:
                    pass
    return total


def peak_memory(fn, *args, **kwargs):
    started = not tracemalloc.is_tracing()
    if started:
        tracemalloc.start()
    try:
        tracemalloc.reset_peak()
        base = tracemalloc.get_traced_memory()[0]
        result = fn(*args, **kwargs)
        peak = tracemalloc.get_traced_memory()[1] - base
    finally:
        if started:
            tracemalloc.stop()
    return result, peak

def stream_records(path, chunk_size=65536):
    if chunk_size < 1:
        raise ValueError("chunk_size must be at least 1")
    return _records(path, chunk_size)


def _records(path, chunk_size):
    buf = b""
    with open(path, "rb") as f:
        while chunk := f.read(chunk_size):
            buf += chunk
            *done, buf = buf.split(b"\x1e")
            for rec in done:
                yield rec.decode("utf-8")
    if buf:
        yield buf.decode("utf-8")

    return max((len(r) for r in stream_records(path)), default=0)

import heapq
import itertools
import os


def chunked(iterable, size):
    if size < 1:
        raise ValueError("size must be at least 1")
    return _chunks(iter(iterable), size)


def _chunks(it, size):
    while chunk := list(itertools.islice(it, size)):
        yield chunk


def _read_ints(path):
    with open(path) as f:
        for line in f:
            yield int(line)


def external_sort(numbers, chunk_size, tmp_dir):
    paths = []
    try:
        for i, chunk in enumerate(chunked(numbers, chunk_size)):
            chunk.sort()
            path = os.path.join(tmp_dir, f"chunk-{i}.txt")
            with open(path, "w") as f:
                f.writelines(f"{n}\n" for n in chunk)
            paths.append(path)
        readers = [_read_ints(p) for p in paths]
        try:
            yield from heapq.merge(*readers)
        finally:
            for r in readers:
                r.close()
    finally:
        for p in paths:
            os.remove(p)

import functools


class CachedLookup:
    def __init__(self, fetch, maxsize=128):
        self._cached = functools.lru_cache(maxsize=maxsize)(fetch)

    def get(self, key):
        return self._cached(key)

    def invalidate(self):
        self._cached.cache_clear()

    def info(self):
        return self._cached.cache_info()

Write a decorator `memoize(fn)` that caches results by hand in a dict, without `functools.lru_cache` or `functools.cache`. The cache key combines the positional arguments with the keyword arguments independent of their order, so `f(a=1, b=2)` and `f(b=2, a=1)` share one entry (positional and keyword spellings of the same argument may be separate entries). Falsy results, including None, are cached like any other; a call that raises is not cached, so the next call tries again. The wrapper exposes its dict as `wrapper.cache`, provides `wrapper.clear()` to empty it, and keeps the wrapped function's `__name__` and `__doc__` via `functools.wraps`. Unhashable arguments raise TypeError. A recursive function that calls itself through its decorated name must benefit from the cache.
@@@ reference
import functools


def memoize(fn):
    cache = {}

    @functools.wraps(fn)
    def wrapper(*args, **kwargs):
        key = (args, frozenset(kwargs.items()))
        try:
            return cache[key]
        except KeyError:
            pass
        result = fn(*args, **kwargs)
        cache[key] = result
        return result

    wrapper.cache = cache
    wrapper.clear = cache.clear
    return wrapper
