def _run(fn, args, kwargs):
    prof = cProfile.Profile()
    prof.enable()
    try:

    finally:
        prof.disable()

def profile_call(fn, *args, **kwargs):
    result, prof = _run(fn, args, kwargs)
    counts = {}
    for (_, _, name), (cc, nc, *_rest) in pstats.Stats(prof).stats.items():

import time
import timeit


def time_per_call(fn, number=1000, repeat=5, timer=time.perf_counter):
    if number < 1 or repeat < 1:
        raise ValueError("number and repeat must be at least 1")
    return min(timeit.Timer(fn, timer=timer).repeat(repeat=repeat, number=number)) / number


class Stopwatch:
    def __init__(self, clock=time.perf_counter):
        self._clock = clock
        self._start = None
        self.laps = []
        self.elapsed = 0.0

    def __enter__(self):
        if self._start is not None:
            raise RuntimeError("stopwatch already running")
        self._start = self._clock()
        return self

    def __exit__(self, *exc):
        lap = self._clock() - self._start
        self._start = None
        self.laps.append(lap)
        self.elapsed += lap
        return False

    def reset(self):
        self.laps = []
        self.elapsed = 0.0

import array
import sys


class IntSeries:
    def __init__(self, values=()):
        self.data = array.array("q", values)

    def append(self, x):
        self.data.append(x)

    def extend(self, iterable):
        self.data.extend(iterable)

    def __len__(self):
        return len(self.data)

    def __iter__(self):
        return iter(self.data)

    def __getitem__(self, index):
        if isinstance(index, slice):
            return IntSeries(self.data[index])
        return self.data[index]

        return sum(self.data)

    def nbytes(self):

    def from_bytes(cls, data):

        a.frombytes(data)

import mmap
import os


def _open_map(f):
    return mmap.mmap(f.fileno(), 0, access=mmap.ACCESS_READ)


def find_all(path, needle):
    if not needle:
        raise ValueError("empty needle")
    if os.path.getsize(path) == 0:
        return []
    with open(path, "rb") as f, _open_map(f) as m:
        out, i = [], m.find(needle)
        while i != -1:
            out.append(i)
            i = m.find(needle, i + 1)
        return out


def count_lines(path):
    size = os.path.getsize(path)
    if size == 0:
        return 0
    with open(path, "rb") as f, _open_map(f) as m:
        n, i = 0, m.find(b"\n")
        while i != -1:
            n += 1
            i = m.find(b"\n", i + 1)
        return n + (m[size - 1:size] != b"\n")


def patch(path, offset, data):
    size = os.path.getsize(path)
    if offset < 0 or offset + len(data) > size:
        raise ValueError("patch outside the file")
    if not data:
        return
    with open(path, "r+b") as f, mmap.mmap(f.fileno(), 0) as m:
        m[offset:offset + len(data)] = data
        m.flush()

import ast
import sys


def _is_literal(node):
    if isinstance(node, ast.Constant):
        return not (node.value is None or node.value is Ellipsis or isinstance(node.value, bool))
    if isinstance(node, ast.UnaryOp) and isinstance(node.op, (ast.USub, ast.UAdd)):
        v = node.operand
        return isinstance(v, ast.Constant) and isinstance(v.value, (int, float, complex)) \
            and not isinstance(v.value, bool)
    return isinstance(node, (ast.Tuple, ast.List, ast.Dict, ast.Set))


def identity_comparisons(source):
    found = []
    for node in ast.walk(ast.parse(source)):
        if isinstance(node, ast.Compare):
            operands = [node.left, *node.comparators]
            for i, op in enumerate(node.ops):
                if isinstance(op, (ast.Is, ast.IsNot)) and (_is_literal(operands[i]) or _is_literal(operands[i + 1])):
                    found.append((node.lineno, "is" if isinstance(op, ast.Is) else "is not"))
    return sorted(found)

    return [{(sys.intern(k) if isinstance(k, str) else k): v for k, v in r.items()} for r in records]

Nothing is computed in `__init__`, and each value is computed at most once until it is invalidated. To make this observable, every computation of one of the three appends its name to the instance's list `computed` (which starts empty). `text` is a property; assigning a new text must invalidate all three cached values so they recompute on next access. `invalidate()` drops the cached values explicitly and must not fail when some were never computed.
@@@ reference
import collections
import functools
import re


class Document:
    _CACHED = ("words", "word_count", "frequencies")

    def __init__(self, text):
        self._text = text
        self.computed = []

    @property
    def text(self):
        return self._text

    @text.setter
    def text(self, value):
        self._text = value
        self.invalidate()

    def invalidate(self):
        for name in self._CACHED:
            self.__dict__.pop(name, None)

    @functools.cached_property
    def words(self):
        self.computed.append("words")
        return re.findall(r"[a-z']+", self._text.lower())

    @functools.cached_property
    def word_count(self):
        self.computed.append("word_count")
        return len(self.words)

    @functools.cached_property
    def frequencies(self):
        self.computed.append("frequencies")
        return collections.Counter(self.words)

import collections
import functools
from solution import Document


def test_nothing_computed_eagerly():
    d = Document("a b c")
    assert d.computed == []


def test_values():
    d = Document("The cat and the hat. Don't!")
    assert d.words == ["the", "cat", "and", "the", "hat", "don't"]
    assert d.word_count == 6 and d.frequencies["the"] == 2


def test_computed_once():
    d = Document("x y x")
    for _ in range(3):
        d.word_count
        d.frequencies
        d.words
    assert collections.Counter(d.computed) == {"words": 1, "word_count": 1, "frequencies": 1}


def test_uses_cached_property():
    assert isinstance(Document.__dict__["words"], functools.cached_property)
    d = Document("q")
    d.words
    assert d.__dict__["words"] == ["q"]


def test_setting_text_invalidates():
    d = Document("one two")
    assert d.word_count == 2
    d.text = "one two three"
    assert d.word_count == 3 and d.frequencies["three"] == 1
    assert d.computed.count("words") == 2


def test_invalidate_partial():
    d = Document("a")
    d.invalidate()
    d.words
    d.invalidate()
    assert d.words == ["a"] and d.computed == ["words", "words"]
