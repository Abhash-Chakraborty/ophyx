Build a small registry of `functools.lru_cache` caches so an application can inspect and invalidate them all at once. `REGISTRY` is a module-level dict mapping a function's `__qualname__` to its cached wrapper. `registered_cache(maxsize=128, typed=False)` is a decorator factory that wraps the function with `functools.lru_cache(maxsize=maxsize, typed=typed)`, registers the wrapper and returns it. `clear_caches() -> int` clears every registered cache with `cache_clear()` and returns the total number of entries dropped (the sum of the `currsize` values before clearing). `cache_report() -> dict[str, tuple[int, int, int]]` maps each registered name to `(hits, misses, currsize)`. Finally, define `describe(value) -> str`, returning `f"{type(value).__name__}:{value!r}"`, decorated with `@registered_cache(maxsize=256, typed=True)`. It has to be typed: `1`, `1.0` and `True` are equal and hash the same, so an untyped cache could hand `describe(True)` the cached answer for `describe(1.0)`.
@@@ reference
import functools

REGISTRY = {}


def registered_cache(maxsize=128, typed=False):
    def decorator(fn):
        wrapped = functools.lru_cache(maxsize=maxsize, typed=typed)(fn)
        REGISTRY[fn.__qualname__] = wrapped
        return wrapped
    return decorator


def clear_caches() -> int:
    dropped = 0
    for wrapped in REGISTRY.values():
        dropped += wrapped.cache_info().currsize
        wrapped.cache_clear()
    return dropped


def cache_report() -> dict[str, tuple[int, int, int]]:
    out = {}
    for name, wrapped in REGISTRY.items():
        info = wrapped.cache_info()
        out[name] = (info.hits, info.misses, info.currsize)
    return out


@registered_cache(maxsize=256, typed=True)
def describe(value) -> str:
    return f"{type(value).__name__}:{value!r}"

Process a large text file in a single streaming pass, iterating over the file object line by line and never holding all lines in memory (no `read()`, `readlines()` or `list(f)`). `line_stats(path) -> tuple[int, int, int]` returns `(line_count, total_chars, longest)`, where characters are counted without the line terminator (`"\n"`, read as UTF-8 text); a final line without a newline still counts, and an empty file gives `(0, 0, 0)`. `longest_lines(path, n) -> list[str]` returns the `n` longest lines (terminators stripped), longest first, and among equal lengths the earlier line first; keep only `n` candidates at a time, for example with `heapq.nlargest(n, generator, key=len)`, which is stable in that way. `n <= 0` gives `[]`.
@@@ reference
import heapq


def _lines(path):
    with open(path, encoding="utf-8") as f:
        for line in f:
            yield line.rstrip("\n")


def line_stats(path):
    count = total = longest = 0
    for line in _lines(path):
        count += 1
        total += len(line)
        longest = max(longest, len(line))
    return count, total, longest


def longest_lines(path, n):
    if n <= 0:
        return []
    return heapq.nlargest(n, _lines(path), key=len)

import tracemalloc
import pytest
from solution import line_stats, longest_lines

def big(tmp_path):

def measure(fn, *args):
    tracemalloc.start()
    try:
        result = fn(*args)
        return result, tracemalloc.get_traced_memory()[1]
    finally:
        tracemalloc.stop()


def test_small_file(tmp_path):
