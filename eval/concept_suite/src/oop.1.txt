from types import MappingProxyType


class ShoppingCart:
    def __init__(self):
        self._lines = {}
        self._qty = {}

    def add(self, sku, price, qty=1):
        if not isinstance(qty, int) or isinstance(qty, bool) or qty <= 0:
            raise ValueError("qty must be a positive int")
        price = Decimal(str(price))
        if sku in self._lines and self._lines[sku] != price:
            raise ValueError(f"price mismatch for {sku}")
        self._lines[sku] = price
        self._qty[sku] = self._qty.get(sku, 0) + qty

    def remove(self, sku, qty=None):
        if sku not in self._qty:
            raise KeyError(sku)
        if qty is None or qty == self._qty[sku]:
            del self._qty[sku]
            del self._lines[sku]
            return
        if qty > self._qty[sku] or qty <= 0:
            raise ValueError("invalid quantity to remove")
        self._qty[sku] -= qty

    @property
    def items(self):
        return MappingProxyType(self._qty)

    @property
    def total(self):
        return sum((self._lines[s] * q for s, q in self._qty.items()), Decimal("0"))

import json


class DictMixin:
    def to_dict(self):
        return {k: self._convert(v) for k, v in vars(self).items() if not k.startswith("_")}

    @classmethod
    def _convert(cls, v):
        if hasattr(v, "to_dict"):
            return v.to_dict()
        if isinstance(v, (list, tuple)):
            return [cls._convert(x) for x in v]
        if isinstance(v, dict):
            return {k: cls._convert(x) for k, x in v.items()}
        return v

@runtime_checkable
class Exporter(Protocol):

class ExporterRegistry:
    def __init__(self):
        self._by_name = {}

    def register(self, exporter):
        if not isinstance(exporter, Exporter):

    def export(self, name, rows):

from decimal import Decimal, InvalidOperation


@dataclass
class Order:

    customer: str
    amount: Decimal


def parse_orders(csv_text):
    out = []
    for row in csv.DictReader(io.StringIO(csv_text)):
        try:
            amount = Decimal(row["amount"])
        except InvalidOperation:
            raise ValueError(f"bad amount {row['amount']!r}") from None

    problems = []
    if not order.customer.strip():

    if order.amount <= 0:

    by_customer = {}
    for o in orders:
        by_customer[o.customer] = by_customer.get(o.customer, Decimal("0")) + o.amount

    lines = [f"Orders: {summary['count']}", f"Total: {summary['total']:.2f}"]

    return "\n".join(lines)

    orders = [o for o in parse_orders(csv_text) if not validate(o)]

        if width <= 0 or height <= 0:
            raise ValueError("dimensions must be positive")

        self._h = height

    def width(self):
        return self._w

    def height(self):
        return self._h

from typing import Protocol, runtime_checkable


@runtime_checkable
class Reader(Protocol):
    def read(self, n: int = -1) -> bytes:
        ...


@runtime_checkable
class Writer(Protocol):
    def write(self, data: bytes) -> int:
        ...


def copy_stream(src, dst, chunk_size=65536):
    if chunk_size < 1:
        raise ValueError("chunk_size must be >= 1")
    total = 0
    while True:
        chunk = src.read(chunk_size)
        if not chunk:
            return total
        dst.write(chunk)
        total += len(chunk)


class MemoryReader:
    def __init__(self, data: bytes):
        self._data = data
        self._pos = 0

    def read(self, n: int = -1) -> bytes:
        if n < 0:
            n = len(self._data) - self._pos
        out = self._data[self._pos:self._pos + n]
        self._pos += len(out)
        return out


class MemoryWriter:
    def __init__(self):
        self._parts = []

    def write(self, data: bytes) -> int:
        self._parts.append(bytes(data))
        return len(data)

    def getvalue(self) -> bytes:
        return b"".join(self._parts)

Write an immutable fluent builder for SQL SELECT statements. `QueryBuilder(table)` has `select(*columns)`, `where(condition, *params)`, `order_by(column, desc=False)` and `limit(n)`; each returns a new builder and leaves the original unchanged, so partially built queries can be reused safely. `build() -> tuple[str, tuple]` returns the SQL and the parameters, e.g. `("SELECT id, name FROM users WHERE age > ? AND name = ? ORDER BY name DESC, id LIMIT 10", (18, "ann"))`: no `select` means `*`, multiple `where` conditions are joined with ` AND `, multiple `order_by` calls are joined with `, ` (`DESC` only when requested), and parameters keep the order of the `where` calls. Table and column names must be identifiers (`[A-Za-z_][A-Za-z0-9_]*`) or a `ValueError` is raised; `limit` must be a non-negative int.

import re

_IDENT = re.compile(r"[A-Za-z_][A-Za-z0-9_]*\Z")


def _check(name):
    if not isinstance(name, str) or not _IDENT.match(name):
        raise ValueError(f"invalid identifier {name!r}")
    return name


class QueryBuilder:
    def __init__(self, table, _columns=(), _where=(), _params=(), _order=(), _limit=None):
        self._table = _check(table)
        self._columns = _columns
        self._where = _where
        self._params = _params
        self._order = _order
        self._limit = _limit

    def _copy(self, **changes):
        state = dict(_columns=self._columns, _where=self._where, _params=self._params,
                     _order=self._order, _limit=self._limit)
        state.update(changes)
        return QueryBuilder(self._table, **state)

    def select(self, *columns):
        return self._copy(_columns=self._columns + tuple(_check(c) for c in columns))

    def where(self, condition, *params):
        return self._copy(_where=self._where + (condition,), _params=self._params + params)

    def order_by(self, column, desc=False):
        return self._copy(_order=self._order + (_check(column) + (" DESC" if desc else ""),))

    def limit(self, n):
        if not isinstance(n, int) or isinstance(n, bool) or n < 0:
            raise ValueError("limit must be a non-negative int")
        return self._copy(_limit=n)

    def build(self):
        sql = f"SELECT {', '.join(self._columns) or '*'} FROM {self._table}"
        if self._where:
            sql += " WHERE " + " AND ".join(self._where)
        if self._order:
            sql += " ORDER BY " + ", ".join(self._order)
        if self._limit is not None:
            sql += f" LIMIT {self._limit}"
        return sql, tuple(self._params)
