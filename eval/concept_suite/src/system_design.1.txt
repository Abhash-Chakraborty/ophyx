class LRUCache:
    def __init__(self, capacity: int):
        if not isinstance(capacity, int) or isinstance(capacity, bool) or capacity <= 0:
            raise ValueError("capacity must be a positive int")
        self.capacity = capacity
        self._data = OrderedDict()

        if key not in self._data:

        return self._data[key]

@@@ reference
from collections import OrderedDict


class EvictingLRU:
    def __init__(self, capacity: int, on_evict=None):
        if capacity < 1:
            raise ValueError("capacity must be >= 1")
        self.capacity = capacity
        self.on_evict = on_evict
        self._d = OrderedDict()

    def _shrink(self):
        while len(self._d) > self.capacity:
            k, v = self._d.popitem(last=False)
            if self.on_evict is not None:
                self.on_evict(k, v)

    def get(self, key, default=None):
        if key not in self._d:
            return default
        self._d.move_to_end(key)
        return self._d[key]

    def put(self, key, value):
        self._d[key] = value
        self._d.move_to_end(key)
        self._shrink()

    def peek(self, key):
        return self._d[key]

    def pop(self, key):
        return self._d.pop(key)

    def resize(self, new_capacity):
        if new_capacity < 1:
            raise ValueError("capacity must be >= 1")
        self.capacity = new_capacity
        self._shrink()

    def items(self):
        return list(self._d.items())

    def __len__(self):
        return len(self._d)

@@@ reference
from collections import OrderedDict, defaultdict


class LFUCache:
    def __init__(self, capacity: int):
        if capacity < 1:
            raise ValueError("capacity must be >= 1")
        self.capacity = capacity
        self._vals = {}
        self._freq = {}
        self._buckets = defaultdict(OrderedDict)
        self._min = 0

    def _touch(self, key):
        f = self._freq[key]
        del self._buckets[f][key]
        if not self._buckets[f]:
            del self._buckets[f]
            if self._min == f:
                self._min = f + 1
        self._freq[key] = f + 1
        self._buckets[f + 1][key] = None

    def get(self, key, default=None):
        if key not in self._vals:
            return default
        self._touch(key)
        return self._vals[key]

    def put(self, key, value):
        if key in self._vals:
            self._vals[key] = value
            self._touch(key)
            return
        if len(self._vals) >= self.capacity:
            old, _ = self._buckets[self._min].popitem(last=False)
            if not self._buckets[self._min]:
                del self._buckets[self._min]
            del self._vals[old]
            del self._freq[old]
        self._vals[key] = value
        self._freq[key] = 1
        self._buckets[1][key] = None
        self._min = 1

    def frequency(self, key):
        return self._freq[key]

    def __len__(self):
        return len(self._vals)

import time


class TTLCache:
    def __init__(self, ttl: float, clock=time.monotonic):
        if ttl <= 0:
            raise ValueError("ttl must be positive")
        self.ttl = ttl
        self.clock = clock
        self._data = {}

    def set(self, key, value, ttl=None):
        d = self.ttl if ttl is None else ttl
        if d <= 0:
            raise ValueError("ttl must be positive")
        self._data[key] = (value, self.clock() + d)

    def _live(self, key, now):
        item = self._data.get(key)
        return item is not None and now < item[1]

    def get(self, key, default=None):
        now = self.clock()
        if self._live(key, now):
            return self._data[key][0]
        self._data.pop(key, None)
        return default

    def __contains__(self, key):
        return self._live(key, self.clock())

    def __len__(self):
        now = self.clock()
        return sum(1 for k in self._data if self._live(k, now))

    def purge(self):
        now = self.clock()
        dead = [k for k, (_, exp) in self._data.items() if now >= exp]
        for k in dead:
            del self._data[k]
        return len(dead)

import functools
import time


def ttl_memoize(ttl: float, clock=time.monotonic):
    def deco(fn):
        cache = {}

        @functools.wraps(fn)
        def wrapper(*args, **kwargs):
            key = (args, tuple(sorted(kwargs.items())))
            now = clock()
            hit = cache.get(key)
            if hit is not None and now < hit[1]:
                return hit[0]
            value = fn(*args, **kwargs)
            cache[key] = (value, now + ttl)
            return value

        wrapper.clear = cache.clear
        return wrapper
    return deco

import time


class TokenBucket:
    def __init__(self, rate: float, capacity: float, clock=time.monotonic):
        if rate <= 0 or capacity <= 0:
            raise ValueError("rate and capacity must be positive")
        self.rate = rate
        self.capacity = capacity
        self.clock = clock
        self._tokens = float(capacity)
        self._last = clock()

    def _refill(self):
        now = self.clock()
        elapsed = max(0.0, now - self._last)
        self._tokens = min(self.capacity, self._tokens + elapsed * self.rate)
        self._last = now

    @property
    def tokens(self):
        self._refill()
        return self._tokens

    def _check(self, n):
        if n > self.capacity:
            raise ValueError("request exceeds capacity")

    def try_acquire(self, n=1):
        self._check(n)
        self._refill()
        if self._tokens >= n:
            self._tokens -= n
            return True
        return False

    def wait_time(self, n=1):
        self._check(n)
        self._refill()
        if self._tokens >= n:
            return 0.0
        return (n - self._tokens) / self.rate

import time


class KeyedRateLimiter:
    def __init__(self, rate: float, capacity: float, clock=time.monotonic, idle_ttl: float = 60.0):
        self.rate = rate
        self.capacity = capacity
        self.clock = clock
        self.idle_ttl = idle_ttl
        self._buckets = {}  # key -> [tokens, last_refill, last_used]

    def allow(self, key, cost=1):
        now = self.clock()
        b = self._buckets.get(key)
        if b is None:
            b = self._buckets[key] = [float(self.capacity), now, now]
        b[0] = min(self.capacity, b[0] + (now - b[1]) * self.rate)
        b[1] = now
        b[2] = now
        if b[0] >= cost:
            b[0] -= cost
            return True
        return False

    def cleanup(self):
        now = self.clock()
        dead = [k for k, b in self._buckets.items() if now - b[2] >= self.idle_ttl]
        for k in dead:
            del self._buckets[k]
        return len(dead)

    def __len__(self):
        return len(self._buckets)

import time
from collections import defaultdict, deque


class SlidingWindowLimiter:
    def __init__(self, limit: int, window: float, clock=time.monotonic):
        self.limit = limit
        self.window = window
        self.clock = clock
        self._log = defaultdict(deque)

    def _prune(self, key, now):
        q = self._log[key]
        while q and now >= q[0] + self.window:
            q.popleft()
        return q

    def allow(self, key="default"):
        now = self.clock()
        q = self._prune(key, now)
        if len(q) < self.limit:
            q.append(now)
            return True
        return False

    def remaining(self, key="default"):
        return max(0, self.limit - len(self._prune(key, self.clock())))

    def retry_after(self, key="default"):
        now = self.clock()
        q = self._prune(key, now)
        if len(q) < self.limit:
            return 0.0
        return q[0] + self.window - now

@@@ reference
import math
import time


class SlidingWindowCounter:
    def __init__(self, limit: int, window: float, clock=time.monotonic):
        self.limit = limit
        self.window = window
        self.clock = clock
        self._idx = None
        self._curr = 0
        self._prev = 0

    def _roll(self, now):
        i = math.floor(now / self.window)
        if self._idx is None:
            self._idx = i
        elif i != self._idx:
            self._prev = self._curr if i == self._idx + 1 else 0
            self._curr = 0
            self._idx = i
        return i

    def estimate(self):
        now = self.clock()
        i = self._roll(now)
        elapsed = now - i * self.window
        return self._prev * (self.window - elapsed) / self.window + self._curr

    def allow(self):
        if self.estimate() < self.limit - 1 + 1e-9:
            self._curr += 1
            return True
        return False

import heapq
import itertools
import time


class JobQueue:
    def __init__(self, max_attempts=3, base_delay=1.0, clock=time.monotonic):
        self.max_attempts = max_attempts
        self.base_delay = base_delay
        self.clock = clock
        self.results = {}
        self.dead_letters = []
        self._heap = []
        self._ids = itertools.count(1)

    def enqueue(self, fn, *args, **kwargs):
        jid = next(self._ids)
        heapq.heappush(self._heap, (self.clock(), jid, fn, args, kwargs, 0))
        return jid

    def run_due(self):
        now = self.clock()
        due = []
        while self._heap and self._heap[0][0] <= now:
            due.append(heapq.heappop(self._heap))
        for _, jid, fn, args, kwargs, attempts in due:
            attempts += 1
            try:
                self.results[jid] = fn(*args, **kwargs)
            except Exception as exc:
                if attempts >= self.max_attempts:
                    self.dead_letters.append((jid, exc))
                else:
                    when = now + self.base_delay * 2 ** (attempts - 1)
                    heapq.heappush(self._heap, (when, jid, fn, args, kwargs, attempts))
        return len(due)

    def pending(self):
        return len(self._heap)

import itertools
import time


class StaleReceipt(Exception):
    pass


class LeaseQueue:
    def __init__(self, visibility_timeout: float, max_receives: int = 3, clock=time.monotonic):
        self.visibility_timeout = visibility_timeout
        self.max_receives = max_receives
        self.clock = clock
        self.dead_letter = []
        self._msgs = {}  # id -> dict(body, visible_at, receives, receipt)
        self._ids = itertools.count(1)
        self._receipts = itertools.count(1)

    def put(self, body):
        mid = next(self._ids)
        self._msgs[mid] = {"body": body, "visible_at": float("-inf"), "receives": 0, "receipt": None}
        return mid

    def receive(self):
        now = self.clock()
        for mid in sorted(self._msgs):
            m = self._msgs[mid]
            if m["visible_at"] > now:
                continue
            if m["receives"] >= self.max_receives:
                self.dead_letter.append(m["body"])
                del self._msgs[mid]
                continue
            m["receives"] += 1
            m["visible_at"] = now + self.visibility_timeout
            m["receipt"] = f"r{next(self._receipts)}"
            return mid, m["body"], m["receipt"]
        return None

    def _find(self, receipt):
        now = self.clock()
        for mid, m in self._msgs.items():
            if m["receipt"] == receipt and m["visible_at"] > now:
                return mid
        raise StaleReceipt(receipt)

    def ack(self, receipt):
        del self._msgs[self._find(receipt)]

    def nack(self, receipt):
        m = self._msgs[self._find(receipt)]
        m["visible_at"] = float("-inf")
        m["receipt"] = None

    def __len__(self):
        return len(self._msgs)

import heapq
import itertools


class TaskScheduler:
    def __init__(self):
        self._heap = []
        self._live = {}
        self._seq = itertools.count()

    def push(self, task, priority: int) -> int:
        h = next(self._seq)
        heapq.heappush(self._heap, (priority, h, task))
        self._live[h] = task
        return h

    def _clean(self):
        while self._heap and self._heap[0][1] not in self._live:
            heapq.heappop(self._heap)

    def peek(self):
        self._clean()
        if not self._heap:
            raise IndexError("peek from empty scheduler")
        return self._heap[0][2]

    def pop(self):
        self._clean()
        if not self._heap:
            raise IndexError("pop from empty scheduler")
        _, h, task = heapq.heappop(self._heap)
        del self._live[h]
        return task

    def cancel(self, handle) -> bool:
        if handle in self._live:
            del self._live[handle]
            return True
        return False

    def __len__(self):
        return len(self._live)

import heapq
import itertools
import math
import time


class DelayedScheduler:
    def __init__(self, clock=time.monotonic):
        self.clock = clock
        self._heap = []
        self._seq = itertools.count()
        self._active = {}  # handle -> interval or None

    def _push(self, when, handle, fn):
        heapq.heappush(self._heap, (when, next(self._seq), handle, fn))

    def call_at(self, when, fn):
        h = next(self._seq)
        self._active[h] = None
        self._push(when, h, fn)
        return h

    def call_later(self, delay, fn):
        return self.call_at(self.clock() + delay, fn)

    def call_every(self, interval, fn):
        if interval <= 0:
            raise ValueError("interval must be positive")
        h = self.call_at(self.clock() + interval, fn)
        self._active[h] = interval
        return h

    def cancel(self, handle):
        return self._active.pop(handle, False) is not False

    def run_pending(self):
        now = self.clock()
        due = []
        while self._heap and self._heap[0][0] <= now:
            due.append(heapq.heappop(self._heap))
        ran = 0
        for when, _, h, fn in due:
            if h not in self._active:
                continue
            interval = self._active[h]
            if interval is None:
                del self._active[h]
            else:
                k = max(1, math.floor((now - when) / interval) + 1)
                nxt = when + k * interval
                while nxt <= now:
                    nxt += interval
                self._push(nxt, h, fn)
            fn()
            ran += 1
        return ran

        self._broker = broker
        self.pattern = pattern
        self.callback = callback
        self.active = True

    def unsubscribe(self):
        if self.active:

def _match(pat, top):
    if not pat:
        return not top
    head = pat[0]
    if head == "#":
        return any(_match(pat[1:], top[i:]) for i in range(len(top) + 1))
    if not top:
        return False
    return (head == "*" or head == top[0]) and _match(pat[1:], top[1:])

        self._subs = []
        self.errors = []

    def subscribe(self, pattern, callback):
        s = _Subscription(self, pattern, callback)
        self._subs.append(s)
        return s

    def publish(self, topic, message):
        parts = topic.split(".")
        n = 0
        for s in list(self._subs):
            if not s.active or not _match(s.pattern.split("."), parts):

            n += 1
            try:
                s.callback(topic, message)
            except Exception as exc:

        return n

from collections import defaultdict, deque


class _Subscriber:
    def __init__(self, bus, topic, maxsize):
        self._bus = bus
        self.topic = topic
        self._buf = deque()
        self.maxsize = maxsize
        self.dropped = 0

    def _offer(self, msg):
        if len(self._buf) >= self.maxsize:
            self._buf.popleft()
            self.dropped += 1
        self._buf.append(msg)

    def poll(self, max_messages=10):
        out = []
        while self._buf and len(out) < max_messages:
            out.append(self._buf.popleft())
        return out

    def close(self):
        subs = self._bus._subs[self.topic]
        if self in subs:
            subs.remove(self)

    def __init__(self):
        self._subs = defaultdict(list)

    def subscribe(self, topic, maxsize=100):
        s = _Subscriber(self, topic, maxsize)
        self._subs[topic].append(s)
        return s

    def publish(self, topic, message):
        subs = self._subs.get(topic, [])
        for s in subs:
            s._offer(message)
        return len(subs)

from collections import defaultdict


class EventBus:
    def __init__(self):
        self._handlers = defaultdict(list)

    def subscribe(self, event_type, handler):
        if handler not in self._handlers[event_type]:
            self._handlers[event_type].append(handler)

    def on(self, event_type):
        def deco(fn):
            self.subscribe(event_type, fn)
            return fn
        return deco

    def unsubscribe(self, event_type, handler):
        hs = self._handlers.get(event_type, [])
        if handler in hs:
            hs.remove(handler)
            return True
        return False

    def publish(self, event):
        n = 0
        for cls in type(event).__mro__:
            for h in list(self._handlers.get(cls, ())):
                h(event)
                n += 1
        return n

import asyncio
import inspect


class AsyncEventBus:
    def __init__(self):
        self._subs = []

    def subscribe(self, event_type, handler):
        self._subs.append((event_type, handler))

    async def publish(self, event):
        errors = []
        coros = []
        n = 0
        for typ, h in list(self._subs):
            if not isinstance(event, typ):
                continue
            n += 1
            if inspect.iscoroutinefunction(h):
                coros.append(h(event))
            else:
                try:
                    h(event)
                except Exception as exc:
                    errors.append(exc)
        results = await asyncio.gather(*coros, return_exceptions=True)
        errors.extend(r for r in results if isinstance(r, BaseException))
        if errors:
            raise ExceptionGroup("event handlers failed", errors)
        return n

import threading
import time
from contextlib import contextmanager


class PoolTimeout(Exception):
    pass


class ConnectionPool:
    def __init__(self, factory, max_size: int, timeout: float = 5.0):
        self._factory = factory
        self._max = max_size
        self._timeout = timeout
        self._idle = []
        self._out = set()
        self._created = 0
        self._cond = threading.Condition()

    @property
    def size(self):
        return self._created

    @property
    def idle(self):
        with self._cond:
            return len(self._idle)

    def acquire(self, timeout=None):
        timeout = self._timeout if timeout is None else timeout
        deadline = time.monotonic() + timeout
        with self._cond:
            while True:
                if self._idle:
                    conn = self._idle.pop()
                    break
                if self._created < self._max:
                    self._created += 1
                    try:
                        conn = self._factory()
                    except BaseException:
                        self._created -= 1
                        raise
                    break
                remaining = deadline - time.monotonic()
                if remaining <= 0:
                    raise PoolTimeout(f"no connection within {timeout}s")
                self._cond.wait(remaining)
            self._out.add(id(conn))
            return conn

    def release(self, conn):
        with self._cond:
            if id(conn) not in self._out:
                raise ValueError("connection not checked out from this pool")
            self._out.discard(id(conn))
            self._idle.append(conn)
            self._cond.notify()

    @contextmanager
    def connection(self):
        conn = self.acquire()
        try:
            yield conn
        finally:
            self.release(conn)

class PoolClosed(Exception):
    pass


class PoolExhausted(Exception):
    pass


class HealthCheckedPool:
    def __init__(self, factory, max_size: int, is_healthy):
        self._factory = factory
        self._max = max_size
        self._healthy = is_healthy
        self._idle = []
        self._in_use = []
        self._closed = False

    def _total(self):
        return len(self._idle) + len(self._in_use)

    def acquire(self):
        if self._closed:
            raise PoolClosed("pool is closed")
        while self._idle:
            conn = self._idle.pop()
            if self._healthy(conn):
                self._in_use.append(conn)
                return conn
            conn.close()
        if self._total() >= self._max:
            raise PoolExhausted("all connections in use")
        conn = self._factory()
        self._in_use.append(conn)
        return conn

    def _take_back(self, conn):
        for i, c in enumerate(self._in_use):
            if c is conn:
                del self._in_use[i]
                return
        raise ValueError("connection not in use")

    def release(self, conn):
        self._take_back(conn)
        if self._closed:
            conn.close()
        else:
            self._idle.append(conn)

    def discard(self, conn):
        self._take_back(conn)
        conn.close()

    def close(self):
        self._closed = True
        while self._idle:
            self._idle.pop().close()

    def stats(self):
        return {"idle": len(self._idle), "in_use": len(self._in_use), "total": self._total()}

    cs = [pool.acquire() for _ in range(3)]
    pool.release(cs[0])
    assert pool.stats() == {"idle": 1, "in_use": 2, "total": 3}
