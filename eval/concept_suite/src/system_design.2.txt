import json
import os
import zlib


class CorruptLog(Exception):
    pass


def _parse(line):
    if not line.endswith(b"\n"):
        return None
    try:
        crc_hex, payload = line[:-1].split(b" ", 1)
        if int(crc_hex, 16) != zlib.crc32(payload):
            return None
        return json.loads(payload)
    except ValueError:
        return None


class WALStore:
    def __init__(self, path):
        self.path = path
        self._data = {}
        self._replay()
        self._f = open(path, "ab")

    def _apply(self, rec):
        if rec["op"] == "set":
            self._data[rec["k"]] = rec["v"]
        else:
            self._data.pop(rec["k"], None)

    def _replay(self):
        if not os.path.exists(self.path):
            return
        with open(self.path, "rb") as f:
            lines = f.readlines()
        good_bytes = 0
        for i, line in enumerate(lines):
            rec = _parse(line)
            if rec is None:
                if i != len(lines) - 1:
                    raise CorruptLog(f"corrupt record at line {i + 1}")
                with open(self.path, "r+b") as f:
                    f.truncate(good_bytes)
                break
            self._apply(rec)
            good_bytes += len(line)

    def _append(self, rec):
        payload = json.dumps(rec, separators=(",", ":"))
        line = f"{zlib.crc32(payload.encode()):08x} {payload}\n".encode()
        self._f.write(line)
        self._f.flush()
        os.fsync(self._f.fileno())
        self._apply(rec)

    def set(self, key, value):
        self._append({"op": "set", "k": key, "v": value})

    def delete(self, key):
        self._append({"op": "del", "k": key})

    def get(self, key, default=None):
        return self._data.get(key, default)

    def __len__(self):
        return len(self._data)

    def close(self):
        self._f.close()

    def __enter__(self):
        return self

    def __exit__(self, *exc):
        self.close()

import json
import os


class LogStore:
    def __init__(self, path):
        self.path = str(path)
        self._index = {}
        if os.path.exists(self.path):
            self._scan()
        self._f = open(self.path, "ab+")

    def _scan(self):
        self._index = {}
        with open(self.path, "rb") as f:
            off = 0
            for line in f:
                rec = json.loads(line)
                if rec.get("d"):
                    self._index.pop(rec["k"], None)
                else:
                    self._index[rec["k"]] = off
                off += len(line)

    def _append(self, rec):
        self._f.seek(0, os.SEEK_END)
        off = self._f.tell()
        self._f.write((json.dumps(rec) + "\n").encode())
        self._f.flush()
        return off

    def put(self, key, value):
        self._index[key] = self._append({"k": key, "v": value})

    def delete(self, key):
        if key in self._index:
            self._append({"k": key, "d": True})
            del self._index[key]

    def get(self, key):
        off = self._index[key]
        self._f.seek(off)
        return json.loads(self._f.readline())["v"]

    def keys(self):
        return sorted(self._index)

    def compact(self):
        before = os.path.getsize(self.path)
        tmp = self.path + ".compact"
        with open(tmp, "wb") as out:
            for k in sorted(self._index):
                out.write((json.dumps({"k": k, "v": self.get(k)}) + "\n").encode())
        self._f.close()
        os.replace(tmp, self.path)
        self._scan()
        self._f = open(self.path, "ab+")
        return before - os.path.getsize(self.path)

    def close(self):
        self._f.close()

import string
from urllib.parse import urlsplit

ALPHABET = string.digits + string.ascii_lowercase + string.ascii_uppercase
_INDEX = {c: i for i, c in enumerate(ALPHABET)}


def encode(n: int) -> str:
    if n < 0:
        raise ValueError("negative")
    if n == 0:
        return "0"
    out = []
    while n:
        n, r = divmod(n, 62)
        out.append(ALPHABET[r])
    return "".join(reversed(out))


def decode(s: str) -> int:
    if not s:
        raise ValueError("empty code")
    n = 0
    for ch in s:
        if ch not in _INDEX:
            raise ValueError(f"invalid character {ch!r}")
        n = n * 62 + _INDEX[ch]
    return n


class Shortener:
    def __init__(self, base_url="https://sho.rt/"):
        self.base_url = base_url
        self._next = 1
        self._by_code = {}
        self._by_url = {}

    def shorten(self, url):
        parts = urlsplit(url)
        if parts.scheme not in ("http", "https") or not parts.netloc:
            raise ValueError(f"not an absolute http(s) URL: {url!r}")
        if url not in self._by_url:
            code = encode(self._next)
            self._next += 1
            self._by_url[url] = code
            self._by_code[code] = url
        return self.base_url + self._by_url[url]

    def resolve(self, code_or_short_url):
        code = code_or_short_url
        if code.startswith(self.base_url):
            code = code[len(self.base_url):]
        return self._by_code[code]

import re
import secrets
import string
import time

_ALPHA = string.digits + string.ascii_letters
_ALIAS_RE = re.compile(r"[A-Za-z0-9_-]{3,32}")


class AliasTaken(Exception):
    pass


class LinkExpired(Exception):
    pass


class LinkService:
    def __init__(self, clock=time.time):
        self.clock = clock
        self._links = {}

    def create(self, url, alias=None, ttl=None):
        if alias is not None:
            if not _ALIAS_RE.fullmatch(alias):
                raise ValueError(f"invalid alias {alias!r}")
            if alias in self._links:
                raise AliasTaken(alias)
            code = alias
        else:
            while True:
                code = "".join(secrets.choice(_ALPHA) for _ in range(6))
                if code not in self._links:
                    break
        now = self.clock()
        self._links[code] = {"url": url, "clicks": 0, "created": now,
                             "expires": None if ttl is None else now + ttl}
        return code

    def _get(self, code):
        link = self._links[code]
        if link["expires"] is not None and self.clock() >= link["expires"]:
            raise LinkExpired(code)
        return link

    def visit(self, code):
        link = self._get(code)
        link["clicks"] += 1
        return link["url"]

    def stats(self, code):
        return dict(self._links[code])

class Account:
    def __init__(self):
        self.id = None
        self.owner = None
        self.balance = 0
        self.closed = False
        self.version = 0
        self.pending_events = []

    @classmethod
    def open(cls, account_id, owner):
        acc = cls()
        acc._record(Opened(account_id, owner))
        return acc

    @classmethod
    def from_events(cls, events):
        acc = cls()
        for e in events:
            acc._apply(e)
        return acc

    def _apply(self, e):
        if isinstance(e, Opened):
            self.id, self.owner = e.account_id, e.owner
        elif isinstance(e, Deposited):
            self.balance += e.amount
        elif isinstance(e, Withdrawn):
            self.balance -= e.amount
        elif isinstance(e, Closed):
            self.closed = True
        else:
            raise DomainError(f"unknown event {e!r}")
        self.version += 1

    def _record(self, e):
        self._apply(e)
        self.pending_events.append(e)

        if self.closed:
            raise DomainError("account is closed")

    def _check_amount(amount):
        if not isinstance(amount, int) or isinstance(amount, bool) or amount <= 0:
            raise DomainError("amount must be a positive int")

        self._check_open()
        self._check_amount(amount)
        self._record(Deposited(amount))

        self._check_open()
        self._check_amount(amount)
        if amount > self.balance:

        self._record(Withdrawn(amount))

        self._check_open()

class ConcurrencyError(Exception):
    pass


class EventStore:
    def __init__(self):
        self._streams = {}
        self._log = []
        self._subs = []

    def append(self, stream_id, events, expected_version):
        stream = self._streams.setdefault(stream_id, [])
        if len(stream) != expected_version:
            raise ConcurrencyError(f"{stream_id}: expected {expected_version}, at {len(stream)}")
        for e in events:
            stream.append(e)
            self._log.append((stream_id, e))
        for e in events:
            for p in self._subs:
                p.handle(stream_id, e)
        return len(stream)

    def load(self, stream_id):
        return list(self._streams.get(stream_id, []))

    def all_events(self):
        return list(self._log)

    def subscribe(self, projection):
        self._subs.append(projection)


class CartTotals:
    def __init__(self):
        self._lines = {}
        self._frozen = set()

    def handle(self, stream_id, event):
        if stream_id in self._frozen:
            return
        lines = self._lines.setdefault(stream_id, {})
        t = event["type"]
        if t == "ItemAdded":
            price, qty = lines.get(event["sku"], (event["price"], 0))
            lines[event["sku"]] = (event["price"], qty + event["qty"])
        elif t == "ItemRemoved":
            lines.pop(event["sku"], None)
        elif t == "CheckedOut":
            self._frozen.add(stream_id)

    def total(self, cart_id):
        return sum(p * q for p, q in self._lines.get(cart_id, {}).values())

    @classmethod
    def rebuild(cls, store):
        proj = cls()
        for sid, e in store.all_events():
            proj.handle(sid, e)
        return proj

import importlib.util
import itertools
import os

_counter = itertools.count()


class PluginRegistry:
    def __init__(self):
        self._plugins = {}
        self.skipped = []
        self.errors = {}

    def register(self, name, obj):
        if name in self._plugins:
            raise ValueError(f"duplicate plugin {name!r}")
        self._plugins[name] = obj

    def get(self, name):
        return self._plugins[name]

    def names(self):
        return sorted(self._plugins)

    def load_directory(self, path):
        loaded = []
        for fname in sorted(os.listdir(path)):
            if not fname.endswith(".py") or fname.startswith("_"):
                continue
            stem = fname[:-3]
            modname = f"_plugin_{next(_counter)}_{stem}"
            try:
                spec = importlib.util.spec_from_file_location(modname, os.path.join(path, fname))
                mod = importlib.util.module_from_spec(spec)
                spec.loader.exec_module(mod)
                reg = getattr(mod, "register", None)
                if not callable(reg):
                    self.skipped.append(stem)
                    continue
                reg(self)
            except Exception as exc:
                self.errors[stem] = exc
                continue
            loaded.append(stem)
        return loaded

import importlib


def load_object(spec: str):
    if spec.count(":") != 1:
        raise ValueError(f"spec must look like 'module:attr', got {spec!r}")
    mod_name, attr_path = spec.split(":")
    if not mod_name or not attr_path:
        raise ValueError(f"spec must look like 'module:attr', got {spec!r}")
    obj = importlib.import_module(mod_name)
    for part in attr_path.split("."):
        try:
            obj = getattr(obj, part)
        except AttributeError:
            raise AttributeError(f"{spec!r}: no attribute {part!r}") from None
    return obj

    def get(self, name):
        if name not in self._cache:
            self._cache[name] = load_object(self._specs[name])
        return self._cache[name]

    def loaded(self):
        return sorted(self._cache)

def _merge(base, extra):
    for k, v in extra.items():
        if isinstance(v, dict) and isinstance(base.get(k), dict):
            _merge(base[k], v)
        else:
            base[k] = copy.deepcopy(v)

_TRUE = {"1", "true", "yes", "on"}
_FALSE = {"0", "false", "no", "off"}


def _coerce(raw, default):
    if isinstance(default, bool):
        low = raw.strip().lower()
        if low in _TRUE:
            return True
        if low in _FALSE:
            return False
        raise ValueError(f"not a boolean: {raw!r}")
    if isinstance(default, int):
        return int(raw)
    if isinstance(default, float):
        return float(raw)
    return raw


def load_config(defaults, file_path=None, env=None, overrides=None, prefix="APP_"):
    cfg = copy.deepcopy(defaults)
    if file_path is not None:
        with open(file_path) as f:
            _merge(cfg, json.load(f))
    for name, raw in (env or {}).items():
        if not name.startswith(prefix):
            continue
        path = name[len(prefix):].lower().split("__")
        node, ref = cfg, defaults
        ok = True
        for p in path[:-1]:
            if not isinstance(ref, dict) or not isinstance(ref.get(p), dict):
                ok = False
                break
            node, ref = node.setdefault(p, {}), ref[p]
        leaf = path[-1]
        if ok and isinstance(ref, dict) and leaf in ref and not isinstance(ref[leaf], dict):
            node[leaf] = _coerce(raw, ref[leaf])
    if overrides:
        _merge(cfg, overrides)
    return cfg

def _lookup(mapping, parts):
    node = mapping
    for p in parts:
        if not isinstance(node, dict) or p not in node:
            return _MISSING
        node = node[p]

def _merge(low, high):
    out = dict(low)
    for k, v in high.items():
        if isinstance(v, dict) and isinstance(out.get(k), dict):
            out[k] = _merge(out[k], v)
        else:
            out[k] = v
    return out


class LayeredConfig:
    def __init__(self, *layers):
        self.layers = list(layers)

    def _values(self, key):
        parts = key.split(".")
        return [(name, v) for name, m in self.layers if (v := _lookup(m, parts)) is not _MISSING]

    def get(self, key, default=_MISSING):
        vals = self._values(key)
        if not vals:
            if default is _MISSING:
                raise KeyError(key)
            return default
        top = vals[0][1]
        if not isinstance(top, dict):
            return top
        result = {}
        for _, v in reversed(vals):

        return result

import hashlib


class FeatureFlags:
    def __init__(self):
        self._flags = {}

    def define(self, name, enabled=True, rollout=100, allow=(), deny=()):
        if not 0 <= rollout <= 100:
            raise ValueError("rollout must be between 0 and 100")
        self._flags[name] = {"enabled": enabled, "rollout": rollout, "allow": set(allow), "deny": set(deny)}

    @staticmethod
    def bucket(name, user_id):
        return int(hashlib.sha256(f"{name}:{user_id}".encode()).hexdigest(), 16) % 100

            if user_id in f["allow"]:

        if f["rollout"] >= 100:
            return True
        if user_id is None:
            return False
        return self.bucket(name, user_id) < f["rollout"]

import operator

_OPS = {
    "eq": operator.eq,
    "neq": operator.ne,
    "in": lambda a, v: a in v,
    "not_in": lambda a, v: a not in v,
    "gte": operator.ge,
    "lt": operator.lt,
    "contains": lambda a, v: v in a,
}
_KEYS = {"attribute", "op", "value", "result"}


class Flag:
    def __init__(self, name, rules, default=False):
        for r in rules:
            if not _KEYS <= set(r):
                raise ValueError(f"rule missing keys: {_KEYS - set(r)}")
            if r["op"] not in _OPS:
                raise ValueError(f"unknown op {r['op']!r}")
        self.name = name
        self.rules = list(rules)
        self.default = default


def evaluate(flag, context):
    for r in flag.rules:
        if r["attribute"] not in context:
            continue
        try:
            matched = _OPS[r["op"]](context[r["attribute"]], r["value"])
        except TypeError:
            continue
        if matched:
            return r["result"]
    return flag.default

import bisect
import hashlib


def _h(s):
    return int(hashlib.md5(s.encode()).hexdigest(), 16)


class HashRing:
    def __init__(self, nodes=(), replicas=100):
        self.replicas = replicas
        self._points = []
        self._owner = {}
        self._nodes = set()
        for n in nodes:
            self.add_node(n)

    def add_node(self, node):
        if node in self._nodes:
            return
        self._nodes.add(node)
        for i in range(self.replicas):
            p = _h(f"{node}#{i}")
            self._owner[p] = node
            bisect.insort(self._points, p)

    def remove_node(self, node):
        if node not in self._nodes:
            raise KeyError(node)
        self._nodes.discard(node)
        dead = {p for p, n in self._owner.items() if n == node}
        for p in dead:
            del self._owner[p]
        self._points = [p for p in self._points if p not in dead]

    def _start(self, key):
        if not self._points:
            raise LookupError("empty ring")
        return bisect.bisect_right(self._points, _h(key)) % len(self._points)

    def get_node(self, key):
        return self._owner[self._points[self._start(key)]]

    def get_nodes(self, key, n):
        i = self._start(key)
        out = []
        for k in range(len(self._points)):
            node = self._owner[self._points[(i + k) % len(self._points)]]
            if node not in out:
                out.append(node)
                if len(out) == n:
                    break
        return out

import hashlib
import math


class BloomFilter:
    def __init__(self, capacity: int, error_rate: float = 0.01):
        if capacity <= 0 or not 0 < error_rate < 1:
            raise ValueError("bad capacity or error rate")
        self.capacity = capacity
        self.error_rate = error_rate
        self.size_bits = math.ceil(-capacity * math.log(error_rate) / math.log(2) ** 2)
        self.num_hashes = max(1, round(self.size_bits / capacity * math.log(2)))
        self._bits = bytearray((self.size_bits + 7) // 8)
        self._count = 0

    def _positions(self, item):
        if isinstance(item, str):
            item = item.encode()
        d = hashlib.sha256(item).digest()
        h1 = int.from_bytes(d[:8], "big")
        h2 = int.from_bytes(d[8:16], "big")
        return [(h1 + i * h2) % self.size_bits for i in range(self.num_hashes)]

    def add(self, item):
        for p in self._positions(item):
            self._bits[p >> 3] |= 1 << (p & 7)
        self._count += 1

    def __contains__(self, item):
        return all(self._bits[p >> 3] & (1 << (p & 7)) for p in self._positions(item))

    def __len__(self):
        return self._count

    def union(self, other):
        if (self.size_bits, self.num_hashes) != (other.size_bits, other.num_hashes):
            raise ValueError("incompatible filters")
        out = BloomFilter(self.capacity, self.error_rate)
        out._bits = bytearray(a | b for a, b in zip(self._bits, other._bits))
        out._count = self._count + other._count
        return out

class Autocomplete:
    def __init__(self):
        self._root = {}
        self._count = 0

    def add(self, word, weight=1):
        node = self._root
        for ch in word.lower():
            node = node.setdefault(ch, {})
        if "$" not in node:
            node["$"] = 0
            self._count += 1
        node["$"] += weight

    def _find(self, prefix):
        node = self._root
        for ch in prefix:
            node = node.get(ch)
            if node is None:
                return None
        return node

    def suggest(self, prefix, limit=5):
        prefix = prefix.lower()
        node = self._find(prefix)
        if node is None:
            return []
        found = []
        stack = [(node, prefix)]
        while stack:
            n, path = stack.pop()
            for ch, child in n.items():
                if ch == "$":
                    found.append((-child, path))
                else:
                    stack.append((child, path + ch))
        found.sort()
        return [w for _, w in found[:limit]]

    def remove(self, word):
        word = word.lower()
        path = [self._root]
        for ch in word:
            nxt = path[-1].get(ch)
            if nxt is None:
                return False
            path.append(nxt)
        if "$" not in path[-1]:
            return False
        del path[-1]["$"]
        self._count -= 1
        for i in range(len(word), 0, -1):
            if path[i]:
                break
            del path[i - 1][word[i - 1]]
        return True

    def __contains__(self, word):
        node = self._find(word.lower())
        return node is not None and "$" in node

    def __len__(self):
        return self._count

Duplicate inserts count once. `count_prefix` must run in O(len(prefix)) by keeping a counter on every node of how many words pass through it (it counts words equal to the prefix too; the empty prefix gives the total). `longest_common_prefix()` returns the longest prefix shared by all stored words ("" if the trie is empty). `delete` returns False for words not present and keeps all counters correct.

class _Node:
    __slots__ = ("children", "count", "end")

    def __init__(self):
        self.children = {}
        self.count = 0
        self.end = False


class Trie:
    def __init__(self):
        self._root = _Node()

    def _walk(self, s):
        node = self._root
        for ch in s:
            node = node.children.get(ch)
            if node is None:
                return None
        return node

    def contains(self, word):
        n = self._walk(word)
        return n is not None and n.end

    def insert(self, word):
        if self.contains(word):
            return
        node = self._root
        node.count += 1
        for ch in word:
            node = node.children.setdefault(ch, _Node())
            node.count += 1
        node.end = True

    def delete(self, word):
        if not self.contains(word):
            return False
        node = self._root
        node.count -= 1
        for ch in word:
            child = node.children[ch]
            child.count -= 1
            if child.count == 0:
                del node.children[ch]
                return True
            node = child
        node.end = False
        return True

    def count_prefix(self, prefix):
        n = self._walk(prefix)
        return 0 if n is None else n.count

    def longest_common_prefix(self):
        node, out = self._root, []
        if node.count == 0:
            return ""
        while len(node.children) == 1 and not node.end:
            ch, node = next(iter(node.children.items()))
            out.append(ch)
        return "".join(out)

    assert t.delete("app") is False and t.delete("xyz") is False
    t.delete("apple")
    t.delete("apply")
    assert t.count_prefix("") == 0 and t.count_prefix("a") == 0


def test_longest_common_prefix():
    assert make(["flower", "flow", "flight"]).longest_common_prefix() == "fl"
    assert make(["interview", "internet", "interval"]).longest_common_prefix() == "inter"
    assert make(["same"]).longest_common_prefix() == "same"
    assert Trie().longest_common_prefix() == ""


def test_lcp_stops_at_word_end():
    assert make(["ab", "abc"]).longest_common_prefix() == "ab"
    assert make(["", "abc"]).longest_common_prefix() == ""


def test_reinsert_after_delete():
    t = make(["cat"])
    t.delete("cat")
    t.insert("cat")
    assert t.contains("cat") and t.count_prefix("c") == 1
