from collections import defaultdict


class OutOfStock(Exception):
    def __init__(self, sku, requested, available):
        super().__init__(f"{sku}: requested {requested}, available {available}")
        self.sku = sku


def _check_qty(qty):
    if not isinstance(qty, int) or isinstance(qty, bool) or qty <= 0:
        raise ValueError(f"quantity must be a positive int, got {qty!r}")


class Inventory:
    def __init__(self):
        self._on_hand = defaultdict(int)
        self._reserved = defaultdict(int)
        self._orders = {}

    def add_stock(self, sku, qty):
        _check_qty(qty)
        self._on_hand[sku] += qty

    def on_hand(self, sku):
        return self._on_hand.get(sku, 0)

    def available(self, sku):
        return self._on_hand.get(sku, 0) - self._reserved.get(sku, 0)

    def reserve(self, order_id, items):
        if order_id in self._orders:
            raise ValueError(f"order {order_id!r} already has a reservation")
        for qty in items.values():
            _check_qty(qty)
        for sku, qty in items.items():
            if self.available(sku) < qty:
                raise OutOfStock(sku, qty, self.available(sku))
        for sku, qty in items.items():
            self._reserved[sku] += qty
        self._orders[order_id] = dict(items)

    def release(self, order_id):
        for sku, qty in self._orders.pop(order_id).items():
            self._reserved[sku] -= qty

    def commit(self, order_id):
        for sku, qty in self._orders.pop(order_id).items():
            self._reserved[sku] -= qty
            self._on_hand[sku] -= qty

    def check_invariants(self):
        for sku in set(self._on_hand) | set(self._reserved):
            assert self._on_hand[sku] >= 0, sku
            assert 0 <= self._reserved[sku] <= self._on_hand[sku], sku

from decimal import Decimal


class LedgerError(Exception):
    pass


def _amount(x):
    if isinstance(x, float):
        raise TypeError("use Decimal, int or str for money, not float")
    if isinstance(x, bool) or not isinstance(x, (int, str, Decimal)):
        raise TypeError(f"bad amount type {type(x).__name__}")
    return Decimal(x)


class Ledger:
    def __init__(self):
        self._bal = {}
        self._neg = {}
        self._hist = {}

    def open(self, name, allow_negative=False):
        if name in self._bal:
            raise LedgerError(f"account {name!r} exists")
        self._bal[name] = Decimal(0)
        self._neg[name] = allow_negative
        self._hist[name] = []

    def post(self, entries, memo=""):
        legs = [(acct, _amount(amt)) for acct, amt in entries]
        if not legs:
            raise LedgerError("no legs")
        if sum(a for _, a in legs) != 0:
            raise LedgerError("legs do not balance")
        new = {}
        for acct, amt in legs:
            if acct not in self._bal:
                raise LedgerError(f"unknown account {acct!r}")
            new[acct] = new.get(acct, self._bal[acct]) + amt
        for acct, bal in new.items():
            if bal < 0 and not self._neg[acct]:
                raise LedgerError(f"{acct} would go negative")
        self._bal.update(new)
        for acct, amt in legs:
            self._hist[acct].append((memo, amt))

    def balance(self, name):
        return self._bal[name]

    def history(self, name):
        return list(self._hist[name])

    def total(self):
        return sum(self._bal.values(), Decimal(0))

import json
import threading
from http.server import BaseHTTPRequestHandler, ThreadingHTTPServer


class _Handler(BaseHTTPRequestHandler):
    def log_message(self, fmt, *args):
        pass

    def _send(self, status, obj):
        body = json.dumps(obj).encode()
        self.send_response(status)
        self.send_header("Content-Type", "application/json")
        self.send_header("Content-Length", str(len(body)))
        self.end_headers()
        self.wfile.write(body)

    def do_GET(self):
        srv = self.server
        parts = self.path.rstrip("/").split("/")
        if self.path == "/notes":
            with srv.lock:
                return self._send(200, list(srv.notes.values()))
        if len(parts) == 3 and parts[1] == "notes" and parts[2].isdigit():
            with srv.lock:
                note = srv.notes.get(int(parts[2]))
            if note is not None:
                return self._send(200, note)
            return self._send(404, {"error": "note not found"})
        self._send(404, {"error": "not found"})

    def do_POST(self):
        if self.path != "/notes":
            return self._send(404, {"error": "not found"})
        length = int(self.headers.get("Content-Length") or 0)
        raw = self.rfile.read(length)
        try:
            data = json.loads(raw)
        except ValueError:
            return self._send(400, {"error": "invalid JSON"})
        if not isinstance(data, dict) or not isinstance(data.get("text"), str):
            return self._send(400, {"error": "'text' must be a string"})
        srv = self.server
        with srv.lock:
            srv.next_id += 1
            note = {"id": srv.next_id, "text": data["text"]}
            srv.notes[note["id"]] = note
        self._send(201, note)


def make_server(port=0):
    srv = ThreadingHTTPServer(("127.0.0.1", port), _Handler)
    srv.notes = {}
    srv.next_id = 0
    srv.lock = threading.Lock()
    srv.daemon_threads = True
    return srv

import re
from typing import Any, NamedTuple


class Response(NamedTuple):
    status: int
    body: Any


_PARAM = re.compile(r"\{(\w+)(?::(int))?\}")


class Router:
    def __init__(self):
        self._routes = []  # (static, regex, converters, methods, handler)

    def route(self, pattern, methods=("GET",)):
        convs = {}

        def repl(m):
            name, kind = m.group(1), m.group(2)
            convs[name] = int if kind == "int" else str
            return rf"(?P<{name}>\d+)" if kind == "int" else rf"(?P<{name}>[^/]+)"
        parts = _PARAM.split(pattern)
        regex = ""
        pos = 0
        for m in _PARAM.finditer(pattern):
            regex += re.escape(pattern[pos:m.start()]) + repl(m)
            pos = m.end()
        regex += re.escape(pattern[pos:])
        static = not convs

        def deco(fn):
            self._routes.append((static, re.compile(regex + r"\Z"), convs, {m.upper() for m in methods}, fn))
            return fn
        return deco

    def dispatch(self, method, path):
        path = path.split("?", 1)[0]
        path_matched = False
        for static in (True, False):
            for st, rx, convs, methods, fn in self._routes:
                if st != static:
                    continue
                m = rx.match(path)
                if not m:
                    continue
                path_matched = True
                if method.upper() not in methods:
                    continue
                kwargs = {k: convs[k](v) for k, v in m.groupdict().items()}
                result = fn(**kwargs)
                return result if isinstance(result, Response) else Response(200, result)
        if path_matched:
            return Response(405, "Method Not Allowed")
        return Response(404, "Not Found")

from dataclasses import dataclass, field


@dataclass
class Request:
    method: str
    path: str
    headers: dict = field(default_factory=dict)


@dataclass
class Response:
    status: int = 200
    body: str = ""
    headers: dict = field(default_factory=dict)


class App:
    def __init__(self):
        self._routes = {}
        self._middleware = []

    def route(self, path):
        def deco(fn):
            self._routes[path] = fn
            return fn
        return deco

    def use(self, middleware):
        self._middleware.append(middleware)
        return middleware

    def _endpoint(self, request):
        handler = self._routes.get(request.path)
        if handler is None:
            return Response(404, "not found")
        return handler(request)

    def handle(self, request):
        call = self._endpoint
        for mw in reversed(self._middleware):
            call = (lambda m, nxt: lambda req: m(req, nxt))(mw, call)
        try:
            return call(request)
        except Exception:
            return Response(500, "internal error")

def _save(path, data):

def build_parser():
    p = argparse.ArgumentParser(prog="todo")
    p.add_argument("--db", default="todo.json")

    a.add_argument("--priority", choices=["low", "normal", "high"], default="normal")

    d = sub.add_parser("done")
    d.add_argument("id", type=int)

def main(argv=None):
    args = build_parser().parse_args(argv)
    data = _load(args.db)
    if args.cmd == "add":
        item = {"id": data["next_id"], "text": args.text, "priority": args.priority, "done": False}
        data["next_id"] += 1
        data["items"].append(item)
        _save(args.db, data)
        print(f"added {item['id']}")
    elif args.cmd == "list":
        for it in data["items"]:
            if it["done"] and not args.all:
                continue
            mark = "x" if it["done"] else " "
            suffix = "" if it["priority"] == "normal" else f" ({it['priority']})"
            print(f"{it['id']}. [{mark}] {it['text']}{suffix}")

            if it["id"] == args.id:

                _save(args.db, data)
                print(f"done {args.id}")
                break
        else:
            print(f"error: no item {args.id}", file=sys.stderr)

Write polling-based change detection for a directory tree. `snapshot(root) -> dict[str, tuple[int, int]]` maps every regular file below `root` (recursively, relative POSIX-style paths like `"sub/a.txt"`) to `(st_mtime_ns, st_size)`. `Changes` is a NamedTuple of three sorted lists `added`, `removed`, `modified`. `diff(old, new) -> Changes` compares two snapshots. `Watcher(root)` takes an initial snapshot on construction; `poll() -> Changes` takes a new snapshot, returns the changes since the previous poll and remembers the new snapshot. @@@ reference
import os
from pathlib import Path
from typing import NamedTuple


class Changes(NamedTuple):
    added: list
    removed: list
    modified: list


def snapshot(root):
    root = Path(root)
    out = {}
    for dirpath, _, files in os.walk(root):
        for name in files:
            p = Path(dirpath) / name
            try:
                st = p.stat()
            except FileNotFoundError:
                continue
            out[p.relative_to(root).as_posix()] = (st.st_mtime_ns, st.st_size)
    return out


def diff(old, new):
    added = sorted(set(new) - set(old))
    removed = sorted(set(old) - set(new))
    modified = sorted(k for k in set(old) & set(new) if old[k] != new[k])
    return Changes(added, removed, modified)


class Watcher:
    def __init__(self, root):
        self.root = root
        self._last = snapshot(root)

    def poll(self):
        new = snapshot(self.root)
        ch = diff(self._last, new)
        self._last = new
        return ch

@@@ reference
import hashlib
import os
from pathlib import Path


class ChangeDetector:
    def __init__(self, root):
        self.root = Path(root)
        self.hashes_computed = 0
        self._state = self._scan_state({})

    def _hash(self, path):
        self.hashes_computed += 1
        h = hashlib.sha256()
        with open(path, "rb") as f:
            for chunk in iter(lambda: f.read(65536), b""):
                h.update(chunk)
        return h.hexdigest()

    def _scan_state(self, old):
        state = {}
        for dirpath, _, files in os.walk(self.root):
            for name in files:
                p = Path(dirpath) / name
                rel = p.relative_to(self.root).as_posix()
                st = p.stat()
                sig = (st.st_mtime_ns, st.st_size)
                prev = old.get(rel)
                digest = prev[1] if prev and prev[0] == sig else self._hash(p)
                state[rel] = (sig, digest)
        return state

    def scan(self):
        old = self._state
        new = self._scan_state(old)
        added = set(new) - set(old)
        removed = set(old) - set(new)
        modified = sorted(k for k in set(old) & set(new) if old[k][1] != new[k][1])
        moved = []
        by_hash = {}
        for r in sorted(removed):
            by_hash.setdefault(old[r][1], []).append(r)
        for a in sorted(added):
            cands = by_hash.get(new[a][1])
            if cands:
                r = cands.pop(0)
                moved.append((r, a))
                removed.discard(r)
                added.discard(a)
        self._state = new
        return {"added": sorted(added), "removed": sorted(removed), "modified": modified, "moved": moved}

import csv
import datetime
import io
from decimal import Decimal, InvalidOperation
from typing import NamedTuple


class Report(NamedTuple):
    loaded: int
    rejected: list


def _validate(row, seen):
    try:
        pid = int(row["id"])
    except (TypeError, ValueError):
        raise ValueError("id is not an integer")
    if pid in seen:
        raise ValueError("duplicate id")
    email = (row["email"] or "").strip()
    if email.count("@") != 1 or "." not in email.split("@")[1] or not email.split("@")[0]:
        raise ValueError("invalid email")
    try:
        amount = Decimal(row["amount"])
    except (InvalidOperation, TypeError):
        raise ValueError("invalid amount")
    if not amount.is_finite() or amount < 0 or amount.as_tuple().exponent < -2:
        raise ValueError("invalid amount")
    try:
        date = datetime.date.fromisoformat(row["date"] or "")
    except ValueError:
        raise ValueError("invalid date")
    return pid, email.lower(), int(amount * 100), date.isoformat()


def run_etl(csv_text, conn):
    reader = csv.DictReader(io.StringIO(csv_text))
    good, rejected, seen = [], [], set()
    for line_no, row in enumerate(reader, start=2):
        try:
            rec = _validate(row, seen)
        except ValueError as exc:
            rejected.append((line_no, str(exc)))
            continue
        seen.add(rec[0])
        good.append(rec)
    with conn:
        conn.execute("CREATE TABLE IF NOT EXISTS payments (id INTEGER PRIMARY KEY, email TEXT, "
                     "amount_cents INTEGER, date TEXT)")
        conn.executemany("INSERT INTO payments (id, email, amount_cents, date) VALUES (?, ?, ?, ?)", good)
    return Report(len(good), rejected)

@@@ reference
class RejectRow(Exception):
    def __init__(self, reason):
        super().__init__(reason)
        self.reason = reason


class PipelineError(Exception):
    def __init__(self, index, step, cause):
        super().__init__(f"row {index} failed in step {step!r}: {cause!r}")
        self.index = index
        self.step = step


class Pipeline:
    def __init__(self, steps):
        self.steps = list(steps)

    def run(self, rows):
        outputs, rejects = [], []
        for i, original in enumerate(rows):
            row = dict(original)
            for name, fn in self.steps:
                try:
                    row = fn(row)
                except RejectRow as r:
                    rejects.append({"index": i, "step": name, "reason": r.reason, "row": original})
                    break
                except Exception as exc:
                    raise PipelineError(i, name, exc) from exc
            else:
                outputs.append(row)
        return outputs, rejects

import time

_NEG = object()


class ReadThroughCache:
    def __init__(self, loader, ttl: float, clock=time.monotonic, negative_ttl=None):
        self.loader = loader
        self.ttl = ttl
        self.clock = clock
        self.negative_ttl = negative_ttl
        self._data = {}
        self.stats = {"hits": 0, "misses": 0, "loads": 0}

    def get(self, key):
        now = self.clock()
        entry = self._data.get(key)
        if entry is not None and now < entry[1]:
            self.stats["hits"] += 1
            if entry[0] is _NEG:
                raise KeyError(key)
            return entry[0]
        self.stats["misses"] += 1
        self.stats["loads"] += 1
        try:
            value = self.loader(key)
        except KeyError:
            if self.negative_ttl is not None:
                self._data[key] = (_NEG, now + self.negative_ttl)
            else:
                self._data.pop(key, None)
            raise
        self._data[key] = (value, now + self.ttl)
        return value

    def invalidate(self, key):
        self._data.pop(key, None)

import threading


class _Call:
    def __init__(self):
        self.done = threading.Event()
        self.result = None
        self.error = None
        self.callers = 1


class SingleFlight:
    def __init__(self):
        self._lock = threading.Lock()
        self._calls = {}

    def inflight(self, key):
        with self._lock:
            c = self._calls.get(key)
            return c.callers if c else 0

    def do(self, key, fn):
        with self._lock:
            call = self._calls.get(key)
            if call is not None:
                call.callers += 1
                leader = False
            else:
                call = self._calls[key] = _Call()
                leader = True
        if leader:
            try:
                call.result = fn()
            except BaseException as exc:
                call.error = exc
            finally:
                with self._lock:
                    del self._calls[key]
                call.done.set()
        else:
            call.done.wait()
        if call.error is not None:
            raise call.error
        return call.result

import hashlib
import json
import time


class IdempotencyConflict(Exception):
    pass


def _fingerprint(payload):
    return hashlib.sha256(json.dumps(payload, sort_keys=True).encode()).hexdigest()


class IdempotencyStore:
    def __init__(self, ttl: float = 86400, clock=time.monotonic):
        self.ttl = ttl
        self.clock = clock
        self._entries = {}

    def _get(self, key):
        e = self._entries.get(key)
        if e is not None and self.clock() >= e[2]:
            del self._entries[key]
            return None
        return e

    def handle(self, key, payload, fn):
        fp = _fingerprint(payload)
        e = self._get(key)
        if e is not None:
            if e[0] != fp:
                raise IdempotencyConflict(f"key {key!r} reused with a different payload")
            return e[1]
        response = fn(payload)
        self._entries[key] = (fp, response, self.clock() + self.ttl)
        return response

    def __len__(self):
        now = self.clock()
        return sum(1 for e in self._entries.values() if now < e[2])

import inspect
import typing


class ResolutionError(Exception):
    pass


class Container:
    def __init__(self):
        self._regs = {}
        self._singletons = {}

    def register(self, interface, factory=None, *, singleton=False):
        self._regs[interface] = (factory or interface, singleton)

    def register_instance(self, interface, obj):
        self._regs[interface] = (lambda: obj, True)
        self._singletons[interface] = obj

    def resolve(self, interface):
        return self._resolve(interface, [])

    def _resolve(self, interface, stack):
        if interface in self._singletons:
            return self._singletons[interface]
        if interface not in self._regs:
            raise ResolutionError(f"no registration for {getattr(interface, '__name__', interface)}")
        if interface in stack:
            chain = " -> ".join(t.__name__ for t in stack + [interface])
            raise ResolutionError(f"dependency cycle: {chain}")
        factory, singleton = self._regs[interface]
        target = factory.__init__ if inspect.isclass(factory) else factory
        hints = typing.get_type_hints(target)
        sig = inspect.signature(factory)
        kwargs = {}
        for name, param in sig.parameters.items():
            if param.kind in (param.VAR_POSITIONAL, param.VAR_KEYWORD):
                continue
            dep = hints.get(name)
            if dep is not None and (dep in self._regs or dep in self._singletons):
                kwargs[name] = self._resolve(dep, stack + [interface])
            elif param.default is not param.empty:
                continue
            elif dep is None:
                raise ResolutionError(f"parameter {name!r} of {factory!r} has no annotation")
            else:
                raise ResolutionError(f"no registration for {getattr(dep, '__name__', dep)} "
                                      f"(needed by {getattr(interface, '__name__', interface)})")
        obj = factory(**kwargs)
        if singleton:
            self._singletons[interface] = obj
        return obj

from contextlib import contextmanager


class History:
    def __init__(self, initial, limit=None):
        self._state = initial
        self._undo = []
        self._redo = []
        self.limit = limit
        self._batch_depth = 0
        self._batch_start = None

    @property
    def state(self):
        return self._state

    @property
    def can_undo(self):
        return bool(self._undo)

    @property
    def can_redo(self):
        return bool(self._redo)

    def _record(self, previous):
        self._undo.append(previous)
        if self.limit is not None and len(self._undo) > self.limit:
            del self._undo[0]
        self._redo.clear()

    def push(self, new_state):
        if self._batch_depth == 0:
            self._record(self._state)
        self._state = new_state

    def undo(self):
        if not self._undo:
            raise IndexError("nothing to undo")
        self._redo.append(self._state)
        self._state = self._undo.pop()
        return self._state

    def redo(self):
        if not self._redo:
            raise IndexError("nothing to redo")
        self._undo.append(self._state)
        self._state = self._redo.pop()
        return self._state

    @contextmanager
    def batch(self):
        outer = self._batch_depth == 0
        if outer:
            self._batch_start = self._state
        self._batch_depth += 1
        try:
            yield self
        except BaseException:
            self._batch_depth -= 1
            if outer:
                self._state = self._batch_start
            raise
        self._batch_depth -= 1
        if outer and self._state is not self._batch_start:
            self._record(self._batch_start)

        score = self._best.pop(player)
        self._sorted.pop(bisect.bisect_left(self._sorted, (-score, player)))

def _labels(d):
    if not d:

    return "{" + ",".join(f'{k}="{v}"' for k, v in sorted(d.items())) + "}"

class _Histogram:
    def __init__(self, buckets):
        self.buckets = sorted(buckets)
        self.counts = [0] * len(self.buckets)
        self.sum = 0
        self.count = 0

    def observe(self, v):
        i = bisect.bisect_left(self.buckets, v)
        if i < len(self.buckets):
            self.counts[i] += 1
        self.sum += v
        self.count += 1

    def samples(self, name, labels):
        out, cum = [], 0
        for b, c in zip(self.buckets, self.counts):
            cum += c
            out.append(f"{name}_bucket{_labels({**labels, 'le': format(b, 'g')})} {format(cum, 'g')}")
        out.append(f"{name}_bucket{_labels({**labels, 'le': '+Inf'})} {format(self.count, 'g')}")
        out.append(f"{name}_sum{_labels(labels)} {format(self.sum, 'g')}")
        out.append(f"{name}_count{_labels(labels)} {format(self.count, 'g')}")
        return out


class MetricsRegistry:
    def __init__(self):
        self._types = {}
        self._metrics = {}

    def _get(self, cls, name, labels, *args):
        if self._types.setdefault(name, cls) is not cls:
            raise TypeError(f"{name} is already a {self._types[name].__name__}")
        key = (name, tuple(sorted(labels.items())))
        if key not in self._metrics:
            self._metrics[key] = cls(*args)
        return self._metrics[key]

    def counter(self, name, **labels):
        return self._get(_Counter, name, labels)

    def gauge(self, name, **labels):
        return self._get(_Gauge, name, labels)

    def histogram(self, name, buckets, **labels):
        return self._get(_Histogram, name, labels, buckets)

    def render(self):
        lines = []
        for (name, labels), m in self._metrics.items():
            lines.extend(m.samples(name, dict(labels)))
        return "\n".join(sorted(lines))

import functools
import math
import time
from collections import deque


class RollingStats:
    def __init__(self, window: float, clock=time.monotonic):
        self.window = window
        self.clock = clock
        self._items = deque()

    def _prune(self):
        now = self.clock()
        while self._items and now >= self._items[0][0] + self.window:
            self._items.popleft()

    def record(self, value):
        self._items.append((self.clock(), value))

    def count(self):
        self._prune()
        return len(self._items)

    def mean(self):
        self._prune()
        if not self._items:
            return None
        return sum(v for _, v in self._items) / len(self._items)

    def percentile(self, p):
        if not 0 < p <= 100:
            raise ValueError("p must be in (0, 100]")
        self._prune()
        if not self._items:
            return None
        vals = sorted(v for _, v in self._items)
        return vals[math.ceil(p * len(vals) / 100) - 1]


class timed:
    def __init__(self, stats, clock=time.perf_counter):
        self.stats = stats
        self.clock = clock
        self._starts = []

    def __enter__(self):
        self._starts.append(self.clock())
        return self

    def __exit__(self, *exc):
        self.stats.record(self.clock() - self._starts.pop())
        return False

    def __call__(self, fn):
        @functools.wraps(fn)
        def wrapper(*a, **kw):
            with self:
                return fn(*a, **kw)
        return wrapper

import abc
import time
from dataclasses import dataclass, replace


class LockLost(Exception):
    pass


class StaleToken(Exception):
    pass


@dataclass(frozen=True)
class Lease:
    name: str
    owner: str
    token: int
    expires_at: float


class LockService(abc.ABC):
    @abc.abstractmethod
    def acquire(self, name, owner, ttl):
        ...

    @abc.abstractmethod
    def renew(self, lease, ttl):
        ...

    @abc.abstractmethod
    def release(self, lease):
        ...


class InMemoryLockService(LockService):
    def __init__(self, clock=time.monotonic):
        self.clock = clock
        self._held = {}
        self._tokens = {}

    def _current(self, name):
        lease = self._held.get(name)
        if lease is not None and self.clock() >= lease.expires_at:
            del self._held[name]
            return None
        return lease

    def acquire(self, name, owner, ttl):
        if self._current(name) is not None:
            return None
        token = self._tokens.get(name, 0) + 1
        self._tokens[name] = token
        lease = Lease(name, owner, token, self.clock() + ttl)
        self._held[name] = lease
        return lease

    def renew(self, lease, ttl):
        cur = self._current(lease.name)
        if cur is None or cur.token != lease.token:
            raise LockLost(lease.name)
        new = replace(cur, expires_at=self.clock() + ttl)
        self._held[lease.name] = new
        return new

    def release(self, lease):
        cur = self._current(lease.name)
        if cur is None or cur.token != lease.token:
            return False
        del self._held[lease.name]
        return True


class FencedResource:
    def __init__(self):
        self.values = []
        self._highest = 0

    def write(self, token, value):
        if token < self._highest:
            raise StaleToken(f"token {token} < {self._highest}")
        self._highest = token
        self.values.append(value)

import time


class TransientError(Exception):
    pass


class PermanentError(Exception):
    pass


class CircuitOpen(Exception):
    pass


class ApiClient:
    def __init__(self, transport, *, max_retries=2, base_delay=0.1, failure_threshold=3,
                 reset_timeout=30.0, clock=time.monotonic, sleep=time.sleep):
        self.transport = transport
        self.max_retries = max_retries
        self.base_delay = base_delay
        self.failure_threshold = failure_threshold
        self.reset_timeout = reset_timeout
        self.clock = clock
        self.sleep = sleep
        self._failures = 0
        self._opened_at = None

    @property
    def state(self):
        if self._opened_at is None:
            return "closed"
        if self.clock() - self._opened_at >= self.reset_timeout:
            return "half_open"
        return "open"

    def get(self, path):
        state = self.state
        if state == "open":
            raise CircuitOpen(path)
        retries = 0 if state == "half_open" else self.max_retries
        for i in range(retries + 1):
            try:
                result = self.transport(path)
            except PermanentError:
                if state == "half_open":
                    self._opened_at = None
                    self._failures = 0
                raise
            except TransientError:
                if i == retries:
                    self._failures += 1
                    if state == "half_open" or self._failures >= self.failure_threshold:
                        self._opened_at = self.clock()
                    raise
                self.sleep(self.base_delay * 2 ** i)
            else:
                self._failures = 0
                self._opened_at = None
                return result
