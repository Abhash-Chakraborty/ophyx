import secrets
import time


class FakeClock:
    def __init__(self, start: float = 0.0):
        self.now = start

    def __call__(self):
        return self.now

    def advance(self, seconds):
        if seconds < 0:
            raise ValueError("time cannot go backwards")
        self.now += seconds


class SessionStore:
    def __init__(self, ttl: float, clock=time.monotonic):
        self.ttl = ttl
        self._clock = clock
        self._sessions = {}

    def create(self, user):
        token = secrets.token_hex(16)
        self._sessions[token] = [user, self._clock()]
        return token

    def _expired(self, last_seen, now):
        return now - last_seen >= self.ttl

    def get(self, token):
        entry = self._sessions.get(token)
        if entry is None:
            return None
        now = self._clock()
        if self._expired(entry[1], now):
            del self._sessions[token]
            return None
        entry[1] = now
        return entry[0]

    def purge(self):
        now = self._clock()
        dead = [t for t, (_, seen) in self._sessions.items() if self._expired(seen, now)]
        for t in dead:
            del self._sessions[t]
        return len(dead)

    def __len__(self):
        return len(self._sessions)

import itertools


def rle_encode(s: str) -> list[tuple[str, int]]:
    return [(ch, len(list(group))) for ch, group in itertools.groupby(s)]


def rle_decode(pairs) -> str:
    return "".join(ch * n for ch, n in pairs)


def random_strings(rng, n, alphabet="ab", max_len=12):
    for _ in range(n):
        yield "".join(rng.choice(alphabet) for _ in range(rng.randint(0, max_len)))


def _fails(prop, value):
    try:
        return prop(value) is False
    except AssertionError:
        return True


def shrink(value, prop):
    changed = True
    while changed:
        changed = False
        for i in range(len(value)):
            candidate = value[:i] + value[i + 1:]
            if _fails(prop, candidate):
                value, changed = candidate, True
                break
    return value


def for_all(prop, inputs):
    for value in inputs:
        if _fails(prop, value):
            shrunk = shrink(value, prop)
            raise AssertionError(f"falsified by {shrunk!r} (original {value!r})")

import re

_TEMP = re.compile(r"\s*(-?\d+(?:\.\d+)?)\s*([cCfF])\s*")


def c_to_f(c: float) -> float:
    """Convert Celsius to Fahrenheit.

    >>> c_to_f(100)
    212.0
    >>> c_to_f(-40)
    -40.0
    >>> c_to_f(36.6)
    97.88
    """
    return round(c * 9 / 5 + 32, 2)


def f_to_c(f: float) -> float:
    """Convert Fahrenheit to Celsius.

    >>> f_to_c(32)
    0.0
    >>> f_to_c(98.6)
    37.0
    """
    return round((f - 32) * 5 / 9, 2)


def parse_temp(text: str) -> float:
    """Parse a temperature with a unit and return Celsius.

    >>> parse_temp("21.5C")
    21.5
    >>> parse_temp(" 70F ")
    21.11
    >>> parse_temp("-3 c")
    -3.0
    >>> parse_temp("warm")
    Traceback (most recent call last):
        ...
    ValueError: not a temperature: 'warm'
    """
    m = _TEMP.fullmatch(text)
    if not m:
        raise ValueError(f"not a temperature: {text!r}")
    value = float(m.group(1))
    return round(value, 2) if m.group(2) in "cC" else f_to_c(value)
