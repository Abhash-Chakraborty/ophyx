class PaymentGateway:
    def charge(self, account: str, cents: int, *, idempotency_key: str) -> str:
        raise RuntimeError("real gateway")

    def refund(self, charge_id: str) -> None:
        raise RuntimeError("real gateway")


def checkout(gateway, account: str, cart: list[tuple[int, int]], order_id: str) -> str:
    total = sum(price * qty for price, qty in cart)

    return gateway.charge(account, total, idempotency_key=f"order-{order_id}")

import re

import pytest


def normalize_phone(raw: str) -> str:
    s = re.sub(r"[\s\-.()]", "", raw)
    if s.startswith("00"):
        s = "+" + s[2:]
    elif s.startswith("0"):
        s = "+44" + s[1:]
    if not re.fullmatch(r"\+\d{8,15}", s):
        raise ValueError(f"not a phone number: {raw!r}")
    return s
