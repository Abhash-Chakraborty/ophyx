def parse_duration(text: str) -> int:
    m = _RE.fullmatch(text)
    if not m or not any(m.groups()):
        raise ValueError(f"bad duration {text!r}")
    return sum(int(g) * UNITS[u] for g, u in zip(m.groups(), "dhms") if g is not None)

from collections import deque
from collections.abc import Iterable, Iterator
from typing import Generic, TypeVar

T = TypeVar("T")


class RingBuffer(Generic[T]):
    def __init__(self, capacity: int) -> None:
        if capacity < 1:
            raise ValueError("capacity must be at least 1")
        self._items: deque[T] = deque()
        self.capacity = capacity

    def push(self, item: T) -> T | None:
        evicted = self._items.popleft() if len(self._items) == self.capacity else None
        self._items.append(item)
        return evicted

    def __iter__(self) -> Iterator[T]:
        return iter(list(self._items))

    def __len__(self) -> int:
        return len(self._items)

    def latest(self) -> T:
        if not self._items:
            raise IndexError("empty buffer")
        return self._items[-1]

    for item in items:
        return item
    return default

from collections.abc import Callable, Iterable

type Matrix[T] = list[list[T]]


def transpose[T](m: Matrix[T]) -> Matrix[T]:
    if not m:
        return []
    width = len(m[0])
    if any(len(row) != width for row in m):
        raise ValueError("ragged matrix")
    return [[row[i] for row in m] for i in range(width)]


class Grouped[K, V]:
    def __init__(self) -> None:
        self._groups: dict[K, list[V]] = {}

    def add(self, key: K, value: V) -> None:
        self._groups.setdefault(key, []).append(value)

    def get(self, key: K) -> list[V]:
        return list(self._groups.get(key, []))

    def keys(self) -> list[K]:
        return list(self._groups)


def group_by[K, V](items: Iterable[V], key: Callable[[V], K]) -> Grouped[K, V]:
    g: Grouped[K, V] = Grouped()
    for item in items:
        g.add(key(item), item)
    return g


def clamp[N: (int, float)](value: N, low: N, high: N) -> N:
    if low > high:
        raise ValueError("low > high")
    return max(low, min(value, high))

from collections.abc import Iterable
from typing import Protocol, runtime_checkable


@runtime_checkable
class Readable(Protocol):
    def read(self, size: int = -1) -> bytes: ...


@runtime_checkable
class Closeable(Protocol):
    def close(self) -> None: ...


def drain(source, chunk_size: int = 4096) -> bytes:
    if isinstance(source, (bytes, bytearray, memoryview)):
        return bytes(source)
    try:
        if isinstance(source, Readable):
            parts = []
            while chunk := source.read(chunk_size):
                parts.append(chunk)
            return b"".join(parts)
        if isinstance(source, Iterable):
            return b"".join(bytes(c) for c in source)
        raise TypeError(f"cannot drain {type(source).__name__}")
    finally:
        if isinstance(source, Closeable):
            source.close()

import functools
import inspect
from collections.abc import Callable
from typing import Concatenate, ParamSpec, TypeVar

P = ParamSpec("P")
R = TypeVar("R")
T = TypeVar("T")


def timed(clock: Callable[[], float], sink: Callable[[str, float], None]) -> Callable[[Callable[P, R]], Callable[P, R]]:
    def decorator(fn: Callable[P, R]) -> Callable[P, R]:
        @functools.wraps(fn)
        def wrapper(*args: P.args, **kwargs: P.kwargs) -> R:
            start = clock()
            try:
                return fn(*args, **kwargs)
            finally:
                sink(fn.__qualname__, clock() - start)
        return wrapper
    return decorator

        def wrapper(*args: P.args, **kwargs: P.kwargs) -> R:
            return fn(value, *args, **kwargs)
        sig = inspect.signature(fn)
        wrapper.__signature__ = sig.replace(parameters=list(sig.parameters.values())[1:])
        return wrapper
    return decorator

_MONEY = re.compile(r"(-?)(\d+)(?:\.(\d{1,2}))?")


def to_cents(amount: str) -> Cents:
    m = _MONEY.fullmatch(amount)
    if not m:
        raise ValueError(f"bad amount {amount!r}")
    sign, whole, frac = m.groups()
    value = int(whole) * 100 + int((frac or "0").ljust(2, "0"))
    return Cents(-value if sign else value)

import copy
from typing import Self


class Query:
    def __init__(self, table: str) -> None:
        self.table = table
        self._where: list[str] = []
        self._order: str | None = None
        self._limit: int | None = None
        self._offset: int | None = None

    def _copy(self) -> Self:
        new = copy.copy(self)
        new._where = list(self._where)
        return new

    def where(self, condition: str) -> Self:
        new = self._copy()
        new._where.append(condition)
        return new

    def order_by(self, column: str, *, desc: bool = False) -> Self:
        new = self._copy()
        new._order = f"{column} DESC" if desc else column
        return new

    def limit(self, n: int) -> Self:
        if n < 0:
            raise ValueError("negative limit")
        new = self._copy()
        new._limit = n
        return new

    def build(self) -> str:
        sql = f"SELECT * FROM {self.table}"
        if self._where:
            sql += " WHERE " + " AND ".join(self._where)
        if self._order:
            sql += f" ORDER BY {self._order}"
        if self._limit is not None:
            sql += f" LIMIT {self._limit}"
        if self._offset is not None:
            sql += f" OFFSET {self._offset}"
        return sql


class PagedQuery(Query):
    def page(self, number: int, size: int) -> Self:
        if number < 1 or size < 1:
            raise ValueError("page number and size must be at least 1")
        new = self.limit(size)
        new._offset = (number - 1) * size
        return new

from typing import TypeVar

K = TypeVar("K")
V = TypeVar("V")
T = TypeVar("T")


def invert(mapping: Mapping[K, V]) -> dict[V, list[K]]:
    out: dict[V, list[K]] = {}
    for k, v in mapping.items():
        out.setdefault(v, []).append(k)
    return out


def windows(seq: Sequence[T], size: int) -> list[Sequence[T]]:
    if not isinstance(seq, Sequence):
        raise TypeError(f"expected a Sequence, got {type(seq).__name__}")
    if size < 1:
        raise ValueError("size must be at least 1")
    return [seq[i:i + size] for i in range(len(seq) - size + 1)]

    total, n = 0.0, 0
    for v in values:
        total += v
        n += 1
        yield total / n

from collections.abc import Mapping

Scalar = str | int | float | bool | None


def flatten_config(cfg: Mapping[str, object], prefix: str = "") -> dict[str, Scalar]:
    out: dict[str, Scalar] = {}

    def walk(value: object, path: str) -> None:
        if isinstance(value, Mapping):
            for k, v in value.items():
                if not isinstance(k, str):
                    raise TypeError(f"non-str key {k!r} under {path or '<root>'}")
                walk(v, f"{path}.{k}" if path else k)
        elif isinstance(value, (list, tuple)):
            for i, v in enumerate(value):
                walk(v, f"{path}.{i}" if path else str(i))
        elif value is None or isinstance(value, (str, int, float, bool)):
            out[path] = value
        else:
            raise TypeError(f"unsupported value of type {type(value).__name__} at {path}")

    walk(cfg, prefix)
    return out


def get_int(flat: Mapping[str, Scalar], key: str, default: int) -> int:
    value = flat.get(key)
    if value is None:
        return default
    if isinstance(value, bool):
        raise TypeError(f"{key} is a bool")
    if isinstance(value, int):
        return value
    if isinstance(value, float):
        if not value.is_integer():
            raise ValueError(f"{key} is not integral: {value}")
        return int(value)
    return int(value.strip())
