import bisect
from collections.abc import Iterable
from typing import Any, Protocol, TypeVar


class SupportsLessThan(Protocol):
    def __lt__(self, other: Any, /) -> bool: ...


CT = TypeVar("CT", bound=SupportsLessThan)


def top_k(items: Iterable[CT], k: int) -> list[CT]:
    if k <= 0:
        return []
    return sorted(items, reverse=True)[:k]


def insert_sorted(seq: list[CT], item: CT) -> int:
    i = bisect.bisect_right(seq, item)
    seq.insert(i, item)
    return i

Write a typed decorator factory `retry(times: int, *, on: tuple[type[BaseException], ...] = (Exception,), sleep: Callable[[float], None] = time.sleep, delay: float = 0.0) -> Callable[[Callable[P, R]], Callable[P, R]]`, with `P = ParamSpec("P")` and `R = TypeVar("R")`. The returned decorator wraps a function so that a call raising one of the exception types in `on` is retried, up to `times` attempts in total, calling `sleep(delay)` between attempts (not after the last); exceptions not in `on` propagate immediately, and after the last failed attempt the last exception propagates. `times < 1` raises ValueError when `retry` is called. The wrapper is annotated `(*args: P.args, **kwargs: P.kwargs) -> R`, uses `functools.wraps`, and must work on methods too (self simply flows through `P`).
@@@ reference
import functools
import time
from collections.abc import Callable
from typing import ParamSpec, TypeVar

P = ParamSpec("P")
R = TypeVar("R")


def retry(times: int, *, on: tuple[type[BaseException], ...] = (Exception,),
          sleep: Callable[[float], None] = time.sleep, delay: float = 0.0) -> Callable[[Callable[P, R]], Callable[P, R]]:
    if times < 1:
        raise ValueError("times must be at least 1")

    def decorator(fn: Callable[P, R]) -> Callable[P, R]:
        @functools.wraps(fn)
        def wrapper(*args: P.args, **kwargs: P.kwargs) -> R:
            for attempt in range(1, times + 1):
                try:
                    return fn(*args, **kwargs)
                except on:
                    if attempt == times:
                        raise
                    sleep(delay)
            raise AssertionError("unreachable")
        return wrapper
    return decorator

import bisect
from collections.abc import Iterable, Sequence
from dataclasses import dataclass
from typing import overload


@dataclass(frozen=True)
class Event:
    at: int
    name: str


class Timeline(Sequence[Event]):
    def __init__(self, events: Iterable[Event] = ()) -> None:
        self._events = sorted(events, key=lambda e: e.at)
        self._times = [e.at for e in self._events]

    @overload
    def __getitem__(self, index: int) -> Event: ...
    @overload
    def __getitem__(self, index: slice) -> "Timeline": ...
    def __getitem__(self, index):
        if isinstance(index, slice):
            return Timeline(self._events[index])
        return self._events[index]

    def __len__(self) -> int:
        return len(self._events)

    def __repr__(self) -> str:
        return f"Timeline({self._events!r})"

    def between(self, start: int, end: int) -> "Timeline":
        lo = bisect.bisect_left(self._times, start)
        hi = bisect.bisect_left(self._times, end)
        return Timeline(self._events[lo:hi])
